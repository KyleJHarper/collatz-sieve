175 static inline level_t st_get_level_by_node_value(const T& value) {
176 level_t level = 0;
177 if constexpr(FixedWidthIntegral<T>) {
178 level = st_floor_log2(value + _offset) + 1;
179 } else if constexpr(GMPIntegral<T>) {
180 // Adding 1 is a waste of alloc here, so use a scratch variable.
181 static thread_local mpz_class junk = 0;
182 mpz_add_ui(junk.get_mpz_t(), value.get_mpz_t(), _offset);
183 level = st_floor_log2(junk) + 1;
184 }
185 return level;

225 static inline level_t st_max_level_of_type() {
226 // Max level is scaled with tree size (diadic) but must respect implementation limits, ergo:
227 // Bit width
228 // Minus 1 because the last node on 2^bit level overflows due to counting starting at 0.
229 // Minus 1 if the type is signed.
230 size_t bits = sizeof(T) * 8;
231 level_t max_level = bits - 1 - (std::is_signed_v<T> ? 1 : 0);
232 return max_level;

324 static inline T st_node_value_by_position_and_level(const T& position, level_t level) {
325 static thread_local T lift;
326 static thread_local T index;
327
328 // Lift: 2^(L-1)
329 if constexpr(FixedWidthIntegral<T>) {
330 lift = Exponents::get_power_of_two<T>(level - 1);
331 lift = T(1) << (level - 1);
332 } else if constexpr(GMPIntegral<T>) {
333 mpz_pow_ui(lift.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - 1));
334 }
335
336 // New Position: bit_reverse_L(pos - 1)
337 if constexpr(FixedWidthIntegral<T>) {
338 index = position - 1;
339 } else if constexpr(GMPIntegral<T>) {
340 mpz_sub_ui(index.get_mpz_t(), position.get_mpz_t(), 1);
341 }
342
343 // Reverse the low bits. For GMP, use an out param to avoid alloc (yes, even with TLS).
344 if constexpr(FixedWidthIntegral<T>) {
345 index = Bit::st_reverse_low_bits(index, (level - 1));
346 } else if constexpr (GMPIntegral<T>) {
347 static thread_local T index_copy;
348 index_copy = index;
349 Bit::st_reverse_low_bits(index_copy, (level - 1), index);
350 }
351
352 // Return
353 if constexpr(FixedWidthIntegral<T>) {
354 return lift + index - _offset;
355 } else if constexpr (GMPIntegral<T>) {
356 static thread_local T result;
357 mpz_set(result.get_mpz_t(), lift.get_mpz_t());
358 mpz_add(result.get_mpz_t(), result.get_mpz_t(), index.get_mpz_t());
359 mpz_sub_ui(result.get_mpz_t(), result.get_mpz_t(), _offset);
360 return result;
361 }

447 static inline T st_node_position(const T& value) {
448 level_t level = BinaryTreeMath<T>::st_get_level_by_node_value(value);
449 T value_plus_offset = value + _offset;
450 T low_bits;
451 static thread_local T position;
452
453 // Get lower bits.
454 if constexpr(FixedWidthIntegral<T>) {
455 low_bits = value_plus_offset & ((T(1) << (level - 1)) - 1);
456 } else if constexpr(GMPIntegral<T>) {
457 mpz_class mask = (mpz_class(1) << (level - 1)) - 1;
458 low_bits = value_plus_offset & mask;
459 }
460
461 // Get the position now.
462 position = Bit::st_reverse_low_bits(low_bits, level - 1);
463 if constexpr(FixedWidthIntegral<T>) {
464 position += 1;
465 } else if constexpr(GMPIntegral<T>) {
466 mpz_add_ui(position.get_mpz_t(), position.get_mpz_t(), 1);
467 }
468
469 return position;
470 }

496 [[deprecated("Calculates positive S1 series correctly, but the whole approach is superseded by bitwise operations")]]
497 static inline T st_s1_summation(const T& position, level_t level) {
498 static thread_local T value = 1;
499 static thread_local T quantity;
500 static thread_local T scale;
501
502 // Leave unless the level is high enough to have any quantity. AKA: L > 1
503 if (level < 2) { return T(0); }
504
505 // Quantity: ceil((pos - 1) / 2)
506 // We can simplify because: ceil((P - 1) / 2) ==> ((P - 1)+1)/2 ==> pos / 2
507 quantity = position / 2;
508
509 // Scale: 2^(L-2)
510 if constexpr(FixedWidthIntegral<T>) {
511 scale = T(1) << (level - 2);
512 } else if constexpr(GMPIntegral<T>) {
513 mpz_pow_ui(scale.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - 2));
514 }
515
516 // Result of multiplication should be correctly typed.
517 return value * quantity * scale;
518 }

547 [[deprecated("Calculates negative S2 series correctly, but the whole approach is superseded by bitwise operations")]]
548 static inline T st_s2_summation(const T& position, level_t level) {
549 static thread_local T quantity;
550 static thread_local T value;
551 static thread_local T scale;
552 static thread_local T summation;
553
554 // Summation Loop
555 summation = 0;
556 for(level_t a=3; a<=level; a++) {
557 // Quantity: ceil((P - 2^(a-2)) / 2^(a-1))
558 // Simplification exists because ceiling behaves this way:
559 // ceil(x/y) ==> (x + y - 1) / y
560 //
561 // Currently the formula contains:
562 // x (numerator) : P - 2^(a-2)
563 // y (denominator): 2^(a-1)
564 // ceiling(x/y) : (x + y - 1) / y
565 //
566 // With ceiling applied it looks like this:
567 // (P - 2^(a-2) + 2^(a-1) - 1) / 2^(a-1)
568 //
569 // The first subtraction can be rewritten as addition of a negative:
570 // (P + (-2^(a-2)) + 2^(a-1) - 1) / 2^(a-1)
571 //
572 // Now it isolate the powers of two sections in the numerator to see how they simplify:
573 // -2^(a-2) + 2^(a-1)
574 // ==> -2^(a-2) + 2 * 2^(a-2)
575 // ==> -2^(a-2) + 2^(a-2) + 2^(a-2)
576 // ==> 2^(a-2)
577 //
578 // The full numerator is now simplified to:
579 // P + 2^(a-2) - 1
580 //
581 // The full expression is now just:
582 // (P + 2^(a-2) - 1) / 2^(a-1)
583 //
584 // Some arithmetic and bit shifting covers the rest:
585 // (P + (T(1) << (a - 2)) - 1) >> (a - 1)
586 quantity = (position + (T(1) << (a - 2)) - 1) >> (a - 1);
587
588 // Value: -(2^(a-1) - 3)
589 // Remember to use a positive value, even though it's technically a decreasing summation (negative).
590 if constexpr(FixedWidthIntegral<T>) {
591 value = (T(1) << (a - 1)) - 3;
592 } else if constexpr(GMPIntegral<T>) {
593 mpz_pow_ui(value.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (a - 1));
594 mpz_sub_ui(value.get_mpz_t(), value.get_mpz_t(), 3);
595 }
596
597 // Scale: 2^(L-a)
598 if constexpr(FixedWidthIntegral<T>) {
599 scale = T(1) << (level - a);
600 } else if constexpr(GMPIntegral<T>) {
601 mpz_pow_ui(scale.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - a));
602 }
603
604 // Add to Summation
605 summation += (quantity * value * scale);
606 }
607
608 // Return it.
609 return summation;
610 }

660 static inline size_t st_floor_log2(const T& val) {
661 if (val == 0) { return 0; }
662 size_t result = 0;
663 if constexpr(FixedWidthIntegral<T>) {
664 if constexpr(sizeof(T) < 64) {
665 result = 63 - __builtin_clzll(val);
666 } else if constexpr(sizeof(T) == 128) {
667 if (val >> 64) {
668 result = 127 - __builtin_clzll((uint64_t)(val >> 64));
669 } else {
670 result = 63 - __builtin_clzll((uint64_t)val);
671 }
672 }
673 } else if constexpr(GMPIntegral<T>) {
674 result = mpz_sizeinbase(val.get_mpz_t(), 2) - 1;
675 }
676 return result;
677 }
