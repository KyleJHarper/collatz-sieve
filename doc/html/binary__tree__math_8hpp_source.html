179 static inline level_t st_get_level_by_node_value(const T& value) {
180 level_t level = 0;

229 static inline level_t st_max_level_of_type() {
230 // Max level is scaled with tree size (diadic) but must respect implementation limits, ergo:
231 // Bit width
232 // Minus 1 because the last node on 2^bit level overflows due to counting starting at 0.
233 // Minus 1 if the type is signed.
234 if (GMPIntegral<T>) {
235 return std::numeric_limits<level_t>::max();
236 }
237 size_t bits = sizeof(T) * 8;
238 level_t max_level = bits - 1 - (std::numeric_limits<T>::is_signed ? 1 : 0);
239 return max_level;

338 static inline T st_node_value_by_position_and_level(const T& position, level_t level) {
339 static thread_local T lift;
340 static thread_local T index;
341
342 // Lift: 2^(L-1)
343 if constexpr(FixedWidthIntegral<T>) {
344 lift = Exponents::get_power_of_two<T>(level - 1);
345 lift = T(1) << (level - 1);
346 } else if constexpr(GMPIntegral<T>) {
347 mpz_pow_ui(lift.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - 1));
348 }
349
350 // New Position: bit_reverse_L(pos - 1)
351 if constexpr(FixedWidthIntegral<T>) {
352 index = position - 1;
353 } else if constexpr(GMPIntegral<T>) {
354 mpz_sub_ui(index.get_mpz_t(), position.get_mpz_t(), 1);
355 }
356
357 // Reverse the low bits. For GMP, use an out param to avoid alloc (yes, even with TLS).
358 if constexpr(FixedWidthIntegral<T>) {
359 index = Bit::st_reverse_low_bits(index, (level - 1));
360 } else if constexpr (GMPIntegral<T>) {
361 static thread_local T index_copy;
362 index_copy = index;
363 Bit::st_reverse_low_bits(index_copy, (level - 1), index);
364 }
365
366 // Return
367 if constexpr(FixedWidthIntegral<T>) {
368 return lift + index - _offset;
369 } else if constexpr (GMPIntegral<T>) {
370 static thread_local T result;
371 mpz_set(result.get_mpz_t(), lift.get_mpz_t());
372 mpz_add(result.get_mpz_t(), result.get_mpz_t(), index.get_mpz_t());
373 mpz_sub_ui(result.get_mpz_t(), result.get_mpz_t(), _offset);
374 return result;
375 }

461 static inline T st_node_position(const T& value) {
462 level_t level = BinaryTreeMath<T>::st_get_level_by_node_value(value);
463 T value_plus_offset = value + _offset;
464 T low_bits;
465 static thread_local T position;
466
467 // Get lower bits.
468 if constexpr(FixedWidthIntegral<T>) {
469 low_bits = value_plus_offset & ((T(1) << (level - 1)) - 1);
470 } else if constexpr(GMPIntegral<T>) {
471 mpz_class mask = (mpz_class(1) << (level - 1)) - 1;
472 low_bits = value_plus_offset & mask;
473 }
474
475 // Get the position now.
476 position = Bit::st_reverse_low_bits(low_bits, level - 1);
477 if constexpr(FixedWidthIntegral<T>) {
478 position += 1;
479 } else if constexpr(GMPIntegral<T>) {
480 mpz_add_ui(position.get_mpz_t(), position.get_mpz_t(), 1);
481 }
482
483 return position;
484 }

510 [[deprecated("Calculates positive S1 series correctly, but the whole approach is superseded by bitwise operations")]]
511 static inline T st_s1_summation(const T& position, level_t level) {
512 static thread_local T value = 1;
513 static thread_local T quantity;
514 static thread_local T scale;
515
516 // Leave unless the level is high enough to have any quantity. AKA: L > 1
517 if (level < 2) { return T(0); }
518
519 // Quantity: ceil((pos - 1) / 2)
520 // We can simplify because: ceil((P - 1) / 2) ==> ((P - 1)+1)/2 ==> pos / 2
521 quantity = position / 2;
522
523 // Scale: 2^(L-2)
524 if constexpr(FixedWidthIntegral<T>) {
525 scale = T(1) << (level - 2);
526 } else if constexpr(GMPIntegral<T>) {
527 mpz_pow_ui(scale.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - 2));
528 }
529
530 // Result of multiplication should be correctly typed.
531 return value * quantity * scale;
532 }

561 [[deprecated("Calculates negative S2 series correctly, but the whole approach is superseded by bitwise operations")]]
562 static inline T st_s2_summation(const T& position, level_t level) {
563 static thread_local T quantity;
564 static thread_local T value;
565 static thread_local T scale;
566 static thread_local T summation;
567
568 // Summation Loop
569 summation = 0;
570 for(level_t a=3; a<=level; a++) {
571 // Quantity: ceil((P - 2^(a-2)) / 2^(a-1))
572 // Simplification exists because ceiling behaves this way:
573 // ceil(x/y) ==> (x + y - 1) / y
574 //
575 // Currently the formula contains:
576 // x (numerator) : P - 2^(a-2)
577 // y (denominator): 2^(a-1)
578 // ceiling(x/y) : (x + y - 1) / y
579 //
580 // With ceiling applied it looks like this:
581 // (P - 2^(a-2) + 2^(a-1) - 1) / 2^(a-1)
582 //
583 // The first subtraction can be rewritten as addition of a negative:
584 // (P + (-2^(a-2)) + 2^(a-1) - 1) / 2^(a-1)
585 //
586 // Now it isolate the powers of two sections in the numerator to see how they simplify:
587 // -2^(a-2) + 2^(a-1)
588 // ==> -2^(a-2) + 2 * 2^(a-2)
589 // ==> -2^(a-2) + 2^(a-2) + 2^(a-2)
590 // ==> 2^(a-2)
591 //
592 // The full numerator is now simplified to:
593 // P + 2^(a-2) - 1
594 //
595 // The full expression is now just:
596 // (P + 2^(a-2) - 1) / 2^(a-1)
597 //
598 // Some arithmetic and bit shifting covers the rest:
599 // (P + (T(1) << (a - 2)) - 1) >> (a - 1)
600 quantity = (position + (T(1) << (a - 2)) - 1) >> (a - 1);
601
602 // Value: -(2^(a-1) - 3)
603 // Remember to use a positive value, even though it's technically a decreasing summation (negative).
604 if constexpr(FixedWidthIntegral<T>) {
605 value = (T(1) << (a - 1)) - 3;
606 } else if constexpr(GMPIntegral<T>) {
607 mpz_pow_ui(value.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (a - 1));
608 mpz_sub_ui(value.get_mpz_t(), value.get_mpz_t(), 3);
609 }
610
611 // Scale: 2^(L-a)
612 if constexpr(FixedWidthIntegral<T>) {
613 scale = T(1) << (level - a);
614 } else if constexpr(GMPIntegral<T>) {
615 mpz_pow_ui(scale.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - a));
616 }
617
618 // Add to Summation
619 summation += (quantity * value * scale);
620 }
621
622 // Return it.
623 return summation;
624 }

674 static inline size_t st_floor_log2(const T& val) {
675 if (val == 0) { return 0; }
676 size_t result = 0;
677 if constexpr(FixedWidthIntegral<T>) {
678 if constexpr(sizeof(T) < 64) {
679 result = 63 - __builtin_clzll(val);
680 } else if constexpr(sizeof(T) == 128) {
681 if (val >> 64) {
682 result = 127 - __builtin_clzll((uint64_t)(val >> 64));
683 } else {
684 result = 63 - __builtin_clzll((uint64_t)val);
685 }
686 }
687 } else if constexpr(GMPIntegral<T>) {
688 result = mpz_sizeinbase(val.get_mpz_t(), 2) - 1;
689 }
690 return result;
691 }
