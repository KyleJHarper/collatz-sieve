181 static inline level_t st_get_level_by_node_value(const T& value) {
182 level_t level = 0;

234 // Max level is scaled with tree size (diadic) but must respect implementation limits, ergo:

334 static inline T st_node_value_by_position_and_level(const T& position, level_t level) {
335 static thread_local T lift;
336 static thread_local T index;
337
338 // Lift: 2^(L-1)
339 if constexpr(BuiltinIntegral<T>) {
340 lift = Exponents::get_power_of_two<T>(level - 1);
341 lift = T(1) << (level - 1);
342 } else if constexpr(GMPIntegral<T>) {
343 mpz_pow_ui(lift.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - 1));
344 } else {
345 throw std::logic_error("Unknown type.");
346 }
347
348 // New Position: bit_reverse_L(pos - 1)
349 if constexpr(BuiltinIntegral<T>) {
350 index = position - 1;
351 } else if constexpr(GMPIntegral<T>) {
352 mpz_sub_ui(index.get_mpz_t(), position.get_mpz_t(), 1);
353 } else {
354 throw std::logic_error("Unknown type.");
355 }
356
357 // Reverse the low bits. For GMP, use an out param to avoid alloc (yes, even with TLS).
358 if constexpr(BuiltinIntegral<T>) {
359 index = st_reverse_low_bits(index, (level - 1));
360 } else {
361 static thread_local T index_copy;
362 index_copy = index;
363 st_reverse_low_bits(index_copy, (level - 1), index);
364 }
365
366 // Return
367 if constexpr(BuiltinIntegral<T>) {
368 return lift + index - _offset;
369 } else {
370 static thread_local T result;
371 mpz_set(result.get_mpz_t(), lift.get_mpz_t());
372 mpz_add(result.get_mpz_t(), result.get_mpz_t(), index.get_mpz_t());
373 mpz_sub_ui(result.get_mpz_t(), result.get_mpz_t(), _offset);
374 return result;
375 }

465 static inline T st_node_position(const T& value) {
466 level_t level = BinaryTreeMath<T>::st_get_level_by_node_value(value);
467 T value_plus_offset = value + _offset;
468 T low_bits;
469 static thread_local T position;
470
471 // Get lower bits.
472 if constexpr(BuiltinIntegral<T>) {
473 low_bits = value_plus_offset & ((T(1) << (level - 1)) - 1);
474 } else if constexpr(GMPIntegral<T>) {
475 mpz_class mask = (mpz_class(1) << (level - 1)) - 1;
476 low_bits = value_plus_offset & mask;
477 } else {
478 throw std::logic_error("Unknown type.");
479 }
480
481 // Get the position now.
482 position = BinaryTreeMath<T>::st_reverse_low_bits(low_bits, level - 1);
483 if constexpr(BuiltinIntegral<T>) {
484 position += 1;
485 } else if constexpr(GMPIntegral<T>) {
486 mpz_add_ui(position.get_mpz_t(), position.get_mpz_t(), 1);
487 } else {
488 throw std::logic_error("Unknown type.");
489 }
490
491 return position;
492 }

518 [[deprecated("Calculates positive S1 series correctly, but the whole approach is superseded by bitwise operations")]]
519 static inline T st_s1_summation(const T& position, level_t level) {
520 static thread_local T value = 1;
521 static thread_local T quantity;
522 static thread_local T scale;
523
524 // Leave unless the level is high enough to have any quantity. AKA: L > 1
525 if (level < 2) { return T(0); }
526
527 // Quantity: ceil((pos - 1) / 2)
528 // We can simplify because: ceil((P - 1) / 2) ==> ((P - 1)+1)/2 ==> pos / 2
529 quantity = position / 2;
530
531 // Scale: 2^(L-2)
532 if constexpr(BuiltinIntegral<T>) {
533 scale = T(1) << (level - 2);
534 } else if constexpr(GMPIntegral<T>) {
535 mpz_pow_ui(scale.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - 2));
536 } else {
537 throw std::logic_error("Unknown type.");
538 }
539
540 // Result of multiplication should be correctly typed.
541 return value * quantity * scale;
542 }

571 [[deprecated("Calculates negative S2 series correctly, but the whole approach is superseded by bitwise operations")]]
572 static inline T st_s2_summation(const T& position, level_t level) {
573 static thread_local T quantity;
574 static thread_local T value;
575 static thread_local T scale;
576 static thread_local T summation;
577
578 // Summation Loop
579 summation = 0;
580 for(level_t a=3; a<=level; a++) {
581 // Quantity: ceil((P - 2^(a-2)) / 2^(a-1))
582 // Simplification exists because ceiling behaves this way:
583 // ceil(x/y) ==> (x + y - 1) / y
584 //
585 // Currently the formula contains:
586 // x (numerator) : P - 2^(a-2)
587 // y (denominator): 2^(a-1)
588 // ceiling(x/y) : (x + y - 1) / y
589 //
590 // With ceiling applied it looks like this:
591 // (P - 2^(a-2) + 2^(a-1) - 1) / 2^(a-1)
592 //
593 // The first subtraction can be rewritten as addition of a negative:
594 // (P + (-2^(a-2)) + 2^(a-1) - 1) / 2^(a-1)
595 //
596 // Now it isolate the powers of two sections in the numerator to see how they simplify:
597 // -2^(a-2) + 2^(a-1)
598 // ==> -2^(a-2) + 2 * 2^(a-2)
599 // ==> -2^(a-2) + 2^(a-2) + 2^(a-2)
600 // ==> 2^(a-2)
601 //
602 // The full numerator is now simplified to:
603 // P + 2^(a-2) - 1
604 //
605 // The full expression is now just:
606 // (P + 2^(a-2) - 1) / 2^(a-1)
607 //
608 // Some arithmetic and bit shifting covers the rest:
609 // (P + (T(1) << (a - 2)) - 1) >> (a - 1)
610 quantity = (position + (T(1) << (a - 2)) - 1) >> (a - 1);
611
612 // Value: -(2^(a-1) - 3)
613 // Remember to use a positive value, even though it's technically a decreasing summation (negative).
614 if constexpr(BuiltinIntegral<T>) {
615 value = (T(1) << (a - 1)) - 3;
616 } else if constexpr(GMPIntegral<T>) {
617 mpz_pow_ui(value.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (a - 1));
618 mpz_sub_ui(value.get_mpz_t(), value.get_mpz_t(), 3);
619 } else {
620 throw std::logic_error("Unknown type.");
621 }
622
623 // Scale: 2^(L-a)
624 if constexpr(BuiltinIntegral<T>) {
625 scale = T(1) << (level - a);
626 } else if constexpr(GMPIntegral<T>) {
627 mpz_pow_ui(scale.get_mpz_t(), _MPZ_TWO.get_mpz_t(), (level - a));
628 } else {
629 throw std::logic_error("Unknown type.");
630 }
631
632 // Add to Summation
633 summation += (quantity * value * scale);
634 }
635
636 // Return it.
637 return summation;
638 }

694 static inline T st_bit_reverse_full(const T& x) {
695 if constexpr(NativeIntegral<T>) {
696 // Native types have a direct builtin. Use it.
697 using U = std::make_unsigned_t<T>;
698 U ux = static_cast<U>(x);
699 if constexpr (sizeof(T) == 1) {
700 return static_cast<T>(bitreverse8(ux));
701 } else if constexpr (sizeof(T) == 2) {
702 return static_cast<T>(bitreverse16(ux));
703 } else if constexpr (sizeof(T) == 4) {
704 return static_cast<T>(bitreverse32(ux));
705 } else if constexpr (sizeof(T) == 8) {
706 return static_cast<T>(bitreverse64(ux));
707 }
708 } else if constexpr(ExtendedIntegral<T>) {
709 // Extended 128-bit integrals need a little juggling.
710 using U = make_unsigned_custom_t<T>;
711 U ux = static_cast<U>(x);
712
713 uint64_t low = (uint64_t)ux;
714 uint64_t high = (uint64_t)(ux >> 64);
715 uint64_t rev_low = bitreverse64(high);
716 uint64_t rev_high = bitreverse64(low);
717 return static_cast<T>(( (U)rev_high << 64 ) | rev_low);
718 } else if constexpr(GMPIntegral<T>) {
719 // GMP is arbitrary precision, and is unsupported currently.
720 throw std::logic_error("Cannot fully reverse mpz_class");
721 } else {
722 static_assert(false, "Must use a supported type T");
723 }
724 }

735 static inline T st_reverse_low_bits(const T& value, size_t bits) {
736 // Test for zero.
737 T result = 0;
738 if (bits == 0) { return result; }
739
740 // Reverse and shift by the correct amount.
741 if constexpr(BuiltinIntegral<T>) {
742 result = st_bit_reverse_full(value);
743 constexpr size_t WIDTH = sizeof(T) * 8;
744 result >>= (WIDTH - bits);
745 } else if constexpr(GMPIntegral<T>) {
746 for (size_t bit = 0; bit < bits; bit++) {
747 if (mpz_tstbit(value.get_mpz_t(), bit)) {
748 mpz_setbit(result.get_mpz_t(), bits - 1 - bit);
749 }
750 }
751 }
752
753 // Return.
754 return result;
755 }

796 static inline size_t st_floor_log2(const T& val) {
797 if (val == 0) { return 0; }
798 size_t result = 0;
799 if constexpr(NativeIntegral<T>) {
800 result = 63 - __builtin_clzll(val);
801 } else if constexpr(ExtendedIntegral<T>) {
802 if (val >> 64) {
803 result = 127 - __builtin_clzll((uint64_t)(val >> 64));
804 } else {
805 result = 63 - __builtin_clzll((uint64_t)val);
806 }
807 } else if constexpr(GMPIntegral<T>) {
808 result = mpz_sizeinbase(val.get_mpz_t(), 2) - 1;
809 } else {
810 throw std::logic_error("Unknown type.");
811 }
812 return result;
813 }
