248 template<AnySupportedIntegral T>
249 inline constexpr T get_max_initial_value_by_bit(size_t bit_size) {
250 // Safety Check
251 constexpr size_t max_bits = get_max_bits_for_max_initial_value_by_type<T>();
252 if (bit_size > max_bits) {
253 throw std::out_of_range("Max initial value for bit size " + std::to_string(bit_size) + " not found because it exceeds max bits: " + std::to_string(max_bits) + ".");
254 }
255
256 if constexpr(FixedWidthIntegral<T>) {
257 return static_cast<T>(MAX_INITIAL_VALUE_BY_BIT[bit_size]);
258 } else if constexpr(GMPIntegral<T>) {
259 return Int128::uint128_to_mpz(MAX_INITIAL_VALUE_BY_BIT[bit_size]);
260 }
261 }
