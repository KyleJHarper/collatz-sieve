86 void init(const T& initial_value) {
87 if (initial_value < 0) {
88 throw std::runtime_error("You cannot create a Collatz sequence with a value lower than 0.");
89 }
90
91 // Reset if necessary.
92 if(_is_initialized) { reset(); }
93
94 // Establish or clear metadata object. Reset() already cleared it, if it existed.
95 _is_initialized = true;
96 _initial_value = initial_value;
97
98 // Process the sequence and store any related metadata, if needed. Otherwise, leave.
99 try {
100 for_each_sequence_step([&](const T& step) {
101 _step_count++;
102 if (step > _peak_value) {
103 _peak_value = step;
104 }
105 if (_ast_index == 0 && step < _initial_value) {
106 _ast_index = _step_count - 1;
107 }
108 return ForEachSignal::CONTINUE;
109 });
110 // Decrement 1 to account for start position
111 if (_step_count > 0) {
112 _step_count--;
113 }
114 } catch(const CollatzSequenceOverflow& err) {
115 _sequence_overflows = true;
116 throw(err);
117 }
118 }

191 template<AnySupportedIntegral U, bool UseIVTable = true>
192 static inline __attribute__((always_inline)) bool st_verify(const U& initial_value, const U& sentinel_value = 1) {
193 // Since Collatz sequences converge to 2->1->2 loops when using accerlated functions F and G, it's possible for an affine
194 // map to terminate on 1 or 2, not just 1. To solve for this without introducing an additional test inside the hot loop,
195 // the sentinel value is required to be at least 3.
196 //
197 // These are addressed differently for fixed-width vs GMP types below.
198
199 // If the value is even, it's automatically verified.
200 if ((initial_value & 1) == 0) {
201 return true;
202 }
203
204 // Type selection and main loops.
205 if constexpr(FixedWidthIntegral<U>) {
206 // Sentinal value safety, solves low-value edge cases (see above). For fixed-width types, use a stack variable.
207 U effective_sentinel_value = sentinel_value > U(2) ? sentinel_value : U(3);
208
209 // When the initial value is below the known (precomputed) max initial values defined in CollatzConstants, there's no
210 // need to track headroom or detect overflow in the hot loop. It can be calculated, promoted, and then use raw
211 // arithmetic without checks, which enables all manner of compiler and CPU optimizations.
212 //
213 // If the table is viable, pick the known-safe type and pass the work to the "unsafe" verifier for speed.
214 if constexpr(UseIVTable) {
215 if (initial_value <= CollatzConstants::MAX_INITIAL_VALUE_BY_BIT.back()) {
216 // Is promotion needed? If not, send it as type U (as-is).
217 if (initial_value <= CollatzConstants::get_max_initial_value_by_type<U>()) {
218 return st_verify_unsafe(initial_value, effective_sentinel_value);
219 }
220
221 // Promotion needed. Current type U cannot handle it.
222 if constexpr(sizeof(U) * 8 < 64) {
223 // Type U is less than 64 bits, just send it as a 64-bit. Even uint32_t can't reach an overflow of 64-bit.
224 // Max initial value of 64 bit is empirically shown to be 12,327,829,502 which is greater than 2^32 which is 4,294,967,296.
225 uint64_t promoted_initial_value = initial_value;
226 uint64_t promoted_effective_sentinel_value = effective_sentinel_value;
227 return st_verify_unsafe(promoted_initial_value, promoted_effective_sentinel_value);
228 } else if constexpr (sizeof(U) * 8 == 64) {
229 // Dealing with U of exactly 64 bits. Uint128_t can handle it.
230 uint128_t promoted_initial_value = initial_value;
231 uint128_t promoted_effective_sentinel_value = effective_sentinel_value;
232 return st_verify_unsafe(promoted_initial_value, promoted_effective_sentinel_value);
233 } else if constexpr (sizeof(U) * 8 == 128) {
234 // Dealing with 128 bits. Upgrade to mpz_class.
235 static thread_local mpz_class promoted_initial_value_tls;
236 static thread_local mpz_class promoted_effective_sentinel_value_tls;
237 Int128::uint128_to_mpz(initial_value, promoted_initial_value_tls);
238 Int128::uint128_to_mpz(effective_sentinel_value, promoted_effective_sentinel_value_tls);
239 return st_verify_unsafe(promoted_initial_value_tls, promoted_effective_sentinel_value_tls);
240 }
241 }
242 //
243 // Table above wasn't viable or used. Process while tracking headroom bits for realtime safety.
244 }
245
246 // Pick a steady stride table.
247 using StrideTable = AffineStride::VerifyFWTable;
248
249 // Use a stack variable for the temp.
250 U current_value = initial_value;
251
252 // Track headroom and make adjustments from Stride to prevent overflow efficiently.
253 uint8_t headroom_bits = static_cast<uint8_t>(Bit::count_leading_zeros_positive(current_value));
254
255 // Run the loop.
256 while (current_value >= effective_sentinel_value) {
257 // Pick the next stride.
258 const AffineStride::Stride& stride = StrideTable::get_stride(current_value);
259
260 // Check headroom to see if we need promotion.
261 if (headroom_bits < stride.bits_required) {
262 if constexpr(sizeof(U) * 8 < 64) {
263 // Type is less than 64 bits, just send it as a 64-bit. Even uint32_t can't reach an overflow of 64-bit.
264 // Max initial value of 64 bit is empirically shown to be 12,327,829,502 which is greater than 2^32 which is 4,294,967,296.
265 uint64_t promoted_current_value = current_value;
266 uint64_t promoted_effective_sentinel_value = effective_sentinel_value;
267 return st_verify(promoted_current_value, promoted_effective_sentinel_value);
268 } else if constexpr (sizeof(U) * 8 == 64) {
269 // Dealing with U of exactly 64 bits. Uint128_t is the next best candidate.
270 uint128_t promoted_current_value = current_value;
271 uint128_t promoted_effective_sentinel_value = effective_sentinel_value;
272 return st_verify(promoted_current_value, promoted_effective_sentinel_value);
273 } else if constexpr (sizeof(U) * 8 == 128) {
274 // Dealing with 128 bits. Upgrade to mpz_class if needed.
275 // Headroom is conservative because Stride.bits_required is pessimistic. It's worth a quick clz() to re-chec headroom before giving up on the GPU entirely.
276 headroom_bits = static_cast<uint8_t>(Bit::count_leading_zeros_positive(current_value));
277 if (headroom_bits < stride.bits_required) {
278 // It's truly too low to handle this value. Upgrade to GMP.
279 static thread_local mpz_class promoted_current_value_tls;
280 static thread_local mpz_class promoted_effective_sentinel_value_tls;
281 Int128::uint128_to_mpz(current_value, promoted_current_value_tls);
282 Int128::uint128_to_mpz(effective_sentinel_value, promoted_effective_sentinel_value_tls);
283 return st_verify(promoted_current_value_tls, promoted_effective_sentinel_value_tls);
284 }
285 // The updated headroom from clz() says it's okay. Fall out of this block and keep going.
286 }
287 }
288
289 // No promotion needed. Apply the stride.
290 StrideTable::apply_stride(current_value, stride);
291
292 // Adjust headroom according to Stride.
293 headroom_bits += stride.shift;
294 headroom_bits -= stride.bits_required;
295 }
296 } else if constexpr(GMPIntegral<U>) {
297 // Sentinal value safety, solves low-value edge cases (see above). For GMP types, use a thread-local.
298 static thread_local U effective_sentinel_value;
299 effective_sentinel_value = sentinel_value > 2 ? sentinel_value : 3;
300
301 // Use a thread-local to avoid GMP allocs.
302 static thread_local U current_value;
303 current_value = initial_value;
304
305 // GMP cannot overflow, so there's no need to track headroom_bits.
306
307 // Pick a steady stride table. Twelve seems to be the sweet spot for this according to microbenchmarking.
308 using StrideTable = AffineStride::VerifyGMPTable;
309
310 // Run the loop.
311 while (current_value >= effective_sentinel_value) {
312 // Pick the next stride.
313 const AffineStride::Stride& stride = StrideTable::get_stride(current_value);
314
315 // Apply the stride.
316 StrideTable::apply_stride(current_value, stride);
317 }
318 }
319
320 // Everything is okay.
321 return true;
322 }

344 template<AnySupportedIntegral U>
345 static inline bool st_verify_unsafe(const U& initial_value, const U& sentinel_value) {
346 // Caller has checked for (or disregarded) overflow potential already. Just blaze through the loop. Make CPU happy.
347 if constexpr(FixedWidthIntegral<U>) {
348 // Stack allocate and fly.
349 U current_value = initial_value;
350 while (current_value >= sentinel_value) {
351 // Must be odd here.
352 current_value = (current_value << 1) + current_value + 1;
353 // Must be even here.
354 current_value >>= Bit::count_trailing_zeros_positive(current_value);
355 }
356 } else if constexpr(GMPIntegral<U>) {
357 // Use a thread-local to avoid GMP allocs.
358 static thread_local U current_value;
359 current_value = initial_value;
360
361 // Use striding with GMP. Always.
362 using StrideTable = AffineStride::VerifyGMPTable;
363
364 // Run the loop.
365 while (current_value >= sentinel_value) {
366 // Pick the next stride.
367 const AffineStride::Stride& stride = StrideTable::get_stride(current_value);
368
369 // Apply the stride.
370 StrideTable::apply_stride(current_value, stride);
371 }
372 }
373
374 // All good.
375 return true;
376 }

387 template<typename Func>
388 static void st_for_each_sequence_step(const T& initial_value, Func&& callback) {
389 // Do not allow non-ref callbacks. Otherwise it can make GMP over and over.
390 static_assert(std::is_invocable_v<Func, const T&>, "Callback must be callable with (const T&)");
391 // Require ForEachSignal return type.
392 static_assert(std::is_same_v<std::invoke_result_t<Func, const T&>, ForEachSignal>, "Callback must return ForEachSignal");
393
394 // Zero is a special case, mostly for BinaryTree building a root.
395 if (initial_value == 0) { return; }
396
397 if constexpr(FixedWidthIntegral<T>) {
398 // Fixed integrals can use intrinsic arithmetic operators for "free", but can overflow.
399 T current_value = initial_value;
400 while (current_value != 1) {
401 if (callback(current_value) == ForEachSignal::BREAK) { return; }
402 if ((current_value & 1) == 0) {
403 current_value >>= 1;
404 } else {
405 if (current_value > CollatzConstants::get_max_3xp1<T>()) {
406 throw CollatzSequenceOverflow("Overflow when building for_each_sequence_step().");
407 }
408 current_value = (current_value << 1) + current_value + 1;
409 }
410 }
411 // Since the while-loop exits prematurely at 1, it needs one more callback before ending.
412 callback(current_value);
413 } else if constexpr(GMPIntegral<T>) {
414 // GMP integers will alloc() with certain arithmetic operators, but can't overflow.
415 thread_local T current_value;
416 current_value = initial_value;
417 while (mpz_cmp_ui(current_value.get_mpz_t(), 1) != 0) {
418 if (callback(current_value) == ForEachSignal::BREAK) { return; }
419 if (mpz_even_p(current_value.get_mpz_t())) {
420 mpz_tdiv_q_2exp(current_value.get_mpz_t(), current_value.get_mpz_t(), 1); // current_step >> 1 ==> current_step /= 2
421 } else {
422 mpz_mul_ui(current_value.get_mpz_t(), current_value.get_mpz_t(), 3); // current_step *= 3
423 mpz_add_ui(current_value.get_mpz_t(), current_value.get_mpz_t(), 1); // current_step += 1
424 }
425 }
426 // Since the while-loop exits prematurely at 1, it needs one more callback before ending.
427 callback(current_value);
428 }
429 }

446 static seq_size_t st_get_step_count(const T& initial_value) {
447 size_t right_shifts = 0;
448 seq_size_t steps = 0;
449
450 if constexpr(FixedWidthIntegral<T>) {
451 // Native types are fast as-is. Affine compression doesn't help, except bit-shifting CTZ.
452 T tmp = initial_value;
453 // Check for overflow here, once, instead of over and over.
454 if (initial_value > CollatzConstants::get_max_initial_value_by_bit<T>(std::numeric_limits<T>::digits)) {
455 throw CollatzSequenceOverflow("Overflow when building st_get_step_count().");
456 }
457 while (tmp > 1) {
458 // Handle odd.
459 if ((tmp & 1) == 1) {
460 tmp = (tmp << 1) + tmp + 1;
461 steps++;
462 }
463 // Always even at this point. Shift zeros out.
464 right_shifts = Bit::count_trailing_zeros(tmp);
465 tmp >>= right_shifts;
466 steps += right_shifts;
467 }
468 } else if constexpr(GMPIntegral<T>) {
469 // GMP types are significantly faster using an Affine map for CTZ and CTO, ergo it's worth the overhead.
470 static thread_local T tmp;
471 tmp = initial_value;
472 size_t trailing_ones = 0;
473 constexpr size_t limit = Exponents::POW3_MPZ_ELEMENT_COUNT - 1;
474 // See collatz_compression.cpp tests for details on how this works and why.
475 while (tmp > 1) {
476 // Handle odd.
477 if ((tmp & 1) == 1) {
478 trailing_ones = Bit::count_trailing_ones(tmp);
479 steps += (2 * trailing_ones);
480 while (trailing_ones > limit) {
481 tmp = ((Exponents::POW3_MPZ[limit] * (tmp + 1)) >> limit) - 1;
482 trailing_ones -= limit;
483 }
484 if (trailing_ones > 0) {
485 tmp = ((Exponents::POW3_MPZ[trailing_ones] * (tmp + 1)) >> trailing_ones) - 1;
486 }
487 }
488 // Always even at this point. Shift zeros out.
489 right_shifts = Bit::count_trailing_zeros(tmp);
490 tmp >>= right_shifts;
491 steps += right_shifts;
492 }
493 }
494 return steps;
495 }

505 static inline void st_get_peak(const T& initial_value, T& out_peak, bool stop_at_ast = false) {
506 size_t right_shifts = 0;
507 out_peak = initial_value;
508
509 if constexpr(FixedWidthIntegral<T>) {
510 // Native types are fast as-is. Affine compression of consecutive ones doesn't help, except bit-shifting CTZ.
511 T current_value = initial_value;
512 T bailout_value = (stop_at_ast && initial_value > 1) ? (T)initial_value - 1 : T(1);
513 // Shift CTZ to ensure odd start.
514 current_value >>= Bit::count_trailing_zeros(current_value);
515 while (current_value > bailout_value) {
516 // Always odd here. Check for overflow.
517 if (current_value > CollatzConstants::get_max_3xp1<T>()) {
518 throw std::out_of_range("Cannot process initial_value " + to_string_any(initial_value) + " any further in st_get_peak.");
519 }
520 current_value = (current_value << 1) + current_value + 1;
521 if (current_value > out_peak) {
522 out_peak = current_value;
523 }
524 // Always even at this point. Shift zeros out. Can't affect peak.
525 current_value >>= Bit::count_trailing_zeros(current_value);
526 }
527 } else if constexpr(GMPIntegral<T>) {
528 // GMP types are significantly faster using an Affine map for CTZ and CTO, ergo it's worth the overhead.
529 static thread_local T current_value;
530 current_value = initial_value;
531 static thread_local T tmp_x_2;
532 static thread_local T bailout_value;
533 bailout_value = 1;
534 if (stop_at_ast && initial_value > 1) {
535 bailout_value = initial_value;
536 mpz_sub_ui(bailout_value.get_mpz_t(), bailout_value.get_mpz_t(), 1);
537 }
538 size_t trailing_ones = 0;
539 constexpr size_t limit = Exponents::POW3_MPZ_ELEMENT_COUNT - 1;
540 // See collatz_compression.cpp tests for details on how this works and why.
541 while (current_value > bailout_value) {
542 // Handle odd.
543 if (mpz_odd_p(current_value.get_mpz_t())) {
544 trailing_ones = Bit::count_trailing_ones(current_value);
545 while (trailing_ones > limit) {
546 mpz_add_ui(current_value.get_mpz_t(), current_value.get_mpz_t(), 1);
547 mpz_mul(current_value.get_mpz_t(), current_value.get_mpz_t(), Exponents::POW3_MPZ[limit].get_mpz_t());
548 mpz_tdiv_q_2exp(current_value.get_mpz_t(), current_value.get_mpz_t(), limit);
549 mpz_sub_ui(current_value.get_mpz_t(), current_value.get_mpz_t(), 1);
550 trailing_ones -= limit;
551 }
552 if (trailing_ones > 0) {
553 mpz_add_ui(current_value.get_mpz_t(), current_value.get_mpz_t(), 1);
554 mpz_mul(current_value.get_mpz_t(), current_value.get_mpz_t(), Exponents::POW3_MPZ[trailing_ones].get_mpz_t());
555 mpz_tdiv_q_2exp(current_value.get_mpz_t(), current_value.get_mpz_t(), trailing_ones);
556 mpz_sub_ui(current_value.get_mpz_t(), current_value.get_mpz_t(), 1);
557 }
558 // Now check peak. However! It applied an accelerated F(x): (3x + 1) / 2
559 // Peak was actually tmp * 2.
560 mpz_mul_2exp(tmp_x_2.get_mpz_t(), current_value.get_mpz_t(), 1);
561 if (mpz_cmp(tmp_x_2.get_mpz_t(), out_peak.get_mpz_t()) > 0) {
562 out_peak = tmp_x_2;
563 }
564 }
565 // Always even at this point. Shift zeros out.
566 right_shifts = Bit::count_trailing_zeros(current_value);
567 current_value >>= right_shifts;
568 }
569 }
570 }

581 template<typename Func>
582 static void st_for_each_fg_chain_link(const T& initial_value, Func&& callback) {
583 // Do not allow non-ref callbacks. Otherwise it makes GMP over and over.
584 static_assert(std::is_invocable_v<Func, bool>, "Callback must be callable as (bool)");
585 // Require ForEachSignal return type.
586 static_assert(std::is_same_v<std::invoke_result_t<Func, bool>, ForEachSignal>, "Callback must return ForEachSignal");
587
588 // Zero is a special case, mostly for BinaryTree building a root.
589 if (initial_value == 0) { return; }
590
591 // Call our basic Collatz iterator.
592 bool is_F = false;
593 Collatz<T>::st_for_each_sequence_step(initial_value, [&](const T& current_value) {
594 // When the previous step was F, skip the next step which is always the latter part of accerlated F(x).
595 if (is_F) {
596 // Reset it so it doesn't loop.
597 is_F = false;
598 return ForEachSignal::CONTINUE;
599 } else {
600 if constexpr(FixedWidthIntegral<T>) {
601 is_F = (current_value & 1) == 1;
602 } else if constexpr(GMPIntegral<T>) {
603 is_F = mpz_odd_p(current_value.get_mpz_t());
604 }
605 return callback(is_F);
606 }
607 });
608 }

710 static std::string st_fg_to_oe(
711 const std::string& fg_pattern
712 , size_t max_oe_chars = std::numeric_limits<size_t>::max()
713 , bool strip_last_e = true
714 ) {
715 std::string oe_string;
716 for (const char& c : fg_pattern) {
717 if (c == 'F') {
718 oe_string += "OE";
719 } else {
