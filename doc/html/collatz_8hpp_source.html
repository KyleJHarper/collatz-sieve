85 void init(const T& initial_value) {
86 if (initial_value < 0) {
87 throw std::runtime_error("You cannot create a Collatz sequence with a value lower than 0.");
88 }
89
90 // Reset if necessary.
91 if(_is_initialized) { reset(); }
92
93 // Establish or clear metadata object. Reset() already cleared it, if it existed.
94 _is_initialized = true;
95 _initial_value = initial_value;
96
97 // Process the sequence and store any related metadata, if needed. Otherwise, leave.
98 try {
99 for_each_sequence_step([&](const T& step) {
100 _step_count++;
101 if (step > _peak_value) {
102 _peak_value = step;
103 }
104 if (_hwm_index == 0 && step < _initial_value) {
105 _hwm_index = _step_count - 1;
106 }
107 return false;
108 });
109 // Decrement 1 to account for start position
110 if (_step_count > 0) {
111 _step_count--;
112 }
113 } catch(const CollatzSequenceOverflow& err) {
114 _sequence_overflows = true;
115 throw(err);
116 }
117 }

187 static inline bool st_verify(const T& initial_value, const T& sentinel_value = 1) {
188 // Quick sanity check. Sentinel value should be within the Collatz space (>=1) or it'll loop forever.
189 // However, there are cases, such as a tree rooted at 0, where this might get tripped up.
190 if (sentinel_value < 1) {
191 throw std::out_of_range("You cannot send a sentinel value for st_verify below 1. You sent: " + to_string_any(sentinel_value));
192 }
193
194 // Select the correct type to upgrade to, if needed for overflow.
195 if constexpr(FixedWidthIntegral<T>) {
196 if constexpr(sizeof(T) * 8 < 64) {
197 // Type is less than 64 bits, just send it as a 64-bit. Even uint32_t can't reach an overflow of 64-bit.
198 // Max initial value of 64 bit is 12,327,829,502 which is greater than 2^32 which is 4,294,967,296.
199 uint64_t safe_initial_value = initial_value;
200 uint64_t safe_sentinel_value = sentinel_value;
201 return st_call_verify_impl(safe_initial_value, safe_sentinel_value);
202 } else if constexpr (sizeof(T) * 8 == 64) {
203 // Dealing with T of exactly 64 bits. If it'll overflow, use 128-bit. Otherwise, pass through.
204 if (initial_value > CollatzConstants::get_max_initial_value_by_bit<T>(64)) {
205 uint128_t safe_initial_value = initial_value;
206 uint128_t safe_sentinel_value = sentinel_value;
207 return st_call_verify_impl(safe_initial_value, safe_sentinel_value);
208 }
209 } else if constexpr (sizeof(T) * 8 == 128) {
210 // Dealing with 128 bits. If it'll overflow, use mpz_class.
211 if (initial_value > CollatzConstants::get_max_initial_value_by_bit<T>(128)) {
212 // Extended will overflow, but mpz_class won't. Assignment of 128 -> mpz_class uses helper.
213 static thread_local mpz_class safe_initial_value;
214 static thread_local mpz_class safe_sentinel_value;
215 Int128::uint128_to_mpz(initial_value, safe_initial_value);
216 Int128::uint128_to_mpz(sentinel_value, safe_sentinel_value);
217 return st_call_verify_impl(safe_initial_value, safe_sentinel_value);
218 }
219 }
220 }
221
222 // If no overflow detected, simply send the call as type T.
223 return st_call_verify_impl(initial_value, sentinel_value);
224 }

257 static inline bool st_verify_impl(const T& initial_value) {
258 // Use a copy in a TLS to avoid alloc on GMP type, and to avoid mangling caller's data.
259 static thread_local T tls_initial_value;
260 tls_initial_value = initial_value;
261 while (tls_initial_value > 1) {
262 if constexpr(FixedWidthIntegral<T>) {
263 // The key optimizations are shifting by CTZ and using modulo instead of bitwise check.
264 if (tls_initial_value % 2 == 1) {
265 // Odd
266 tls_initial_value = (tls_initial_value << 1) + tls_initial_value + 1;
267 }
268 // Always even at this point, so shift by CTZ.
269 tls_initial_value >>= Bit::count_trailing_zeros(tls_initial_value);
270 } else if constexpr(GMPIntegral<T>) {
271 // GMP optimizations are to use bitwise check, shift by CTZ, and leverage an affine map for consecutive ones.
272 static thread_local CollatzAffineMap<T> af_map;
273 static thread_local T tls_out;
274 if ((tls_initial_value & 1) == 1) {
275 // Odd
276 af_map.reset();
277 size_t trailing_ones = Bit::count_trailing_ones(tls_initial_value);
278 for (size_t i = 0; i < trailing_ones; i++) {
279 af_map.apply_F();
280 }
281 af_map.calculate(tls_initial_value, tls_out);
282 tls_initial_value = tls_out;
283 }
284 // Always even at this point, so shift by CTZ.
285 tls_initial_value >>= Bit::count_trailing_zeros(tls_initial_value);
286 }
287 }
288
289 return true;
290 }

344 template<typename Func>
345 static void st_for_each_sequence_step(const T& initial_value, Func&& callback) {
346 // Do not allow non-ref callbacks. Otherwise it can make GMP over and over.
347 static_assert(std::is_invocable_v<Func, const T&>, "Callback must be callable with (const T&)");
348
349 // Zero is a special case, mostly for BinaryTree building a root.
350 if (initial_value == 0) { return; }
351
352 if constexpr(FixedWidthIntegral<T>) {
353 // Fixed integrals can use intrinsic arithmetic operators for "free", but can overflow.
354 T current_value = initial_value;
355 while (current_value != 1) {
356 bool stop = callback(current_value);
357 if (stop) { return; }
358 if ((current_value & 1) == 0) {
359 current_value >>= 1;
360 } else {
361 if (current_value > CollatzConstants::get_max_3xp1<T>()) {
362 throw CollatzSequenceOverflow("Overflow when building for_each_sequence_step().");
363 }
364 current_value = (current_value << 1) + current_value + 1;
365 }
366 }
367 // Since the while-loop exits prematurely at 1, it needs one more callback before ending.
368 callback(current_value);
369 } else if constexpr(GMPIntegral<T>) {
370 // GMP integers will alloc() with certain arithmetic operators, but can't overflow.
371 thread_local T current_value;
372 current_value = initial_value;
373 while (mpz_cmp_ui(current_value.get_mpz_t(), 1) != 0) {
374 bool stop = callback(current_value);
375 if (stop) { return; }
376 if (mpz_even_p(current_value.get_mpz_t())) {
377 mpz_tdiv_q_2exp(current_value.get_mpz_t(), current_value.get_mpz_t(), 1); // current_step >> 1 ==> current_step /= 2
378 } else {
379 mpz_mul(current_value.get_mpz_t(), current_value.get_mpz_t(), CollatzConstants::MPZ_THREE.get_mpz_t()); // current_step *= 3
380 mpz_add(current_value.get_mpz_t(), current_value.get_mpz_t(), CollatzConstants::MPZ_ONE.get_mpz_t()); // current_step += 1
381 }
382 }
383 // Since the while-loop exits prematurely at 1, it needs one more callback before ending.
384 callback(current_value);
385 }
386 }

403 static seq_size_t st_get_step_count(const T& initial_value) {
404 size_t right_shifts = 0;
405 seq_size_t steps = 0;
406
407 if constexpr(FixedWidthIntegral<T>) {
408 // Native types are fast as-is. Affine compression doesn't help, except bit-shifting CTZ.
409 T tmp = initial_value;
410 // Check for overflow here, once, instead of over and over.
411 if (initial_value > CollatzConstants::get_max_initial_value_by_bit<T>(std::numeric_limits<T>::digits)) {
412 throw CollatzSequenceOverflow("Overflow when building st_get_step_count().");
413 }
414 while (tmp > 1) {
415 // Handle odd.
416 if ((tmp & 1) == 1) {
417 tmp = (tmp << 1) + tmp + 1;
418 steps++;
419 }
420 // Always even at this point. Shift zeros out.
421 right_shifts = Bit::count_trailing_zeros(tmp);
422 tmp >>= right_shifts;
423 steps += right_shifts;
424 }
425 } else if constexpr(GMPIntegral<T>) {
426 // GMP types are significantly faster using an Affine map for CTZ and CTO, ergo it's worth the overhead.
427 static thread_local T tmp;
428 tmp = initial_value;
429 size_t trailing_ones = 0;
430 constexpr size_t limit = Exponents::POW3_MPZ_ELEMENT_COUNT - 1;
431 // See collatz_compression.cpp tests for details on how this works and why.
432 while (tmp > 1) {
433 // Handle odd.
434 if ((tmp & 1) == 1) {
435 trailing_ones = Bit::count_trailing_ones(tmp);
436 steps += (2 * trailing_ones);
437 while (trailing_ones > limit) {
438 tmp = ((Exponents::POW3_MPZ[limit] * (tmp + 1)) >> limit) - 1;
439 trailing_ones -= limit;
440 }
441 if (trailing_ones > 0) {
442 tmp = ((Exponents::POW3_MPZ[trailing_ones] * (tmp + 1)) >> trailing_ones) - 1;
443 }
444 }
445 // Always even at this point. Shift zeros out.
446 right_shifts = Bit::count_trailing_zeros(tmp);
447 tmp >>= right_shifts;
448 steps += right_shifts;
449 }
450 }
451 return steps;
452 }

462 static inline void st_get_peak(const T& initial_value, T& out_peak, bool stop_at_hwm = false) {
463 size_t right_shifts = 0;
464 out_peak = initial_value;
465
466 if constexpr(FixedWidthIntegral<T>) {
467 // Native types are fast as-is. Affine compression doesn't help, except bit-shifting CTZ.
468 T tmp = initial_value;
469 T bailout_value = (stop_at_hwm && initial_value > 1) ? (T)initial_value - 1 : T(1);
470 while (tmp > bailout_value) {
471 // Handle odd.
472 if ((tmp & 1) == 1) {
473 if (tmp > CollatzConstants::get_max_3xp1<T>()) {
474 throw std::out_of_range("Cannot process initial_value " + to_string_any(initial_value) + " any further in st_get_peak.");
475 }
476 tmp = (tmp << 1) + tmp + 1;
477 if (tmp > out_peak) {
478 out_peak = tmp;
479 }
480 }
481 // Always even at this point. Shift zeros out. Can't affect peak.
482 right_shifts = Bit::count_trailing_zeros(tmp);
483 tmp >>= right_shifts;
484 }
485 } else if constexpr(GMPIntegral<T>) {
486 // GMP types are significantly faster using an Affine map for CTZ and CTO, ergo it's worth the overhead.
487 static thread_local T tmp;
488 tmp = initial_value;
489 static thread_local T tmp_x_2;
490 static thread_local T bailout_value;
491 bailout_value = 1;
492 if (stop_at_hwm && initial_value > 1) {
493 bailout_value = initial_value;
494 mpz_sub_ui(bailout_value.get_mpz_t(), bailout_value.get_mpz_t(), 1);
495 }
496 size_t trailing_ones = 0;
497 constexpr size_t limit = Exponents::POW3_MPZ_ELEMENT_COUNT - 1;
498 // See collatz_compression.cpp tests for details on how this works and why.
499 while (tmp > bailout_value) {
500 // Handle odd.
501 if (mpz_odd_p(tmp.get_mpz_t())) {
502 trailing_ones = Bit::count_trailing_ones(tmp);
503 while (trailing_ones > limit) {
504 mpz_add_ui(tmp.get_mpz_t(), tmp.get_mpz_t(), 1);
505 mpz_mul(tmp.get_mpz_t(), tmp.get_mpz_t(), Exponents::POW3_MPZ[limit].get_mpz_t());
506 mpz_tdiv_q_2exp(tmp.get_mpz_t(), tmp.get_mpz_t(), limit);
507 mpz_sub_ui(tmp.get_mpz_t(), tmp.get_mpz_t(), 1);
508 trailing_ones -= limit;
509 }
510 if (trailing_ones > 0) {
511 mpz_add_ui(tmp.get_mpz_t(), tmp.get_mpz_t(), 1);
512 mpz_mul(tmp.get_mpz_t(), tmp.get_mpz_t(), Exponents::POW3_MPZ[trailing_ones].get_mpz_t());
513 mpz_tdiv_q_2exp(tmp.get_mpz_t(), tmp.get_mpz_t(), trailing_ones);
514 mpz_sub_ui(tmp.get_mpz_t(), tmp.get_mpz_t(), 1);
515 }
516 // Now check peak. However! It applied an accelerated F(x): (3x + 1) / 2
517 // Peak was actually tmp * 2.
518 mpz_mul_2exp(tmp_x_2.get_mpz_t(), tmp.get_mpz_t(), 1);
519 if (mpz_cmp(tmp_x_2.get_mpz_t(), out_peak.get_mpz_t()) > 0) {
520 out_peak = tmp_x_2;
521 }
522 }
523 // Always even at this point. Shift zeros out.
524 right_shifts = Bit::count_trailing_zeros(tmp);
525 tmp >>= right_shifts;
526 }
527 }
528 }

547 // Call our basic Collatz iterator.
548 bool is_F = false;
549 Collatz<T>::st_for_each_sequence_step(initial_value, [&](const T& current_value) {

662 static std::string st_fg_to_oe(
663 const std::string& fg_pattern
664 , size_t max_oe_chars = std::numeric_limits<size_t>::max()
665 , bool strip_last_e = true
666 ) {
667 std::string oe_string;
668 for (const char& c : fg_pattern) {
669 if (c == 'F') {
670 oe_string += "OE";
671 } else {
672 oe_string += "E";
673 }
674 if (oe_string.size() > max_oe_chars) {
675 break;
676 }
677 }
678
679 // If this didn't hit the max_chars, remove the last E. Because all sequences end in 1 (odd) and therefore get an "F",
680 // which traslates to "OE", but that last "E" is 1 going to 4 (1 * 3 + 1).
681 // If the caller forbids this, skip it.
682 if (strip_last_e && oe_string.size() < max_oe_chars && oe_string.size() > 0) {
683 oe_string.pop_back();
684 }
685
686 // If it's over the max_chars requested, trim.
687 if (oe_string.size() > max_oe_chars) {
688 oe_string.resize(max_oe_chars);
689 }
690
691 return oe_string;
692 }
