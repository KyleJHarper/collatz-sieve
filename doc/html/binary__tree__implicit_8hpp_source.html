355 void add_level() {
356 // When there are no levels, this simply crafts a 0- or 1-based root node and manually sets level map and coverage.
357 if (_level_count == 0) {
358 _level_count = 1;
359 _root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
360 _coverage_map[_level_count].set_covered(0);
361 _coverage_map[_level_count].set_total(1);
362 _uncovered_positions.add(1);
363 return;
364 }
365
366 // Bump level count.
367 _level_count++;
368
369 // Establish TLS for our callback.
370 std::vector<AddLevelTLS> callback_storage;
371 // Track our own local High-Water Mark. Since nodes are not sequentially ordered, we can only set this to first node - 1.
372 T add_level_high_water_mark = BinaryTreeMath<T>::st_first_node_value_of_level(_level_count) - 1;
373 if (add_level_high_water_mark < 1) {
374 add_level_high_water_mark = 1;
375 }
376
377 // Now loop.
378 _uncovered_positions.for_each_transformer(BitmapTransformerPolicy::PARALLEL, callback_storage, [&](const T& cb_position, AddLevelTLS& tls) {
379 // Scale the uncovered position to this level. Make these TLS storage to avoid alloc() on GMP path.
380 if constexpr(BuiltinIntegral<T>) {
381 tls.left_position = (cb_position << 1) - 1;
382 tls.right_position = cb_position << 1;
383 } else {
384 mpz_mul_2exp(tls.left_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
385 mpz_sub_ui(tls.left_position.get_mpz_t(), tls.left_position.get_mpz_t(), 1);
386 mpz_mul_2exp(tls.right_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
387 }
388
389 // Unroll this loop manually.
390 // for (const T& position : tls.positions) {...}
391
392 // Calculate the values from positions. Use an out param on GMP path.
393 if constexpr(BuiltinIntegral<T>) {
394 tls.left_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count);
395 tls.right_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count);
396 } else {
397 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count, tls.left_value);
398 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count, tls.right_value);
399 }
400
401 // Initialize the node and test HWM for ancestry.
402 tls.left_node.init(tls.left_value);
403 tls.right_node.init(tls.right_value);
404 if (_is_preserving_ancestors) {
405 if (tls.left_node.is_below_high_water_mark()) {
406 tls.new_ancestors.push_back(new Node<T>(tls.left_value));
407 }
408 if (tls.right_node.is_below_high_water_mark()) {
409 tls.new_ancestors.push_back(new Node<T>(tls.right_value));
410 }
411 }
412
413 // Perform verification, if requested.
414 if (_is_verifying_non_hwm_nodes) {
415 if (tls.left_node.is_below_high_water_mark() == false) {
416 if (Collatz<T>::st_verify(tls.left_value, add_level_high_water_mark) == false) {
417 throw std::logic_error("Node value " + to_string_any(tls.left_value) + " didn't verify. How?");
418 }
419 }
420 if (tls.right_node.is_below_high_water_mark() == false) {
421 if (Collatz<T>::st_verify(tls.right_value, add_level_high_water_mark) == false) {
422 throw std::logic_error("Node value " + to_string_any(tls.right_value) + " didn't verify. How?");
423 }
424 }
425 }
426
427 // Write the POSITION to the TLS bitmap if it didn't hit HWM.
428 if (tls.left_node.is_below_high_water_mark() == false) {
429 tls.uncovered_bitmap.add(tls.left_position);
430 }
431 if (tls.right_node.is_below_high_water_mark() == false) {
432 tls.uncovered_bitmap.add(tls.right_position);
433 }
434
435 // Return false. There is no stopping condition.
436 return false;
437 });
438
439 // Merge TLS data into the tree's trackers.
440 _uncovered_positions.clear();
441 for (const AddLevelTLS& storage : callback_storage) {
442 _uncovered_positions |= storage.uncovered_bitmap;
443 _ancestors.insert(_ancestors.end(), storage.new_ancestors.begin(), storage.new_ancestors.end());
444 }
445
446 // Always sort ancestors.
447 if (_is_preserving_ancestors) {
448 tbb::parallel_sort(_ancestors.begin(), _ancestors.end(), [](const Node<T>* a, const Node<T>* b) {
449 if constexpr(BuiltinIntegral<T>) {
450 return a->get_value() < b->get_value();
451 } else {
452 return mpz_cmp(a->get_value().get_mpz_t(), b->get_value().get_mpz_t()) < 0;
453 }
454 });
455 }
456
457 // Persist the coverage to the new level. It's just a sum of the uncovered size() values subtracted from the node total.
458 T total = BinaryTreeMath<T>::st_node_count_of_level(_level_count);
459 T uncovered = _uncovered_positions.cardinality();
460 _coverage_map[_level_count] = BinaryTreeCoverage<T>(total - uncovered, total);
461
462 // Invoke optimize above level 32.
463 if (_level_count > 32) {
464 _uncovered_positions.optimize();
465 }
466 }
