379 void add_level() {
380 // Ensure initialization is set.
381 _is_initialized = true;
382
383 // When there are no levels, this simply crafts a 0- or 1-based root node and manually sets level map and coverage.
384 if (_level_count == 0) {
385 _level_count = 1;
386 _root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
387 _coverage_map[_level_count].set_covered(0);
388 _coverage_map[_level_count].set_total(1);
389 _uncovered_positions.add(1);
390 return;
391 }
392
393 // Bump level count.
394 _level_count++;
395
396 // Establish TLS for our callback.
397 std::vector<AddLevelTLS> callback_storage;
398 // Track our own local High-Water Mark. Since nodes are not sequentially ordered, we can only set this to first node - 1.
399 T add_level_high_water_mark = BinaryTreeMath<T>::st_first_node_value_of_level(_level_count) - 1;
400 if (add_level_high_water_mark < 1) {
401 add_level_high_water_mark = 1;
402 }
403
404 // Now loop.
405 _uncovered_positions.for_each_transformer(BitmapTransformerPolicy::PARALLEL, callback_storage, [&](const T& cb_position, AddLevelTLS& tls) {
406 // Scale the uncovered position to this level. Make these TLS storage to avoid alloc() on GMP path.
407 if constexpr(FixedWidthIntegral<T>) {
408 tls.left_position = (cb_position << 1) - 1;
409 tls.right_position = cb_position << 1;
410 } else if constexpr(GMPIntegral<T>) {
411 mpz_mul_2exp(tls.left_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
412 mpz_sub_ui(tls.left_position.get_mpz_t(), tls.left_position.get_mpz_t(), 1);
413 mpz_mul_2exp(tls.right_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
414 }
415
416 // Unroll this loop manually.
417 // for (const T& position : tls.positions) {...}
418
419 // Calculate the values from positions. Use an out param on GMP path.
420 if constexpr(FixedWidthIntegral<T>) {
421 tls.left_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count);
422 tls.right_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count);
423 } else if constexpr(GMPIntegral<T>) {
424 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count, tls.left_value);
425 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count, tls.right_value);
426 }
427
428 // Initialize the node and test HWM for ancestry.
429 tls.left_node.init(tls.left_value);
430 tls.right_node.init(tls.right_value);
431 if (_is_preserving_ancestors) {
432 if (tls.left_node.is_below_high_water_mark()) {
433 tls.new_ancestors.push_back(new Node<T>(tls.left_value));
434 }
435 if (tls.right_node.is_below_high_water_mark()) {
436 tls.new_ancestors.push_back(new Node<T>(tls.right_value));
437 }
438 }
439
440 // Perform verification, if requested.
441 if (_is_verifying_non_hwm_nodes) {
442 if (tls.left_node.is_below_high_water_mark() == false) {
443 if (Collatz<T>::st_verify(tls.left_value, add_level_high_water_mark) == false) {
444 throw std::logic_error("Node value " + to_string_any(tls.left_value) + " didn't verify. How?");
445 }
446 }
447 if (tls.right_node.is_below_high_water_mark() == false) {
448 if (Collatz<T>::st_verify(tls.right_value, add_level_high_water_mark) == false) {
449 throw std::logic_error("Node value " + to_string_any(tls.right_value) + " didn't verify. How?");
450 }
451 }
452 }
453
454 // Write the POSITION to the TLS bitmap if it didn't hit HWM.
455 if (tls.left_node.is_below_high_water_mark() == false) {
456 tls.uncovered_bitmap.add(tls.left_position);
457 }
458 if (tls.right_node.is_below_high_water_mark() == false) {
459 tls.uncovered_bitmap.add(tls.right_position);
460 }
461
462 // Return false. There is no stopping condition.
463 return false;
464 });
465
466 // Merge TLS data into the tree's trackers.
467 _uncovered_positions.clear();
468 for (const AddLevelTLS& storage : callback_storage) {
469 _uncovered_positions |= storage.uncovered_bitmap;
470 _ancestors.insert(_ancestors.end(), storage.new_ancestors.begin(), storage.new_ancestors.end());
471 }
472
473 // Always sort ancestors.
474 if (_is_preserving_ancestors) {
475 tbb::parallel_sort(_ancestors.begin(), _ancestors.end(), [](const Node<T>* a, const Node<T>* b) {
476 if constexpr(FixedWidthIntegral<T>) {
477 return a->get_value() < b->get_value();
478 } else if constexpr(GMPIntegral<T>) {
479 return mpz_cmp(a->get_value().get_mpz_t(), b->get_value().get_mpz_t()) < 0;
480 }
481 });
482 }
483
484 // Persist the coverage to the new level. It's just a sum of the uncovered size() values subtracted from the node total.
485 T total = BinaryTreeMath<T>::st_node_count_of_level(_level_count);
486 T uncovered = _uncovered_positions.cardinality();
487 _coverage_map[_level_count] = BinaryTreeCoverage<T>(total - uncovered, total);
488
489 // Invoke optimize above level 32.
490 if (_level_count > 32) {
491 _uncovered_positions.optimize();
492 }
493 }
