356 void add_level() {
357 // When there are no levels, this simply crafts a 0- or 1-based root node and manually sets level map and coverage.
358 if (_level_count == 0) {
359 _level_count = 1;
360 _root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
361 _coverage_map[_level_count].set_covered(0);
362 _coverage_map[_level_count].set_total(1);
363 _uncovered_positions.add(1);
364 return;
365 }
366
367 // Bump level count.
368 _level_count++;
369
370 // Establish TLS for our callback.
371 std::vector<AddLevelTLS> callback_storage;
372 // Track our own local High-Water Mark. Since nodes are not sequentially ordered, we can only set this to first node - 1.
373 T add_level_high_water_mark = BinaryTreeMath<T>::st_first_node_value_of_level(_level_count) - 1;
374 if (add_level_high_water_mark < 1) {
375 add_level_high_water_mark = 1;
376 }
377
378 // Now loop.
379 _uncovered_positions.for_each_transformer(BitmapTransformerPolicy::PARALLEL, callback_storage, [&](const T& cb_position, AddLevelTLS& tls) {
380 // Scale the uncovered position to this level. Make these TLS storage to avoid alloc() on GMP path.
381 if constexpr(FixedWidthIntegral<T>) {
382 tls.left_position = (cb_position << 1) - 1;
383 tls.right_position = cb_position << 1;
384 } else if constexpr(GMPIntegral<T>) {
385 mpz_mul_2exp(tls.left_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
386 mpz_sub_ui(tls.left_position.get_mpz_t(), tls.left_position.get_mpz_t(), 1);
387 mpz_mul_2exp(tls.right_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
388 }
389
390 // Unroll this loop manually.
391 // for (const T& position : tls.positions) {...}
392
393 // Calculate the values from positions. Use an out param on GMP path.
394 if constexpr(FixedWidthIntegral<T>) {
395 tls.left_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count);
396 tls.right_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count);
397 } else if constexpr(GMPIntegral<T>) {
398 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count, tls.left_value);
399 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count, tls.right_value);
400 }
401
402 // Initialize the node and test HWM for ancestry.
403 tls.left_node.init(tls.left_value);
404 tls.right_node.init(tls.right_value);
405 if (_is_preserving_ancestors) {
406 if (tls.left_node.is_below_high_water_mark()) {
407 tls.new_ancestors.push_back(new Node<T>(tls.left_value));
408 }
409 if (tls.right_node.is_below_high_water_mark()) {
410 tls.new_ancestors.push_back(new Node<T>(tls.right_value));
411 }
412 }
413
414 // Perform verification, if requested.
415 if (_is_verifying_non_hwm_nodes) {
416 if (tls.left_node.is_below_high_water_mark() == false) {
417 if (Collatz<T>::st_verify(tls.left_value, add_level_high_water_mark) == false) {
418 throw std::logic_error("Node value " + to_string_any(tls.left_value) + " didn't verify. How?");
419 }
420 }
421 if (tls.right_node.is_below_high_water_mark() == false) {
422 if (Collatz<T>::st_verify(tls.right_value, add_level_high_water_mark) == false) {
423 throw std::logic_error("Node value " + to_string_any(tls.right_value) + " didn't verify. How?");
424 }
425 }
426 }
427
428 // Write the POSITION to the TLS bitmap if it didn't hit HWM.
429 if (tls.left_node.is_below_high_water_mark() == false) {
430 tls.uncovered_bitmap.add(tls.left_position);
431 }
432 if (tls.right_node.is_below_high_water_mark() == false) {
433 tls.uncovered_bitmap.add(tls.right_position);
434 }
435
436 // Return false. There is no stopping condition.
437 return false;
438 });
439
440 // Merge TLS data into the tree's trackers.
441 _uncovered_positions.clear();
442 for (const AddLevelTLS& storage : callback_storage) {
443 _uncovered_positions |= storage.uncovered_bitmap;
444 _ancestors.insert(_ancestors.end(), storage.new_ancestors.begin(), storage.new_ancestors.end());
445 }
446
447 // Always sort ancestors.
448 if (_is_preserving_ancestors) {
449 tbb::parallel_sort(_ancestors.begin(), _ancestors.end(), [](const Node<T>* a, const Node<T>* b) {
450 if constexpr(FixedWidthIntegral<T>) {
451 return a->get_value() < b->get_value();
452 } else if constexpr(GMPIntegral<T>) {
453 return mpz_cmp(a->get_value().get_mpz_t(), b->get_value().get_mpz_t()) < 0;
454 }
455 });
456 }
457
458 // Persist the coverage to the new level. It's just a sum of the uncovered size() values subtracted from the node total.
459 T total = BinaryTreeMath<T>::st_node_count_of_level(_level_count);
460 T uncovered = _uncovered_positions.cardinality();
461 _coverage_map[_level_count] = BinaryTreeCoverage<T>(total - uncovered, total);
462
463 // Invoke optimize above level 32.
464 if (_level_count > 32) {
465 _uncovered_positions.optimize();
466 }
467 }
