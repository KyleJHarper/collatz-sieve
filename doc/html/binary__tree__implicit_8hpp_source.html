393 void add_level() {
394 // Ensure initialization is set.
395 _is_initialized = true;
396
397 // Clear the value bitmap since it would otherwise be stale/incorrect.
398 _uncovered_values.clear();
399
400 // When there are no levels, this simply crafts a 0- or 1-based root node and manually sets level map and coverage.
401 if (_level_count == 0) {
402 _level_count = 1;
403 _root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
404 _coverage_map[_level_count].set_covered(0);
405 _coverage_map[_level_count].set_total(1);
406 _uncovered_positions.add(1);
407 return;
408 }
409
410 // Bump level count.
411 _level_count++;
412
413 // Establish TLS for our callback.
414 std::vector<AddLevelTLS> callback_storage;
415 // Track our own local High-Water Mark. Since nodes are not sequentially ordered, we can only set this to first node - 1.
416 T add_level_high_water_mark = BinaryTreeMath<T>::st_first_node_value_of_level(_level_count) - 1;
417 if (add_level_high_water_mark < 1) {
418 add_level_high_water_mark = 1;
419 }
420
421 // Now loop.
422 _uncovered_positions.for_each_value_with_tls(ForEachPolicy::PARALLEL, callback_storage, [&](const T& cb_position, AddLevelTLS& tls) {
423 // Scale the uncovered position to this level. Make these TLS storage to avoid alloc() on GMP path.
424 if constexpr(FixedWidthIntegral<T>) {
425 tls.left_position = (cb_position << 1) - 1;
426 tls.right_position = cb_position << 1;
427 } else if constexpr(GMPIntegral<T>) {
428 mpz_mul_2exp(tls.left_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
429 mpz_sub_ui(tls.left_position.get_mpz_t(), tls.left_position.get_mpz_t(), 1);
430 mpz_mul_2exp(tls.right_position.get_mpz_t(), cb_position.get_mpz_t(), 1);
431 }
432
433 // Unroll this loop manually.
434 // for (const T& position : tls.positions) {...}
435
436 // Calculate the values from positions. Use an out param on GMP path.
437 if constexpr(FixedWidthIntegral<T>) {
438 tls.left_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count);
439 tls.right_value = BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count);
440 } else if constexpr(GMPIntegral<T>) {
441 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.left_position, _level_count, tls.left_value);
442 BinaryTreeMath<T>::st_node_value_by_position_and_level(tls.right_position, _level_count, tls.right_value);
443 }
444
445 // Initialize the node and test HWM for ancestry.
446 tls.left_node.init(tls.left_value);
447 tls.right_node.init(tls.right_value);
448 if (_is_preserving_ancestors) {
449 if (tls.left_node.is_below_high_water_mark()) {
450 tls.new_ancestors.push_back(new Node<T>(tls.left_value));
451 }
452 if (tls.right_node.is_below_high_water_mark()) {
453 tls.new_ancestors.push_back(new Node<T>(tls.right_value));
454 }
455 }
456
457 // Perform verification, if requested.
458 if (_is_verifying_non_hwm_nodes) {
459 if (tls.left_node.is_below_high_water_mark() == false) {
460 if (Collatz<T>::st_verify(tls.left_value, add_level_high_water_mark) == false) {
461 throw std::logic_error("Node value " + to_string_any(tls.left_value) + " didn't verify. How?");
462 }
463 }
464 if (tls.right_node.is_below_high_water_mark() == false) {
465 if (Collatz<T>::st_verify(tls.right_value, add_level_high_water_mark) == false) {
466 throw std::logic_error("Node value " + to_string_any(tls.right_value) + " didn't verify. How?");
467 }
468 }
469 }
470
471 // Write the POSITION to the TLS bitmap if it didn't hit HWM.
472 if (tls.left_node.is_below_high_water_mark() == false) {
473 tls.uncovered_bitmap.add(tls.left_position);
474 }
475 if (tls.right_node.is_below_high_water_mark() == false) {
476 tls.uncovered_bitmap.add(tls.right_position);
477 }
478
479 // Return continue. There is no stopping condition.
480 return ForEachSignal::CONTINUE;
481 });
482
483 // Merge TLS data into the tree's trackers.
484 _uncovered_positions.clear();
485 for (const AddLevelTLS& storage : callback_storage) {
486 _uncovered_positions |= storage.uncovered_bitmap;
487 _ancestors.insert(_ancestors.end(), storage.new_ancestors.begin(), storage.new_ancestors.end());
488 }
489
490 // Always sort ancestors.
491 if (_is_preserving_ancestors) {
492 tbb::parallel_sort(_ancestors.begin(), _ancestors.end(), [](const Node<T>* a, const Node<T>* b) {
493 if constexpr(FixedWidthIntegral<T>) {
494 return a->get_value() < b->get_value();
495 } else if constexpr(GMPIntegral<T>) {
496 return mpz_cmp(a->get_value().get_mpz_t(), b->get_value().get_mpz_t()) < 0;
497 }
498 });
499 }
500
501 // Persist the coverage to the new level. It's just a sum of the uncovered size() values subtracted from the node total.
502 T total = BinaryTreeMath<T>::st_node_count_of_level(_level_count);
503 T uncovered = _uncovered_positions.cardinality();
504 _coverage_map[_level_count] = BinaryTreeCoverage<T>(total - uncovered, total);
505
506 // Invoke optimize above level 32.
507 if (_level_count > 32) {
508 _uncovered_positions.optimize();
509 }
510 }

521 template<size_t BUFFER_SIZE = 1 << 24>
522 void generate_value_map(ForEachPolicy policy = ForEachPolicy::SERIAL) {
523 _uncovered_values.clear();
524
525 if (policy == ForEachPolicy::SERIAL) {
526 std::vector<T> buffer;
527 buffer.reserve(BUFFER_SIZE);
528 _uncovered_positions.for_each_value(ForEachPolicy::SERIAL, [&](const T& position) {
529 buffer.push_back(BinaryTreeMath<T>::st_node_value_by_position_and_level(position, _level_count));
530 if (buffer.size() >= BUFFER_SIZE) {
531 _uncovered_values.add_many(buffer.size(), buffer.data());
532 buffer.clear();
533 }
534 return ForEachSignal::CONTINUE;
535 });
536 _uncovered_values.add_many(buffer.size(), buffer.data());
537 } else {
538 struct ValueMapTLS {
539 NodeBitmap<T> bitmap;
540 std::vector<T> buffer;
541 };
542 std::vector<ValueMapTLS> tls_vector;
543 _uncovered_positions.for_each_value_with_tls(ForEachPolicy::PARALLEL, tls_vector, [&](const T& position, ValueMapTLS& tls) {
544 tls.buffer.push_back(BinaryTreeMath<T>::st_node_value_by_position_and_level(position, _level_count));
545 if (tls.buffer.size() >= BUFFER_SIZE) {
546 tls.bitmap.add_many(tls.buffer.size(), tls.buffer.data());
547 tls.buffer.clear();
548 }
549 return ForEachSignal::CONTINUE;
550 });
551 // Now merge bitmaps and remaining buffers.
552 for (ValueMapTLS& callback_tls : tls_vector) {
553 _uncovered_values |= callback_tls.bitmap;
554 callback_tls.bitmap.clear();
555 _uncovered_values.add_many(callback_tls.buffer.size(), callback_tls.buffer.data());
556 }
557 }
558
559 // Optimize
560 _uncovered_values.optimize();
561 }
