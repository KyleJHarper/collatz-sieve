105 void init(const T& value, Node *parent = nullptr) {
106 // Reset object if necessary.
107 if(_is_initialized) { reset(); }
108
109 // Establish or clear metadata object. Reset() already cleared it, if it existed.
110 _is_initialized = true;
111 if constexpr(FixedWidthIntegral<T>) {
112 _value = value;
113 } else if constexpr (GMPIntegral<T>) {
114 mpz_set(_value.get_mpz_t(), value.get_mpz_t());
115 }
116 _parent = parent;
117
118 // Call a helper with a promoted type to process the sequence and build our FG/HWM info. Doing this allows trees to build
119 // up to 2^<bit-1>.
120 if constexpr(FixedWidthIntegral<T>) {
121 if constexpr (sizeof(T) * 8 <= 64) {
122 // Dealing with T of 64 bits or less. If it'll overflow, use 128-bit.
123 if (_value <= CollatzConstants::get_max_initial_value_by_bit<uint64_t>(64)) {
124 init_sequence_helper<uint64_t>();
125 } else {
126 init_sequence_helper<uint128_t>();
127 }
128 } else if constexpr(sizeof(T) * 8 == 128) {
129 // Dealing with 128 bits. If it'll overflow, use mpz_class.
130 if (_value <= CollatzConstants::get_max_initial_value_by_bit<uint128_t>(128)) {
131 init_sequence_helper<uint128_t>();
132 } else {
133 init_sequence_helper<mpz_class>();
134 }
135 }
136 } else if constexpr(GMPIntegral<T>) {
137 // Dealing with mpz_class. Send as-is.
138 init_sequence_helper<mpz_class>();
139 }
140
141 // Use our parent to decide who the high-water mark ancestor is, if any. Since it's a lineage, there's no
142 // reason to scan everything manually. The parent's data is all we need.
143 if (parent != nullptr) {
144 // Assign whoever the ancestor is, even if it's nullptr.
145 _hwm_ancestor = parent->get_hwm_ancestor();
146 // If we have an ancestor, just flag our boolean.
147 if (_hwm_ancestor != nullptr) {
148 _has_hwm_ancestor = true;
149 }
150 // If we don't have an ancestor, see if the parent can be.
151 if (_hwm_ancestor == nullptr && parent->is_below_high_water_mark()) {
152 _hwm_ancestor = parent;
153 _has_hwm_ancestor = true;
154 }
155 }
156 };

164 template<AnySupportedIntegral U>
165 inline void init_sequence_helper() {
166 // Cache the FG chain length.
167 const size_t fg_chain_length = get_fg_chain_length();
168
169 // We need to elevate _value to type U in case it's not the same.
170 static thread_local U u_value;
171 static thread_local U u_current_value;
172 if constexpr(FixedWidthIntegral<T> && FixedWidthIntegral<U>) {
173 // Both types are fixed-width. They can cast directly, even if they're the same. Dirt cheap, so don't overthink this.
174 u_value = U(_value);
175 u_current_value = u_value;
176 } else if constexpr(FixedWidthIntegral<T> && GMPIntegral<U>) {
177 // Node is fixed-width (probably uint128_t) but required U is mpz_class. Elevate.
178 Int128::uint128_to_mpz(_value, u_value);
179 u_current_value = u_value;
180 } else if constexpr(GMPIntegral<T> && GMPIntegral<U>) {
181 // Both are GMP. Just set, which is cheap.
182 mpz_set(u_value.get_mpz_t(), _value.get_mpz_t());
183 mpz_set(u_current_value.get_mpz_t(), u_value.get_mpz_t());
184 }
185
186 // Now loop and stride.
187 using StrideTable = std::conditional_t<
188 FixedWidthIntegral<U>
189 , AffineStride::NodeInitFWTable
190 , AffineStride::NodeInitGMPTable
191 >;
192 size_t strides = fg_chain_length / StrideTable::STRIDE_SIZE;
193 size_t steps_taken = strides * StrideTable::STRIDE_SIZE;
194 while (strides > 0) {
195 StrideTable::apply_stride(u_current_value);
196 strides -= 1;
197 }
198
199 // Mop up leftovers after striding.
200 while (steps_taken < fg_chain_length) {
201 // We're going to shift no matter what, so all we're checking for is odd.
202 if constexpr(FixedWidthIntegral<T>) {
203 // If it's odd, F step.
204 if (u_current_value % 2 == 1) {
205 u_current_value = (u_current_value << 1) + u_current_value + 1;
206 }
207 } else if constexpr (GMPIntegral<T>) {
208 // If it's odd, F step.
209 if (mpz_odd_p(u_current_value.get_mpz_t())) {
210 mpz_mul(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), CollatzConstants::MPZ_THREE.get_mpz_t());
211 mpz_add(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), CollatzConstants::MPZ_ONE.get_mpz_t());
212 }
213 }
214 // Always even at this point. Shift by CTZ, clamped by FG chain length.
215 // This technique isn't faster for 64/128-bit types, but it's faster for GMP, so we'll do it.
216 size_t zeros_count = std::min(static_cast<size_t>(Bit::count_trailing_zeros(u_current_value)), fg_chain_length - steps_taken);
217 if constexpr(FixedWidthIntegral<T>) {
218 u_current_value >>= zeros_count;
219 } else {
220 mpz_tdiv_q_2exp(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), zeros_count);
221 }
222 steps_taken += zeros_count;
223 }
224
225 // Now compare the current value to our original value.
226 _is_below_hwm = u_current_value < u_value;
227 }

285 void release_child(Node<T>* child) {
286 for (size_t i = 0; i < MAX_CHILDREN; i++) {
287 // Find the Child
288 if (_children[i] == child) {
289 // Delete if owned.
290 if (_owns_children) {
291 delete _children[i];
292 }
293 // Null out our reference.
294 _children[i] = nullptr;
295 // Push any other children toward the front.
296 for (size_t j = i + 1; j < MAX_CHILDREN; j++) {
297 _children[j - 1] = _children[j];
298 }
299 // Decrement the count and leave.
300 _child_count--;
301 // Set the last element to nullptr.
302 _children[_child_count] = nullptr;
303 break;
304 }
305 }
306 }

477 static bool st_equal(const Node<T>& first, const Node<T>& second, std::string* err = nullptr) {
478 EqualityHelper eq(err);
479 eq.set_category("Node");
480
481 // Values
482 if (! eq.equal(first.get_value(), second.get_value())) {
483 return eq.fail("Node values mismatch");
484 }
485
486 // Parents
487 if (! eq.pointers_null_agree(first.get_parent(), second.get_parent())) {
488 return eq.fail("Parent nodes' pointers don't agree on null state");
489 }
490 if (first.get_parent() != nullptr) {
491 if (! eq.equal(first.get_parent()->get_value(), second.get_parent()->get_value())) {
492 return eq.fail("Nodes' parents' values mismatch");
493 }
494 }
495
496 // HWM Ancestor
497 if (! eq.equal(first.has_high_water_mark_ancestor(), second.has_high_water_mark_ancestor())) {
498 return eq.fail("Has HWM ancestor mismatch");
499 }
500 if (! eq.pointers_null_agree(first.get_hwm_ancestor(), second.get_hwm_ancestor())) {
501 return eq.fail("HWM ancestors don't agree on null state");
502 }
503 if (first.get_hwm_ancestor() != nullptr) {
504 if (! eq.equal(first.get_hwm_ancestor()->get_value(), second.get_hwm_ancestor()->get_value())) {
505 return eq.fail("HWM ancestor's value mismatch");
506 }
507 }
508
509 // Children
510 if (! eq.equal(first.get_child_count(), second.get_child_count())) {
511 return eq.fail("Node child counts mismatch");
512 }
513 if (! eq.equal(first.does_own_children(), second.does_own_children())) {
514 return eq.fail("Nodes' down_own_children mismatch");
515 }
516 for (size_t child_id = 0; child_id < first.get_child_count(); child_id++) {
517 if (! eq.pointers_null_agree(first.get_child(child_id), second.get_child(child_id))) {
518 return eq.fail("Child ID " + to_string_any(child_id) + " mismatch on null state");
519 }
520 if (first.get_child(child_id) != nullptr) {
521 if (! eq.equal(first.get_child(child_id)->get_value(), second.get_child(child_id)->get_value())) {
522 return eq.fail("Child ID " + to_string_any(child_id) + " values mismatch");
523 }
524 // Ensure each child knows we are the parent.
525 if (! eq.same_address(first.get_child(child_id)->get_parent(), &first)) {
526 return eq.fail("First's node with Child ID " + to_string_any(child_id) + " doesn't have link back to parent");
527 }
528 if (! eq.same_address(second.get_child(child_id)->get_parent(), &second)) {
529 return eq.fail("Second's node with Child ID " + to_string_any(child_id) + " doesn't have link back to parent");
530 }
531 }
532 }
533
534 // Remaining flags.
535 if (! eq.equal(first.is_below_high_water_mark(), second.is_below_high_water_mark())) {
536 return eq.fail("Nodes' is_below_high_water_mark mismatch");
537 }
538 if (! eq.equal(first.is_initialized(), second.is_initialized())) {
539 return eq.fail("Nodes' is_initialize mismatch");
540 }
541
542 // FG Chain Bits
543 if (! eq.equal(first.get_fg_chain_length(), second.get_fg_chain_length())) {
544 return eq.fail("Nodes' FG chain length mismatch");
545 }
546 if (! eq.equal(first.get_fg_chain_string(), second.get_fg_chain_string())) {
547 return eq.fail("Nodes' FG chain string mismatch");
548 }
549
550 // All good.
551 return true;
552 }

566 static bool st_equal(const Node<T>* first, const Node<T>* second, std::string* err = nullptr) {
567 EqualityHelper eq(err);
568 eq.set_category("Node");
569
570 // They should both be real objects or nullptr.
571 // When they disagree because one is real and one is a nullptr, fail.
572 if (! eq.pointers_null_agree(first, second)) {
573 return eq.fail("Node objects don't agree on null state");
574 }
575
576 // If they're null, we're done.
577 if (first == nullptr) {
578 return true;
579 }
580
581 // If they both have the same value, they're either both nullptr or pointing to the same node, which is fine.
582 if (first == second) {
583 return true;
584 }
585
586 // At this point, we have two objects that null-agree but are different, so they must be real objects. We have to assume
587 // they might be coming from different collections or trees, so we have to compare values, not pointers.
588 return Node<T>::st_equal(*first, *second);
589 }

624 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
625 StreamHelper sh(nullptr, &out, err);
626 sh.set_category("Node");
627
628 // Value
629 if (! sh.serialize_integral(_value)) {
630 return sh.fail(" _value==" + to_string_any(_value));
631 }
632
633 // Parent
634 if (_parent == nullptr) {
635 if (! sh.serialize_integral(T(0))) {
636 return sh.fail("no parent (aka: T(0))");
637 }
638 } else {
639 if (! sh.serialize_integral(_parent->get_value())) {
640 return sh.fail("_parent->get_value()==" + to_string_any(_parent->get_value()));
641 }
642 }
643
644 // HWM Ancestor
645 if (_hwm_ancestor == nullptr) {
646 if (! sh.serialize_integral(T(0))) {
647 return sh.fail("no hwm_ancestor (aka: T(0))");
648 }
649 } else {
650 if (! sh.serialize_integral(_hwm_ancestor->get_value())) {
651 return sh.fail("_hwm_ancestor->get_value()==" + to_string_any(_hwm_ancestor->get_value()));
652 }
653 }
654
655 // Children
656 // We don't actually need these. We rebuild (deserialize) from the parents down to the children, therefore reconstruction
657 // is merely adding a node, linking to the parent value (which we wrote above), and then telling that parent to own "this".
658
659 // Flags
660 bool b_is_below_hwm = _is_below_hwm;
661 bool b_has_hwm_ancestor = _has_hwm_ancestor;
662 bool b_is_initialized = _is_initialized;
663 bool b_owns_children = _owns_children;
664 if (! sh.serialize_bool(b_is_below_hwm)) {
665 return sh.fail("_is_below_hwm==" + std::to_string(b_is_below_hwm));
666 }
667 if (! sh.serialize_bool(b_has_hwm_ancestor)) {
668 return sh.fail("_has_hwm_ancestor==" + std::to_string(b_has_hwm_ancestor));
669 }
670 if (! sh.serialize_bool(b_is_initialized)) {
671 return sh.fail("_is_initialized==" + std::to_string(b_is_initialized));
672 }
673 if (! sh.serialize_bool(b_owns_children)) {
674 return sh.fail("_owns_children==" + std::to_string(b_owns_children));
675 }
676
677 // Child Count
678 if (! sh.serialize_integral(_child_count)) {
679 return sh.fail("_child_count==" + to_string_any(_child_count));
680 }
681
682 // All Good
683 return true;
684 }

701 [[nodiscard]] bool deserialize(std::istream& in, T& parent_v, T& hwm_ancestor_v, uint8_t& child_count, std::string* err = nullptr) {
702 StreamHelper sh(&in, nullptr, err);
703 sh.set_category("Node");
704
705 // Reset
706 reset();
707
708 // Value
709 if (! sh.deserialize_integral(_value)) {
710 return sh.fail("couldn't read _value");
711 }
712
713 // Parent
714 if (! sh.deserialize_integral(parent_v)) {
715 return sh.fail("couldn't read _parent's value");
716 }
717
718 // HWM Ancestor
719 if (! sh.deserialize_integral(hwm_ancestor_v)) {
720 return sh.fail("couldn't read hwm ancestor's value");
721 }
722
723 // Flags
724 // We use bit fields, which cannot bind to bool&. So we need a tmp.
725 bool b_tmp;
726 if (! sh.deserialize_bool(b_tmp)) {
727 return sh.fail("couldn't read _is_below_hwm");
728 }
729 _is_below_hwm = b_tmp;
730 if (! sh.deserialize_bool(b_tmp)) {
731 return sh.fail("couldn't read _has_hwm_ancestor");
732 }
733 _has_hwm_ancestor = b_tmp;
734 if (! sh.deserialize_bool(b_tmp)) {
735 return sh.fail("couldn't read _is_initialized");
736 }
737 _is_initialized = b_tmp;
738 if (! sh.deserialize_bool(b_tmp)) {
739 return sh.fail("couldn't read _owns_children");
740 }
741 _owns_children = b_tmp;
742
743 // Child count is returned, because deserializing a node doesn't actually make the kids.
744 if (! sh.deserialize_integral(child_count)) {
745 return sh.fail("couldn't read child_count");
746 }
747 _child_count = 0;
748
749 // All good
750 return true;
751 }
