90 void init(const T& value, Node *parent = nullptr) {
91 // Reset object if necessary.
92 if(_is_initialized) { reset(); }
93
94 // Establish or clear metadata object. Reset() already cleared it, if it existed.
95 _is_initialized = true;
96 if constexpr(BuiltinIntegral<T>) {
97 _value = value;
98 } else {
99 mpz_set(_value.get_mpz_t(), value.get_mpz_t());
100 }
101 _parent = parent;
102
103 // Call a helper with a promoted type to process the sequence and build our FG/HWM info. Doing this allows trees to build
104 // up to 2^<bit-1>.
105 if constexpr(NativeIntegral<T>) {
106 // Dealing with T of 64 bits or less. If it'll overflow, use 128-bit.
107 if (_value <= CollatzConstants::get_max_initial_value_by_bit<uint64_t>(64)) {
108 init_sequence_helper<uint64_t>();
109 } else {
110 init_sequence_helper<uint128_t>();
111 }
112 } else if constexpr(ExtendedIntegral<T>) {
113 // Dealing with 128 bits. If it'll overflow, use mpz_class.
114 if (_value <= CollatzConstants::get_max_initial_value_by_bit<uint128_t>(128)) {
115 init_sequence_helper<uint128_t>();
116 } else {
117 init_sequence_helper<mpz_class>();
118 }
119 } else {
120 // Dealing with mpz_class. Send as-is.
121 init_sequence_helper<mpz_class>();
122 }
123
124 // Use our parent to decide who the high-water mark ancestor is, if any. Since it's a lineage, there's no
125 // reason to scan everything manually. The parent's data is all we need.
126 if (parent != nullptr) {
127 // Assign whoever the ancestor is, even if it's nullptr.
128 _hwm_ancestor = parent->get_hwm_ancestor();
129 // If we have an ancestor, just flag our boolean.
130 if (_hwm_ancestor != nullptr) {
131 _has_hwm_ancestor = true;
132 }
133 // If we don't have an ancestor, see if the parent can be.
134 if (_hwm_ancestor == nullptr && parent->is_below_high_water_mark()) {
135 _hwm_ancestor = parent;
136 _has_hwm_ancestor = true;
137 }
138 }
139 };

147 template<AnySupportedIntegral U>
148 inline void init_sequence_helper() {
149 // Cache the FG chain length.
150 const size_t fg_chain_length = get_fg_chain_length();
151
152 // We need to elevate _value to type U in case it's not the same.
153 static thread_local U u_value;
154 static thread_local U u_current_value;
155 if constexpr(BuiltinIntegral<T> && BuiltinIntegral<U>) {
156 // Both types are builtin. They can cast directly, even if they're the same. Dirt cheap, so don't overthink this.
157 u_value = U(_value);
158 u_current_value = u_value;
159 } else if constexpr(ExtendedIntegral<T> && GMPIntegral<U>) {
160 // Node is uint128_t but required U is mpz_class. Elevate.
161 uint128_to_mpz(_value, u_value);
162 u_current_value = u_value;
163 } else if constexpr(GMPIntegral<T> && GMPIntegral<U>) {
164 // Both are GMP. Just set, which is cheap.
165 mpz_set(u_value.get_mpz_t(), _value.get_mpz_t());
166 mpz_set(u_current_value.get_mpz_t(), u_value.get_mpz_t());
167 }
168
169 // Now loop and stride.
170 size_t strides = fg_chain_length / AffineStride::STRIDE_SIZE;
171 size_t steps_taken = strides * AffineStride::STRIDE_SIZE;
172 while (strides > 0) {
173 AffineStride::apply_stride(u_current_value);
174 strides -= 1;
175 }
176
177 // Mop up leftovers after striding.
178 while (steps_taken < fg_chain_length) {
179 // We're going to shift no matter what, so all we're checking for is odd.
180 if constexpr(BuiltinIntegral<T>) {
181 // If it's odd, F step.
182 if (u_current_value % 2 == 1) {
183 u_current_value = (u_current_value << 1) + u_current_value + 1;
184 }
185 } else {
186 // If it's odd, F step.
187 if (mpz_odd_p(u_current_value.get_mpz_t())) {
188 mpz_mul(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), CollatzConstants::MPZ_THREE.get_mpz_t());
189 mpz_add(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), CollatzConstants::MPZ_ONE.get_mpz_t());
190 }
191 }
192 // Always even at this point. Shift by CTZ, clamped by FG chain length.
193 // This technique isn't faster for 64/128-bit types, but it's faster for GMP, so we'll do it.
194 size_t zeros_count = std::min(static_cast<size_t>(count_trailing_zeros(u_current_value)), fg_chain_length - steps_taken);
195 if constexpr(BuiltinIntegral<T>) {
196 u_current_value >>= zeros_count;
197 } else {
198 mpz_tdiv_q_2exp(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), zeros_count);
199 }
200 steps_taken += zeros_count;
201 }
202
203 // Now compare the current value to our original value.
204 _is_below_hwm = u_current_value < u_value;
205 }

262 void release_child(Node<T>* child) {
263 for (size_t i = 0; i < MAX_CHILDREN; i++) {
264 // Find the Child
265 if (_children[i] == child) {
266 // Delete if owned.
267 if (_owns_children) {
268 delete _children[i];
269 }
270 // Null out our reference.
271 _children[i] = nullptr;
272 // Push any other children toward the front.
273 for (size_t j = i + 1; j < MAX_CHILDREN; j++) {
274 _children[j - 1] = _children[j];
275 }
276 // Decrement the count and leave.
277 _child_count--;
278 break;
279 }
280 }

459 static bool st_equal (const Node<T>* first, const Node<T>* second, std::string* err = nullptr) {
460 EqualityHelper eq(err);
461 eq.set_category("Node");
462
463 // They should both be real objects or nullptr.
464 // When they disagree because one is real and one is a nullptr, fail.
465 if (! eq.pointers_null_agree(first, second)) {
466 return eq.fail("Node objects don't agree on null state");
467 }
468
469 // If they're null, we're done.
470 if (first == nullptr) {
471 return true;
472 }
473
474 // If they both have the same value, they're either both nullptr or pointing to the same node, which is fine.
475 if (first == second) {
476 return true;
477 }
478
479 // At this point, we have two objects that null-agree but are different, so they must be real objects. We have to assume
480 // they might be coming from different collections or trees, so we have to compare values, not pointers.
481
482 // Values
483 if (eq.unequal(first->get_value(), second->get_value())) {
484 return eq.fail("Node values mismatch");
485 }
486
487 // Parents
488 if (! eq.pointers_null_agree(first->get_parent(), second->get_parent())) {
489 return eq.fail("Parent nodes' pointers don't agree on null state");
490 }
491 if (first->get_parent() != nullptr) {
492 if (eq.unequal(first->get_parent()->get_value(), second->get_parent()->get_value())) {
493 return eq.fail("Nodes' parents' values mismatch");
494 }
495 }
496
497 // HWM Ancestor
498 if (eq.unequal(first->has_high_water_mark_ancestor(), second->has_high_water_mark_ancestor())) {
499 return eq.fail("Has HWM ancestor mismatch");
500 }
501 if (! eq.pointers_null_agree(first->get_hwm_ancestor(), second->get_hwm_ancestor())) {
502 return eq.fail("HWM ancestors don't agree on null state");
503 }
504 if (first->get_hwm_ancestor() != nullptr) {
505 if (eq.unequal(first->get_hwm_ancestor()->get_value(), second->get_hwm_ancestor()->get_value())) {
506 return eq.fail("HWM ancestor's value mismatch");
507 }
508 }
509
510 // Children
511 if (eq.unequal(first->get_child_count(), second->get_child_count())) {
512 return eq.fail("Node child counts mismatch");
513 }
514 if (eq.unequal(first->does_own_children(), second->does_own_children())) {
515 return eq.fail("Nodes' down_own_children mismatch");
516 }
517 for (size_t child_id = 0; child_id < first->get_child_count(); child_id++) {
518 if (! eq.pointers_null_agree(first->get_child(child_id), second->get_child(child_id))) {
519 return eq.fail("Child ID " + to_string_any(child_id) + " mismatch on null state");
520 }
521 if (first->get_child(child_id) != nullptr) {
522 if (eq.unequal(first->get_child(child_id)->get_value(), second->get_child(child_id)->get_value())) {
523 return eq.fail("Child ID " + to_string_any(child_id) + " values mismatch");
524 }
525 // Ensure each child knows we are the parent.
526 if (! eq.same_address(first->get_child(child_id)->get_parent(), first)) {
527 return eq.fail("First's node with Child ID " + to_string_any(child_id) + " doesn't have link back to parent");
528 }
529 if (! eq.same_address(second->get_child(child_id)->get_parent(), second)) {
530 return eq.fail("Second's node with Child ID " + to_string_any(child_id) + " doesn't have link back to parent");
531 }
532 }
533 }
534
535 // Remaining flags.
536 if (eq.unequal(first->is_below_high_water_mark(), second->is_below_high_water_mark())) {
537 return eq.fail("Nodes' is_below_high_water_mark mismatch");
538 }
539 if (eq.unequal(first->is_initialized(), second->is_initialized())) {
540 return eq.fail("Nodes' is_initialize mismatch");
541 }
542
543 // FG Chain Bits
544 if (eq.unequal(first->get_fg_chain_length(), second->get_fg_chain_length())) {
545 return eq.fail("Nodes' FG chain length mismatch");
546 }
547 if (eq.unequal(first->get_fg_chain_string(), second->get_fg_chain_string())) {
548 return eq.fail("Nodes' FG chain string mismatch");
549 }
550
551 // Guess we made it here. All good.
552 return true;
553 }

588 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
589 StreamHelper sh(nullptr, &out, err);
590 sh.set_category("Node");
591
592 // Value
593 if (! sh.serialize_integral(_value)) {
594 return sh.fail(" _value==" + to_string_any(_value));
595 }
596
597 // Parent
598 if (_parent == nullptr) {
599 if (! sh.serialize_integral(T(0))) {
600 return sh.fail("no parent (aka: T(0))");
601 }
602 } else {
603 if (! sh.serialize_integral(_parent->get_value())) {
604 return sh.fail("_parent->get_value()==" + to_string_any(_parent->get_value()));
605 }
606 }
607
608 // HWM Ancestor
609 if (_hwm_ancestor == nullptr) {
610 if (! sh.serialize_integral(T(0))) {
611 return sh.fail("no hwm_ancestor (aka: T(0))");
612 }
613 } else {
614 if (! sh.serialize_integral(_hwm_ancestor->get_value())) {
615 return sh.fail("_hwm_ancestor->get_value()==" + to_string_any(_hwm_ancestor->get_value()));
616 }
617 }
618
619 // Children
620 // We don't actually need these. We rebuild (deserialize) from the parents down to the children, therefore reconstruction
621 // is merely adding a node, linking to the parent value (which we wrote above), and then telling that parent to own "this".
622
623 // Flags
624 bool b_is_below_hwm = _is_below_hwm;
625 bool b_has_hwm_ancestor = _has_hwm_ancestor;
626 bool b_is_initialized = _is_initialized;
627 bool b_owns_children = _owns_children;
628 if (! sh.serialize_bool(b_is_below_hwm)) {
629 return sh.fail("_is_below_hwm==" + std::to_string(b_is_below_hwm));
630 }
631 if (! sh.serialize_bool(b_has_hwm_ancestor)) {
632 return sh.fail("_has_hwm_ancestor==" + std::to_string(b_has_hwm_ancestor));
633 }
634 if (! sh.serialize_bool(b_is_initialized)) {
635 return sh.fail("_is_initialized==" + std::to_string(b_is_initialized));
636 }
637 if (! sh.serialize_bool(b_owns_children)) {
638 return sh.fail("_owns_children==" + std::to_string(b_owns_children));
639 }
640
641 // Child Count
642 if (! sh.serialize_integral(_child_count)) {
643 return sh.fail("_child_count==" + to_string_any(_child_count));
644 }
645
646 // All Good
647 return true;
648 }

665 [[nodiscard]] bool deserialize(std::istream& in, T& parent_v, T& hwm_ancestor_v, uint8_t& child_count, std::string* err = nullptr) {
666 StreamHelper sh(&in, nullptr, err);
667 sh.set_category("Node");
668
669 // Reset
670 reset();
671
672 // Value
673 if (! sh.deserialize_integral(_value)) {
674 return sh.fail("couldn't read _value");
675 }
676
677 // Parent
678 if (! sh.deserialize_integral(parent_v)) {
679 return sh.fail("couldn't read _parent's value");
680 }
681
682 // HWM Ancestor
683 if (! sh.deserialize_integral(hwm_ancestor_v)) {
684 return sh.fail("couldn't read hwm ancestor's value");
685 }
686
687 // Flags
688 // We use bit fields, which cannot bind to bool&. So we need a tmp.
689 bool b_tmp;
690 if (! sh.deserialize_bool(b_tmp)) {
691 return sh.fail("couldn't read _is_below_hwm");
692 }
693 _is_below_hwm = b_tmp;
694 if (! sh.deserialize_bool(b_tmp)) {
695 return sh.fail("couldn't read _has_hwm_ancestor");
696 }
697 _has_hwm_ancestor = b_tmp;
698 if (! sh.deserialize_bool(b_tmp)) {
699 return sh.fail("couldn't read _is_initialized");
700 }
701 _is_initialized = b_tmp;
702 if (! sh.deserialize_bool(b_tmp)) {
703 return sh.fail("couldn't read _owns_children");
704 }
705 _owns_children = b_tmp;
706
707 // Child count is returned, because deserializing a node doesn't actually make the kids.
708 if (! sh.deserialize_integral(child_count)) {
709 return sh.fail("couldn't read child_count");
710 }
711 _child_count = 0;
712
713 // All good
714 return true;
715 }
