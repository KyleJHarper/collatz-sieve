19//
20// Node
21// Nodes are the objects within the BinaryTree, which allow us to track the odd-even/f-g patterns and give us the information we
22// need without a full Collatz object.
23//
24// BIG FAT NOTE!
25// By default, we OWN the children. If you delete a node, it will delete the children, which will cascade. If you want to delete
26// a node without affecting its children, call own_children(false).
27//
28template <AnySupportedIntegral T>
29class Node {
30 private:
31 static constexpr size_t MAX_CHILDREN = 2;
32 // Object members.
33 // Memory packing and alignment matter! Keep this class LIGHT.
34 // All data must fit within one cache line.
35 // uint64_t | total | uint128_t | total | mpz_class | total
36 T _value; // 8 | 8 | 16 | 16 | 16 | 16
37 Node *_parent = nullptr; // 8 | 16 | 8 | 24 | 8 | 24
38 Node *_hwm_ancestor = nullptr; // 8 | 24 | 8 | 32 | 8 | 32
39 Node *_children[MAX_CHILDREN] = {nullptr, nullptr}; // 16 | 40 | 16 | 48 | 16 | 48
40 bool _is_below_hwm : 1 = false; // 1:1 | 41 | 1:1 | 49 | 1:1 | 49
41 bool _has_hwm_ancestor : 1 = false; // 1:2 | 41 | 1:2 | 49 | 1:2 | 49
42 bool _is_initialized : 1 = false; // 1:3 | 41 | 1:3 | 49 | 1:3 | 49
43 bool _owns_children : 1 = true; // 1:4 | 41 | 1:4 | 49 | 1:4 | 49 (4 bits padding)
44 uint8_t _child_count = 0; // 1 | 42 | 1 | 50 | 1 | 50
45 // Alignment Padding (u128 is 16 bytes) // 6 | 48 | 14 | 64 | 6 | 56
46 // Free Padding to Cacheline // 16 | 64 | 0 | 64 | 8 | 64
47 // -- Cache Line --

83 //
84 // Initialize
85 // Builds the object, reusing it if necessary.
86 //
87 void init(const T& value, Node *parent = nullptr) {
88 // Reset object if necessary.
89 if(_is_initialized) { reset(); }
90
91 // Establish or clear metadata object. Reset() already cleared it, if it existed.
92 _is_initialized = true;
93 if constexpr(BuiltinIntegral<T>) {
94 _value = value;
95 } else {
96 mpz_set(_value.get_mpz_t(), value.get_mpz_t());
97 }
98 _parent = parent;
99
100 // Call a helper with a promoted type to process the sequence and build our FG/HWM info. Doing this allows trees to build
101 // up to 2^<bit-1>.
102 if constexpr(NativeIntegral<T>) {
103 // Dealing with T of 64 bits or less. If it'll overflow, use 128-bit.
104 if (_value <= CollatzConstants::get_max_initial_value_by_bit<T>(64)) {
105 init_sequence_helper<uint64_t>();
106 } else {
107 init_sequence_helper<uint128_t>();
108 }
109 } else if constexpr(ExtendedIntegral<T>) {
110 // Dealing with 128 bits. If it'll overflow, use mpz_class.
111 if (_value <= CollatzConstants::get_max_initial_value_by_bit<T>(128)) {
112 init_sequence_helper<uint128_t>();
113 } else {
114 init_sequence_helper<mpz_class>();
115 }
116 } else {
117 // Dealing with mpz_class. Send as-is.
118 init_sequence_helper<mpz_class>();
119 }
120
121 // Use our parent to decide who the high-water mark ancestor is, if any. Since it's a lineage, there's no
122 // reason to scan everything manually. The parent's data is all we need.
123 if (parent != nullptr) {
124 // Assign whoever the ancestor is, even if it's nullptr.
125 _hwm_ancestor = parent->get_hwm_ancestor();
126 // If we have an ancestor, just flag our boolean.
127 if (_hwm_ancestor != nullptr) {
128 _has_hwm_ancestor = true;
129 }
130 // If we don't have an ancestor, see if the parent can be.
131 if (_hwm_ancestor == nullptr && parent->is_below_high_water_mark()) {
132 _hwm_ancestor = parent;
133 _has_hwm_ancestor = true;
134 }
135 }
136 };
137 //
138 // Quick helper to DRY up affine logic during init().
139 template<AnySupportedIntegral U>
140 inline void init_sequence_helper() {
141 // Cache the FG chain length.
142 const size_t fg_chain_length = get_fg_chain_length();
143
144 // We need to elevate _value to type U in case it's not the same.
145 static thread_local U u_value;
146 static thread_local U u_current_value;
147 if constexpr(BuiltinIntegral<T> && BuiltinIntegral<U>) {
148 // Both types are builtin. They can cast directly, even if they're the same. Dirt cheap, so don't overthink this.
149 u_value = U(_value);
150 u_current_value = u_value;
151 } else if constexpr(ExtendedIntegral<T> && GMPIntegral<U>) {
152 // Node is uint128_t but required U is mpz_class. Elevate.
153 uint128_to_mpz(_value, u_value);
154 u_current_value = u_value;
155 } else if constexpr(GMPIntegral<T> && GMPIntegral<U>) {
156 // Both are GMP. Just set, which is cheap.
157 mpz_set(u_value.get_mpz_t(), _value.get_mpz_t());
158 mpz_set(u_current_value.get_mpz_t(), u_value.get_mpz_t());
159 }
160
161 // Now loop and stride.
162 size_t strides = fg_chain_length / AffineStride::STRIDE_SIZE;
163 size_t steps_taken = strides * AffineStride::STRIDE_SIZE;
164 while (strides > 0) {
165 AffineStride::apply_stride(u_current_value);
166 strides -= 1;
167 }
168
169 // Mop up leftovers after striding.
170 while (steps_taken < fg_chain_length) {
171 // We're going to shift no matter what, so all we're checking for is odd.
172 if constexpr(BuiltinIntegral<T>) {
173 // If it's odd, F step.
174 if (u_current_value % 2 == 1) {
175 u_current_value = (u_current_value << 1) + u_current_value + 1;
176 }
177 } else {
178 // If it's odd, F step.
179 if (mpz_odd_p(u_current_value.get_mpz_t())) {
180 mpz_mul(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), CollatzConstants::MPZ_THREE.get_mpz_t());
181 mpz_add(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), CollatzConstants::MPZ_ONE.get_mpz_t());
182 }
183 }
184 // Always even at this point. Shift by CTZ, clamped by FG chain length.
185 // This technique isn't faster for 64/128-bit types, but it's faster for GMP, so we'll do it.
186 size_t zeros_count = std::min(static_cast<size_t>(count_trailing_zeros(u_current_value)), fg_chain_length - steps_taken);
187 if constexpr(BuiltinIntegral<T>) {
188 u_current_value >>= zeros_count;
189 } else {
190 mpz_tdiv_q_2exp(u_current_value.get_mpz_t(), u_current_value.get_mpz_t(), zeros_count);
191 }
192 steps_taken += zeros_count;
193 }
194
195 // Now compare the current value to our original value.
196 _is_below_hwm = u_current_value < u_value;
197 }

267 //
268 // Release Child
269 // Release (delete) a single child if _owns_children is true, based on the object passed in. Compares pointer value to memory.
270 //
271 void release_child(Node<T>* child) {
272 for (size_t i = 0; i < MAX_CHILDREN; i++) {
273 // Find the Child
274 if (_children[i] == child) {
275 // Delete if owned.
276 if (_owns_children) {
277 delete _children[i];
278 }
279 // Null out our reference.
280 _children[i] = nullptr;
281 // Push any other children toward the front.
282 for (size_t j = i + 1; j < MAX_CHILDREN; j++) {
283 _children[j - 1] = _children[j];
284 }
285 // Decrement the count and leave.
286 _child_count--;
287 break;
288 }
289 }
290 }

310 // Children are usually generated by a BinaryTree, but in free-form mode, allow a node to generate its own children.
311 static void st_spawn_children(Node<T>* node) {
312 if (node->get_child_count() > 0) {
313 throw std::logic_error("You can't spawn children on a node that already has children.");
314 }
315 T step = BinaryTreeMath<T>::st_step(node->get_level());
316
317 T child_value = node->get_value();
318 for (uint8_t i = 0; i < MAX_CHILDREN; i++) {
319 // Step forward. Avoid alloc() with GMP in the += operator.
320 if constexpr(BuiltinIntegral<T>) {
321 child_value += step;
322 } else if constexpr(GMPIntegral<T>) {
323 mpz_add(child_value.get_mpz_t(), child_value.get_mpz_t(), step.get_mpz_t());
324 }
325 Node<T>* child = new Node<T>(child_value, node);
326 node->assign_child(child);
327 }
328 }

391 //
392 // Equal
393 // Compares this node to "other". Returns true if equal.
394 //
395 // Returns true if they are equal in representation. False otherwise.
396 // Will explain what failed to *err if sent.
397 //
398 static bool st_equal (const Node<T>* first, const Node<T>* second, std::string* err = nullptr) {
399 EqualityHelper eq(err);
400 eq.set_category("Node");
401
402 // They should both be real objects or nullptr.
403 // When they disagree because one is real and one is a nullptr, fail.
404 if (! eq.pointers_null_agree(first, second)) {
405 return eq.fail("Node objects don't agree on null state");
406 }
407
408 // If they're null, we're done.
409 if (first == nullptr) {
410 return true;
411 }
412
413 // If they both have the same value, they're either both nullptr or pointing to the same node, which is fine.
414 if (first == second) {
415 return true;
416 }
417
418 // At this point, we have two objects that null-agree but are different, so they must be real objects. We have to assume
419 // they might be coming from different collections or trees, so we have to compare values, not pointers.
420
421 // Values
422 if (eq.unequal(first->get_value(), second->get_value())) {
423 return eq.fail("Node values mismatch");
424 }
425
426 // Parents
427 if (! eq.pointers_null_agree(first->get_parent(), second->get_parent())) {
428 return eq.fail("Parent nodes' pointers don't agree on null state");
429 }
430 if (first->get_parent() != nullptr) {
431 if (eq.unequal(first->get_parent()->get_value(), second->get_parent()->get_value())) {
432 return eq.fail("Nodes' parents' values mismatch");
433 }
434 }
435
436 // HWM Ancestor
437 if (eq.unequal(first->has_high_water_mark_ancestor(), second->has_high_water_mark_ancestor())) {
438 return eq.fail("Has HWM ancestor mismatch");
439 }
440 if (! eq.pointers_null_agree(first->get_hwm_ancestor(), second->get_hwm_ancestor())) {
441 return eq.fail("HWM ancestors don't agree on null state");
442 }
443 if (first->get_hwm_ancestor() != nullptr) {
444 if (eq.unequal(first->get_hwm_ancestor()->get_value(), second->get_hwm_ancestor()->get_value())) {
445 return eq.fail("HWM ancestor's value mismatch");
446 }
447 }
448
449 // Children
450 if (eq.unequal(first->get_child_count(), second->get_child_count())) {
451 return eq.fail("Node child counts mismatch");
452 }
453 if (eq.unequal(first->does_own_children(), second->does_own_children())) {
454 return eq.fail("Nodes' down_own_children mismatch");
455 }
456 for (size_t child_id = 0; child_id < first->get_child_count(); child_id++) {
457 if (! eq.pointers_null_agree(first->get_child(child_id), second->get_child(child_id))) {
458 return eq.fail("Child ID " + to_string_any(child_id) + " mismatch on null state");
459 }
460 if (first->get_child(child_id) != nullptr) {
461 if (eq.unequal(first->get_child(child_id)->get_value(), second->get_child(child_id)->get_value())) {
462 return eq.fail("Child ID " + to_string_any(child_id) + " values mismatch");
463 }
464 // Ensure each child knows we are the parent.
465 if (! eq.same_address(first->get_child(child_id)->get_parent(), first)) {
466 return eq.fail("First's node with Child ID " + to_string_any(child_id) + " doesn't have link back to parent");
467 }
468 if (! eq.same_address(second->get_child(child_id)->get_parent(), second)) {
469 return eq.fail("Second's node with Child ID " + to_string_any(child_id) + " doesn't have link back to parent");
470 }
471 }
472 }
473
474 // Remaining flags.
475 if (eq.unequal(first->is_below_high_water_mark(), second->is_below_high_water_mark())) {
476 return eq.fail("Nodes' is_below_high_water_mark mismatch");
477 }
478 if (eq.unequal(first->is_initialized(), second->is_initialized())) {
479 return eq.fail("Nodes' is_initialize mismatch");
480 }
481
482 // FG Chain Bits
483 if (eq.unequal(first->get_fg_chain_length(), second->get_fg_chain_length())) {
484 return eq.fail("Nodes' FG chain length mismatch");
485 }
486 if (eq.unequal(first->get_fg_chain_string(), second->get_fg_chain_string())) {
487 return eq.fail("Nodes' FG chain string mismatch");
488 }
489
490 // Guess we made it here. All good.
491 return true;
492 }

501 //
502 // Serialize
503 // Serialize the node with its value, flags, and child values. Since children are pointers, we will emit just their values.
504 // During deserialization, this is handled to re-establish pointers.
505 //
506 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
507 StreamHelper sh(nullptr, &out, err);
508 sh.set_category("Node");
509
510 // Value
511 if (! sh.serialize_integral(_value)) {
512 return sh.fail(" _value==" + to_string_any(_value));
513 }
514
515 // Parent
516 if (_parent == nullptr) {
517 if (! sh.serialize_integral(T(0))) {
518 return sh.fail("no parent (aka: T(0))");
519 }
520 } else {
521 if (! sh.serialize_integral(_parent->get_value())) {
522 return sh.fail("_parent->get_value()==" + to_string_any(_parent->get_value()));
523 }
524 }
525
526 // HWM Ancestor
527 if (_hwm_ancestor == nullptr) {
528 if (! sh.serialize_integral(T(0))) {
529 return sh.fail("no hwm_ancestor (aka: T(0))");
530 }
531 } else {
532 if (! sh.serialize_integral(_hwm_ancestor->get_value())) {
533 return sh.fail("_hwm_ancestor->get_value()==" + to_string_any(_hwm_ancestor->get_value()));
534 }
535 }
536
537 // Children
538 // We don't actually need these. We rebuild (deserialize) from the parents down to the children, therefore reconstruction
539 // is merely adding a node, linking to the parent value (which we wrote above), and then telling that parent to own "this".
540
541 // Flags
542 bool b_is_below_hwm = _is_below_hwm;
543 bool b_has_hwm_ancestor = _has_hwm_ancestor;
544 bool b_is_initialized = _is_initialized;
545 bool b_owns_children = _owns_children;
546 if (! sh.serialize_bool(b_is_below_hwm)) {
547 return sh.fail("_is_below_hwm==" + std::to_string(b_is_below_hwm));
548 }
549 if (! sh.serialize_bool(b_has_hwm_ancestor)) {
550 return sh.fail("_has_hwm_ancestor==" + std::to_string(b_has_hwm_ancestor));
551 }
552 if (! sh.serialize_bool(b_is_initialized)) {
553 return sh.fail("_is_initialized==" + std::to_string(b_is_initialized));
554 }
555 if (! sh.serialize_bool(b_owns_children)) {
556 return sh.fail("_owns_children==" + std::to_string(b_owns_children));
557 }
558
559 // Child Count
560 if (! sh.serialize_integral(_child_count)) {
561 return sh.fail("_child_count==" + to_string_any(_child_count));
562 }
563
564 // All Good
565 return true;
566 }

570 //
571 // Deserialize
572 // Converts data from "in" to this object. Returns the parent and hwm_ancestor values for later linking.
573 //
574 [[nodiscard]] bool deserialize(std::istream& in, T& parent_v, T& hwm_ancestor_v, uint8_t& child_count, std::string* err = nullptr) {
575 StreamHelper sh(&in, nullptr, err);
576 sh.set_category("Node");
577
578 // Reset
579 reset();
580
581 // Value
582 if (! sh.deserialize_integral(_value)) {
583 return sh.fail("couldn't read _value");
584 }
585
586 // Parent
587 if (! sh.deserialize_integral(parent_v)) {
588 return sh.fail("couldn't read _parent's value");
589 }
590
591 // HWM Ancestor
592 if (! sh.deserialize_integral(hwm_ancestor_v)) {
593 return sh.fail("couldn't read hwm ancestor's value");
594 }
595
596 // Flags
597 // We use bit fields, which cannot bind to bool&. So we need a tmp.
598 bool b_tmp;
599 if (! sh.deserialize_bool(b_tmp)) {
600 return sh.fail("couldn't read _is_below_hwm");
601 }
602 _is_below_hwm = b_tmp;
603 if (! sh.deserialize_bool(b_tmp)) {
604 return sh.fail("couldn't read _has_hwm_ancestor");
605 }
606 _has_hwm_ancestor = b_tmp;
607 if (! sh.deserialize_bool(b_tmp)) {
608 return sh.fail("couldn't read _is_initialized");
609 }
610 _is_initialized = b_tmp;
611 if (! sh.deserialize_bool(b_tmp)) {
612 return sh.fail("couldn't read _owns_children");
613 }
614 _owns_children = b_tmp;
615
616 // Child count is returned, because deserializing a node doesn't actually make the kids.
617 if (! sh.deserialize_integral(child_count)) {
618 return sh.fail("couldn't read child_count");
619 }
620 _child_count = 0;
621
622 // All good
623 return true;
624 }
