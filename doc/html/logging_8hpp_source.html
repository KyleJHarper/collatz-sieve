105 fill = maybe_fill;

107 break;
108 }

112 char type = '\0';

114 if (std::isalpha(static_cast<unsigned char>(*it))) { type = *it; break; }

118 bool neg = value < 0;
119 mpz_class absval = neg ? -value : value;
120 std::string digits;
121 if (type == 'x' || type == 'X') digits = absval.get_str(16);
122 else if (type == 'o') digits = absval.get_str(8);
123 else if (type == 'b') digits = absval.get_str(2);
124 else digits = absval.get_str(10);
