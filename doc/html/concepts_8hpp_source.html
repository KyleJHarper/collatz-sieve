40constexpr uint128_t parse_u128(std::string_view view) {
41 uint128_t value = 0;
42 for (char c : view) {
43 if (c < '0' || c > '9')
44 throw "invalid digit in _u128 literal";
45 value = value * 10 + (c - '0');

287inline std::string int128_to_string(int128_t value) {
288 if (value == 0) return "0";
289
290 bool negative = value < 0;
291 unsigned __int128 v = negative ? -value : value;
292
293 std::string result;
294 while (v > 0) {
295 int digit = v % 10;
296 result.push_back('0' + digit);
297 v /= 10;
298 }
299 if (negative) result.push_back('-');
300 std::reverse(result.begin(), result.end());
301 return result;
302}

330inline uint128_t str_to_uint128(const std::string& s) {
331 if (s.empty()) {
332 throw std::invalid_argument("Empty string for uint128_t");
333 }
334 uint128_t result = 0;
335 for (char c : s) {
336 if (!std::isdigit(c)) {
337 throw std::invalid_argument("Invalid character in uint128_t: " + s);
338 }
339 int digit = c - '0';
340 result = result * 10 + digit;
341 }
342 return result;
343}

394template <typename H>
395H AbslHashValue(H h, const mpz_class& value) {
396 // Get the limb count and add it to the hash.
397 const mp_limb_t* limbs = mpz_limbs_read(value.get_mpz_t());
398 size_t limb_count = mpz_size(value.get_mpz_t());
399 h = H::combine(std::move(h), limb_count);
400
401 // Now include the limbs themselves.
402 for (size_t i = 0; i < limb_count; i++) {
403 h = H::combine(std::move(h), limbs[i]);
404 }
405 return h;
406}
