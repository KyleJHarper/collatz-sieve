107 template<FixedWidthIntegral T>
108 static inline T st_bit_reverse_full(const T& x) {
109 // When the type is signed, convert it to unsigned and try again.
110 if constexpr(std::numeric_limits<T>::is_signed) {
111 using U = std::make_unsigned_t<T>;
112 U ux = static_cast<U>(x);
113 return st_bit_reverse_full(ux);
114 }
115
116 // Type is already unsigned. Begin reversing.
117 if constexpr (sizeof(T) == 1) {
118 return static_cast<T>(bitreverse8(x));
119 } else if constexpr (sizeof(T) == 2) {
120 return static_cast<T>(bitreverse16(x));
121 } else if constexpr (sizeof(T) == 4) {
122 return static_cast<T>(bitreverse32(x));
123 } else if constexpr (sizeof(T) == 8) {
124 return static_cast<T>(bitreverse64(x));
125 } else if constexpr (sizeof(T) == 16) {
126 uint64_t low = (uint64_t)x;
127 uint64_t high = (uint64_t)(x >> 64);
128 uint64_t rev_low = bitreverse64(high);
129 uint64_t rev_high = bitreverse64(low);
130 return static_cast<T>(( (T)rev_high << 64 ) | rev_low);
131 }
132 }

145 static inline T st_reverse_low_bits(const T& value, size_t bits) {
146 // Test for zero.
147 T result = 0;
148 if (bits == 0) { return result; }
149
150 // Reverse and shift by the correct amount.
151 if constexpr(FixedWidthIntegral<T>) {
152 result = st_bit_reverse_full(value);
153 constexpr size_t WIDTH = sizeof(T) * 8;
154 result >>= (WIDTH - bits);
155 } else if constexpr(GMPIntegral<T>) {
156 for (size_t bit = 0; bit < bits; bit++) {
157 if (mpz_tstbit(value.get_mpz_t(), bit)) {
158 mpz_setbit(result.get_mpz_t(), bits - 1 - bit);
159 }
160 }
161 }
162
163 // Return.
164 return result;
165 }

206 template<AnySupportedIntegral T>
207 inline int count_trailing_zeros(const T& x) {
208 // Zero is a special and undefined case for builtins. Return 0.
209 if (x == 0) {
210 return 0;
211 }
212
213 if constexpr(FixedWidthIntegral<T>) {
214 // When it's below 64-bits (long long), we can use the builtin.
215 if constexpr(sizeof(T) <= 8) {
216 return std::countr_zero(x);
217 }
218
219 // Larger fixed-widths require one-off handling.
220 if constexpr(Int128Integral<T>) {
221 uint64_t low_bits = uint64_t(x);
222 if (low_bits != 0) {
223 // There are ones in the lower bits. Don't need high bits at all.
224 return std::countr_zero(low_bits);
225 } else {
226 // There are no ones in the lower bits. Return all 64 zeros plus whatever zeros are in high.
227 // High bits can't be all zeros because that would mean low+high == 0, which is guarded against.
228 uint64_t high_bits = uint64_t(x >> 64);
229 return 64 + std::countr_zero(high_bits);
230 }
231 }
232 } else if constexpr(GMPIntegral<T>) {
233 // GMP has a scan1 function to find the first 1, so we can use it.
234 return mpz_scan1(x.get_mpz_t(), 0);
235 }
236 }

283 inline int count_trailing_ones(const T& n) {
284 // Trailing ones is equivalent to the trailing zeros of inverted n: ctz(~n).
285 // GMP will perform an alloc for this, so we'll use TLS on that path.
286 if constexpr(FixedWidthIntegral<T>) {

291 inverted_n = ~n;

366 template<FixedWidthIntegral T>
367 static T byteswap(T& value) {
368 if constexpr (sizeof(T) == 1) {
369 return value;
370 } else if constexpr (sizeof(T) == 2) {
371 return __builtin_bswap16(value);
372 } else if constexpr (sizeof(T) == 4) {
373 return __builtin_bswap32(value);
374 } else if constexpr (sizeof(T) == 8) {
375 return __builtin_bswap64(value);
376 } else if constexpr (sizeof(T) == 16) {
377 // 128-bit needs broken into two chunks.
378 uint64_t high = static_cast<uint64_t>(value >> 64);
379 uint64_t low = static_cast<uint64_t>(value);
380 high = __builtin_bswap64(high);
381 low = __builtin_bswap64(low);
382 return (static_cast<T>(low) << 64) | high;
383 } else {
384 static_assert(sizeof(T) <= 16, "Unsupported size for byteswap");
385 return value;
386 }
387 }
