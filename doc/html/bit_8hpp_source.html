105 template<FixedWidthIntegral T>
106 static inline T st_bit_reverse_full(const T& x) {
107 // When the type is signed, convert it to unsigned and try again.
108 if constexpr(std::is_signed_v<T>) {
109 using U = std::make_unsigned_t<T>;
110 U ux = static_cast<U>(x);
111 return st_bit_reverse_full(ux);
112 }
113
114 // Type is already unsigned. Begin reversing.
115 if constexpr (sizeof(T) == 1) {
116 return static_cast<T>(bitreverse8(x));
117 } else if constexpr (sizeof(T) == 2) {
118 return static_cast<T>(bitreverse16(x));
119 } else if constexpr (sizeof(T) == 4) {
120 return static_cast<T>(bitreverse32(x));
121 } else if constexpr (sizeof(T) == 8) {
122 return static_cast<T>(bitreverse64(x));
123 } else if constexpr (sizeof(T) == 16) {
124 uint64_t low = (uint64_t)x;
125 uint64_t high = (uint64_t)(x >> 64);
126 uint64_t rev_low = bitreverse64(high);
127 uint64_t rev_high = bitreverse64(low);
128 return static_cast<T>(( (T)rev_high << 64 ) | rev_low);
129 }
130 }

143 static inline T st_reverse_low_bits(const T& value, size_t bits) {
144 // Test for zero.
145 T result = 0;
146 if (bits == 0) { return result; }
147
148 // Reverse and shift by the correct amount.
149 if constexpr(FixedWidthIntegral<T>) {
150 result = st_bit_reverse_full(value);
151 constexpr size_t WIDTH = sizeof(T) * 8;
152 result >>= (WIDTH - bits);
153 } else if constexpr(GMPIntegral<T>) {
154 for (size_t bit = 0; bit < bits; bit++) {
155 if (mpz_tstbit(value.get_mpz_t(), bit)) {
156 mpz_setbit(result.get_mpz_t(), bits - 1 - bit);
157 }
158 }
159 }
160
161 // Return.
162 return result;
163 }

204 template<AnySupportedIntegral T>
205 inline int count_trailing_zeros(const T& x) {
206 // Zero is a special and undefined case for builtins. Return sizeof T in bits.
207 if (x == 0) {
208 return 0;
209 }
210
211 if constexpr(FixedWidthIntegral<T>) {
212 // When it's below 64-bits (long long), we can use the builtin.
213 if constexpr(sizeof(T) <= 8) {
214 return __builtin_ctzll(x);
215 }
216
217 // Larger fixed-widths require one-off handling.
218 if constexpr(Int128Integral<T>) {
219 uint64_t low_bits = uint64_t(x);
220 if (low_bits != 0) {
221 // There are ones in the lower bits. Don't need high bits at all.
222 return __builtin_ctzll(low_bits);
223 } else {
224 // There are no ones in the lower bits. Return all 64 zeros plus whatever zeros are in high, if allowed.
225 uint64_t high_bits = uint64_t(x >> 64);
226 if (high_bits == 0) {
227 return 64;
228 } else {
229 return 64 + __builtin_ctzll(high_bits);
230 }
231 }
232 }
233
234 } else if constexpr(GMPIntegral<T>) {
235 // GMP has a scan1 function to find the first 1, so we can use it.
236 return mpz_scan1(x.get_mpz_t(), 0);
237 }
238
239 // If execution ever reaches this, no valid path was found.
240 throw std::logic_error("Cannot discern data type when calling count_trailing_zeros.");
241 }

251 template<AnySupportedIntegral T>
252 inline int count_trailing_ones(const T& n) {
253 // Trailing ones is equivalent to the trailing zeros of inverted n: ctz(~n).
254 // GMP will perform an alloc for this, so we'll use TLS on that path.
255 if constexpr(FixedWidthIntegral<T>) {
256 T inverted_n = ~n;
257 return count_trailing_zeros(inverted_n);
258 } else if constexpr(GMPIntegral<T>) {
259 static thread_local mpz_class inverted_n = 0;
260 inverted_n = ~n;
261 return count_trailing_zeros(inverted_n);
262 }
263
264 // If execution ever reaches this, no valid path was found.
265 throw std::logic_error("Cannot discern data type when calling count_trailing_ones.");
266 }

276 template<FixedWidthIntegral T>
277 static T byteswap(T& value) {
278 if constexpr (sizeof(T) == 1) {
279 return value;
280 } else if constexpr (sizeof(T) == 2) {
281 return __builtin_bswap16(value);
282 } else if constexpr (sizeof(T) == 4) {
283 return __builtin_bswap32(value);
284 } else if constexpr (sizeof(T) == 8) {
285 return __builtin_bswap64(value);
286 } else if constexpr (sizeof(T) == 16) {
287 // 128-bit needs broken into two chunks.
288 uint64_t high = static_cast<uint64_t>(value >> 64);
289 uint64_t low = static_cast<uint64_t>(value);
290 high = __builtin_bswap64(high);
291 low = __builtin_bswap64(low);
292 return (static_cast<T>(low) << 64) | high;
293 } else {
294 static_assert(sizeof(T) <= 16, "Unsupported size for byteswap");
295 return value;
296 }
297 }
