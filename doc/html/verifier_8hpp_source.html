237 void start() {
238 // Need to check if the worker has already been started.
239 if (_state.load(std::memory_order_relaxed) == VerifierState::RUNNING) {
240 return;
241 }
242
243 // Cannot start running when the state is anything other than STOPPED.
244 if (_state.load(std::memory_order_relaxed) != VerifierState::STOPPED) {
245 throw std::logic_error("Cannot call start() unless the job is in the STOPPED state.");
246 }
247
248 // Join the previous executor thread (if any) to make this re-entrant safe-ish.
249 if (_executor_thread.joinable()) {
250 _executor_thread.join();
251 }
252
253 // Reset metrics and time tracking.
254 _published_metrics.reset();
255
256 // Change the state.
257 _state.store(VerifierState::RUNNING);
258
259 // Fire up the executor in a thread.
260 _executor_thread = std::thread([this] { run_executor(); });
261
262 // Start the timer.
263 start_timer();
264 }

269 void stop() {
270 // When stopped or stopping, the state is already correct.
271 if (_state.load(std::memory_order_relaxed) == VerifierState::STOPPED || _state.load(std::memory_order_relaxed) == VerifierState::STOPPING) {
272 return;
273 }
274
275 // Flag STOPPING (not stopped; only the _worker can flag STOPPED).
276 _state.store(VerifierState::STOPPING);
277
278 // Do not sync or stop the timer here. Executor thread will do that when all the child threads are done.
279
280 // Join the executor thread. Both `start()` and the destructor have added protection for this, fyi.
281 if (_executor_thread.joinable()) {
282 _executor_thread.join();
283 }
284 }

289 void pause() {
290 // When already paused, do nothing.
291 if (_state.load(std::memory_order_relaxed) == VerifierState::PAUSED) {
292 return;
293 }
294
295 // If stopped, exception. Cannot pause a stopped worker.
296 if (_state.load(std::memory_order_relaxed) == VerifierState::STOPPED || _state.load(std::memory_order_relaxed) == VerifierState::STOPPING) {
297 throw std::logic_error("Cannot call pause() when a worker is STOPPED or STOPPING.");
298 }
299
300 // Flag PAUSED. Worker will read this and go to sleep eventually.
301 _state.store(VerifierState::PAUSED);
302
303 // Sync the timer. The `resume()` method will call `start_timer()` to reset it.
304 sync_timer();
305 }
306
307
308
310 void resume() {
311 // When already running, do nothing.
312 if (_state.load(std::memory_order_relaxed) == VerifierState::RUNNING) {
313 return;
314 }
315
316 // If stopped, exception. Cannot resume a stopped worker.
317 if (_state.load(std::memory_order_relaxed) == VerifierState::STOPPED || _state.load(std::memory_order_relaxed) == VerifierState::STOPPING) {
318 throw std::logic_error("Cannot call resume() when a worker is STOPPED or STOPPING.");
319 }
320
321 // Flag RUNNING. Worker will read this and resume operation.
322 _state.store(VerifierState::RUNNING);
323
324 // Start the timer again.
325 start_timer();
326 }
