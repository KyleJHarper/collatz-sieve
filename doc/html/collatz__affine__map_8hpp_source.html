12//
13// Collatz Affine Maps
14//
15// An FG chain can be stored and repsented as an affine map with powers of two and three, and with a constant portion. This avoids
16// any realtime tracking of fractional values (floats).
17//
18// Note, there are overflow concerns:
19// The constant_portion grows the most when the chain is all F. This worst case leaves constant_portion == ~3^k. We need about
20// 1.6 bits to represent this in binary, so we get the following:
21// 64 bits ==> 40
22// 128 bits ==> 80
23//
24// To resolve this, Node uses an upgraded T when nessary. Anyone else using this needs to provide the same protection.
25//

97 void apply_F() {
98 // Adjust the constant portion before we modify any exponents.
99 if constexpr(FixedWidthIntegral<T>) {
100 // Built-in types can bit shift cheaper than looking up a table. Do that instead.
101 _constant_portion = ((_constant_portion << 1) + _constant_portion) + (T(1) << _twos_exp);
102 } else if constexpr(GMPIntegral<T>) {
103 // MPZ Class can do an mpz_add from a precaulated item in a table faster than an initialization + shift operator.
104 mpz_mul(_constant_portion.get_mpz_t(), _constant_portion.get_mpz_t(), CollatzConstants::MPZ_THREE.get_mpz_t());
105 mpz_add(_constant_portion.get_mpz_t(), _constant_portion.get_mpz_t(), Exponents::get_power_of_two<T>(_twos_exp).get_mpz_t());
106 }
107
108 // Now bump the exponents.
109 _threes_exp += 1;
110 _twos_exp += 1;
111 }

127 T calculate(const T& value) const {
128 T result = Exponents::get_power_of_three<T>(_threes_exp);
129 if constexpr(FixedWidthIntegral<T>) {
130 if (__builtin_mul_overflow(result, value, &result) || __builtin_add_overflow(result, _constant_portion, &result)) {
131 throw std::overflow_error(
132 "Overflow in CollatzAffineMap calculate() method. Value is " + to_string_any(value)
133 + " Threes exponent is " + to_string_any(_threes_exp)
134 + " Constant portion is " + to_string_any(_constant_portion)
135 + "."
136 );
137 }
138 result >>= _twos_exp;
139 } else if constexpr(GMPIntegral<T>) {
140 mpz_mul(result.get_mpz_t(), result.get_mpz_t(), value.get_mpz_t());
141 mpz_add(result.get_mpz_t(), result.get_mpz_t(), _constant_portion.get_mpz_t());
142 mpz_fdiv_q_2exp(result.get_mpz_t(), result.get_mpz_t(), _twos_exp);
143 }
144 return result;
145 }
