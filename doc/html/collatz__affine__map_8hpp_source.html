11//
12// Collatz Affine Maps
13//
14// An FG chain can be stored and repsented as an affine map with powers of two and three, and with a constant portion. This avoids
15// any realtime tracking of fractional values (floats).
16//
17// Note, there are overflow concerns:
18// The constant_portion grows the most when the chain is all F. This worst case leaves constant_portion == ~3^k. We need about
19// 1.6 bits to represent this in binary, so we get the following:
20// 64 bits ==> 40
21// 128 bits ==> 80
22//
23// To resolve this, Node uses an upgraded T when nessary. Anyone else using this needs to provide the same protection.
24//

82 void apply_F() {
83 // Adjust the constant portion before we modify any exponents.
84 if constexpr(BuiltinIntegral<T>) {
85 // Built-in types can bit shift cheaper than looking up a table. Do that instead.
86 _constant_portion = ((_constant_portion << 1) + _constant_portion) + (T(1) << _twos_exp);
87 } else {
88 // MPZ Class can do an mpz_add from a precaulated item in a table faster than an initialization + shift operator.
89 mpz_mul(_constant_portion.get_mpz_t(), _constant_portion.get_mpz_t(), CollatzConstants::MPZ_THREE.get_mpz_t());
90 mpz_add(_constant_portion.get_mpz_t(), _constant_portion.get_mpz_t(), Exponents::get_power_of_two<T>(_twos_exp).get_mpz_t());
91 }

112 T calculate(const T& value) const {
113 T result = Exponents::get_power_of_three<T>(_threes_exp);
114 if constexpr(BuiltinIntegral<T>) {
115 if (__builtin_mul_overflow(result, value, &result) || __builtin_add_overflow(result, _constant_portion, &result)) {
116 throw std::overflow_error(
117 "Overflow in CollatzAffineMap calculate() method. Value is " + to_string_any(value)
118 + " Threes exponent is " + to_string_any(_threes_exp)
119 + " Constant portion is " + to_string_any(_constant_portion)
120 + "."
121 );
122 }
123 result >>= _twos_exp;
124 } else if constexpr(GMPIntegral<T>) {
125 mpz_mul(result.get_mpz_t(), result.get_mpz_t(), value.get_mpz_t());
126 mpz_add(result.get_mpz_t(), result.get_mpz_t(), _constant_portion.get_mpz_t());
127 mpz_fdiv_q_2exp(result.get_mpz_t(), result.get_mpz_t(), _twos_exp);
128 } else {
129 throw std::logic_error("Unknown data type for calculate().");
130 }
131 return result;
132 }
