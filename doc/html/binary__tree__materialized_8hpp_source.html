93 ~BinaryTreeMaterializedImpl() {
94 // Clean up ancestors. These are new-placement, so they aren't linked to the same memory as _level_map.
95 for (Node<T>* node : _ancestors) {
96 delete node;
97 }
98 _ancestors.clear();
99
100 // Now clear out the level_map. If pruning, handle it carefully.
101 if (_is_pruning_ast_nodes) {
102 // Scan all levels manually and call delete explicitly.
103 for (level_t level = 0; level <= _level_count; level++) {
104 for (Node<T>* node : _level_map[level]) {
105 if (node != nullptr) {
106 node->own_children(false);
107 delete node;
108 }
109 }
110 }
111 } else {
112 // This will cascade to children because Node.own_children is default true.
113 delete _root_node;
114 }
115 }

264 size_t deep_size() const {
265 size_t total = sizeof(*this);
266
267 // Account for _level_map
268 total += sizeof(_level_map);
269 for (const auto& [level, nodes] : _level_map) {
270 total += sizeof(level);
271 total += sizeof(nodes);
272 total += nodes.capacity() * sizeof(Node<T>*);
273 for (const Node<T>* node : nodes) {
274 if (node) {
275 total += node->deep_size();
276 }
277 }
278 }
279
280 // Add uncovered values bitmap.
281 total += _uncovered_values.deep_size();
282
283 // Account for _coverage_map
284 total += sizeof(_coverage_map);
285 for (const auto& [level, coverage] : _coverage_map) {
286 total += sizeof(level);
287 total += sizeof(coverage);
288 }
289
290 // Unordered map buckets.
291 total += _level_map.bucket_count() * sizeof(void*);
292 return total;
293 }

313 static bool st_equal(const BinaryTreeMaterializedImpl<T>& first, const BinaryTreeMaterializedImpl<T>& second, std::string* err = nullptr) {
314 EqualityHelper eq(err);
315 eq.set_category("BinaryTreeMaterializedImpl");
316
317 // Flags
318 if (! eq.equal(first.is_pruning_ast_nodes(), second.is_pruning_ast_nodes())) {
319 return eq.fail("Is pruning ast nodes mismatch");
320 }
321 if (! eq.equal(first.is_pruning_parent_levels(), second.is_pruning_parent_levels())) {
322 return eq.fail("Is pruning parent levels mismatch");
323 }
324
325 // Level Map
326 if (! eq.equal(first.get_level_map().size(), second.get_level_map().size())) {
327 return eq.fail("Level map count mismatch");
328 }
329 for (const auto& [level, f_nodes] : first.get_level_map()) {
330 if (! second.get_level_map().contains(level)) {
331 return eq.fail("First level map has level " + to_string_any(level) + ", but the second level map doesn't.");
332 }
333 const std::vector<Node<T>*>& s_nodes = second.get_level_map().at(level);
334 if (! eq.equal(f_nodes.size(), s_nodes.size())) {
335 return eq.fail("Node counts on level " + to_string_any(level) + " mismatch");
336 }
337 for (size_t i = 0; i < f_nodes.size(); i++) {
338 if(! Node<T>::st_equal(f_nodes.at(i), s_nodes.at(i), err)) {
339 return false;
340 }
341 }
342 }
343
344 // All good.
345 return true;
346 }

380 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
381 // !!! BIG FAT ANNOYING ERROR !!!
382 if (_is_pruning_ast_nodes || _is_pruning_parent_levels) {
383 std::string msg = "You've activated my trap card!";
384 msg+= " Serializing and deserializing a materialized tree is already a bad idea.";
385 msg+= " Doing it with any form of node pruning enabled is NOT supported NOR a good idea.";
386 msg+= " Oh, and rebuilding a materialized tree is almost always faster than trying to import one.";
387 msg+= " Be well.";
388 throw std::logic_error(msg);
389 }
390
391 StreamHelper sh(nullptr, &out, err);
392 sh.set_category("BinaryTreeMaterializedImpl");
393
394 // Unique properties.
395 if (! sh.serialize_bool(_is_pruning_ast_nodes)) {
396 return sh.fail("_is_pruning_ast_nodes==" + std::to_string(_is_pruning_ast_nodes));
397 }
398 if (! sh.serialize_bool(_is_pruning_parent_levels)) {
399 return sh.fail("_is_pruning_parent_levels==" + std::to_string(_is_pruning_parent_levels));
400 }
401
402 // Level Map
403 // Write how many levels we need to loop through later.
404 //
405 // Note: do NOT serialize level 1, because it's only the root node, which is a common item. It is linked in deserialize().
406 uint64_t u64_level_map_count = get_level_map().size() - 1;
407 if (! sh.serialize_integral(u64_level_map_count)) {
408 return sh.fail("level_map_count==" + to_string_any(u64_level_map_count));
409 }
410 // Get the level numbers in order for guaranteed write order.
411 std::vector<level_t> level_map_levels;
412 level_map_levels.reserve(_level_map.size());
413 for (const auto& [level, _] : _level_map) {
414 if (level == 1) { continue; }
415 level_map_levels.push_back(level);
416 }
417 std::sort(level_map_levels.begin(), level_map_levels.end());
418 // Loop through each level found.
419 for (level_t level : level_map_levels) {
420 const std::vector<Node<T>*>& nodes = _level_map.at(level);
421 // Write the level number and then the number of nodes on it.
422 if (! sh.serialize_integral(level)) {
423 return sh.fail("level map's level==" + to_string_any(level));
424 }
425 uint64_t u64_node_size = static_cast<uint64_t>(nodes.size());
426 if (! sh.serialize_integral(u64_node_size)) {
427 return sh.fail("level map on level==" + to_string_any(level) + " nodes.size()==" + to_string_any(nodes.size()));
428 }
429 // Write each node out.
430 for (const Node<T>* node : nodes) {
431 if (! node->serialize(out, err)) {
432 return sh.fail("level map node with value==" + to_string_any(node->get_value()));
433 }
434 }
435 }
436
437 // All good.
438 return true;
439 }

454 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
455 StreamHelper sh(&in, nullptr, err);
456 sh.set_category("BinaryTreeMaterializedImpl");
457
458 // Unique properties
459 bool b_tmp;
460 if (! sh.deserialize_bool(b_tmp)) {
461 return sh.fail("couldn't read _is_pruning_ast_nodes");
462 }
463 _is_pruning_ast_nodes = b_tmp;
464 if (! sh.deserialize_bool(b_tmp)) {
465 return sh.fail("couldn't read _is_pruning_parent_levels");
466 }
467 _is_pruning_parent_levels = b_tmp;
468
469 // Level map
470 // Start by assigning the root node to level 1, because it's not exported in serialize().
471 _level_map[1].resize(1);
472 _level_map[1][0] = _root_node;
473 // Now get the remaining levels, if any.
474 uint64_t u64_level_map_count = 0;
475 T parent_v;
476 T ast_ancestor_v;
477 uint8_t child_count;
478 if (! sh.deserialize_integral(u64_level_map_count)) {
479 return sh.fail("couldn't read level_map_count");
480 }
481 for (uint64_t i = 0; i < u64_level_map_count; i++) {
482 uint64_t parent_index = 0;
483 level_t level = 0;
484 if (! sh.deserialize_integral(level)) {
485 return sh.fail("couldn't read level for index i==" + to_string_any(i));
486 }
487 uint64_t u64_node_size = 0;
488 if (! sh.deserialize_integral(u64_node_size)) {
489 return sh.fail("couldn't read node size for level==" + to_string_any(level));
490 }
491 size_t node_size = static_cast<size_t>(u64_node_size);
492 _level_map[level].reserve(node_size);
493 for (uint64_t j = 0; j < u64_node_size; j++) {
494 // Make a new heap-allocated node to deserialize into.
495 Node<T>* new_node = new Node<T>();
496 if (! new_node->deserialize(in, parent_v, ast_ancestor_v, child_count, err)) {
497 return sh.fail("couldn't get node on level==" + to_string_any(level) + " and position j==" + to_string_any(j));
498 }
499 // Add it to the map.
500 _level_map[level].push_back(new_node);
501 // Find the parent by-value. Must be in the previous level. Use while() to avoid GMP allocs.
502 // These were added left-to-right, so we can scan with an index efficiently.
503 Node<T>* parent = nullptr;
504 while (parent_index < _level_map[level - 1].size()) {
505 if (_level_map[level - 1][parent_index]->get_value() == parent_v) {
506 parent = _level_map[level - 1][parent_index];
507 break;
508 }
509 // Didnt' find it, bump the index.
510 parent_index++;
511 }
512 if (parent == nullptr) {
513 return sh.fail("couldn't find parent on level-1==" + to_string_any(level - 1) + " with parent value==" + to_string_any(parent_v));
514 }
515 // Link up parents and children.
516 new_node->assign_parent(parent);
517 parent->assign_child(new_node);
518 // Follow the same ancestor tracking logic from Node: inherit parent's, and if still null, see if parent is one.
519 new_node->assign_ast_ancestor(parent->get_ast_ancestor());
520 if (new_node->get_ast_ancestor() == nullptr && parent->is_below_ast()) {
521 new_node->assign_ast_ancestor(parent);
522 }
523 }
524 }
525
526 // All good
527 return true;
528 }

544 void add_level() {
545 // Ensure initialization is set.
546 _is_initialized = true;
547
548 // Clear the value bitmap since it would otherwise be stale/incorrect.
549 _uncovered_values.clear();
550
551 // When we have no levels, this simply crafts a 0- or 1-based root node and manually sets level map and coverage.
552 if (_level_count == 0) {
553 _level_count = 1;
554 _root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
555 _level_map[_level_count].resize(1);
556 _level_map[_level_count][0] = _root_node;
557 _coverage_map[_level_count].set_covered(0);
558 _coverage_map[_level_count].set_total(1);
559 return;
560 }
561
562 // Get the parent and child levels. Then confirm the new one will fit. Then bump count.
563 level_t parent_level = _level_count;
564 level_t child_level = _level_count + 1;
565 _level_count++;
566
567 // Build the scaling factor value. Each level doubles the tree, so we need to respect T with GMP-size values.
568 T scaling_factor = BinaryTreeMath<T>::st_scaling_factor(child_level);
569
570 // Calculate parent and child counts for looping and indexing.
571 // Note: size_t is safe, because no one can fit 2^64 nodes in RAM.
572 auto& parents = _level_map[parent_level];
573 size_t parent_count = parents.size();
574 size_t child_count = parent_count * 2;
575
576 // Resize the vector to contain all children. Otherwise assigning to indexes ([]) will fail.
577 _level_map[child_level].resize(child_count);
578
579 // Keep a record of covered or pruned values for coverage math after the loop. Use guaranteed-width integral.
580 uint64_t covered_or_pruned = 0;
581
582 // Create an external tracker for ancestors for OMP local threads to utilize by-ref.
583 std::vector<std::vector<Node<T>*>> omp_local_ancestors_group(omp_get_max_threads());
584
585 // Begin the critical section, but don't loop yet.
586 std::exception_ptr eptr = nullptr;
587 #pragma omp parallel reduction(+:covered_or_pruned) default(none) shared(parents, _level_map, scaling_factor, child_level, parent_count, _is_pruning_ast_nodes, _is_preserving_ancestors, omp_local_ancestors_group, _is_verifying_non_ast_nodes, eptr)
588 {
589 try {
590 // Use local child values.
591 T child_value_1;
592 T child_value_2;
593
594 // Grab this thread's personal ancestors vector to avoid locking.
595 auto& my_ancestors = omp_local_ancestors_group[omp_get_thread_num()];
596
597 // Track this thread's local AST. Since nodes are not sequentially ordered, it can only set this to first node - 1.
598 T my_ast_value = BinaryTreeMath<T>::st_first_node_value_of_level(_level_count) - 1;
599 if (my_ast_value < 1) {
600 my_ast_value = 1;
601 }
602
603 // Now Loop
604 #pragma omp for schedule(static)
605 for(size_t parent_idx = 0; parent_idx < parent_count; parent_idx++) {
606 // Find the parent.
607 Node<T>* parent = parents[parent_idx];
608
609 // Compute the child values. Avoid alloc() with GMP with arithmetic operators.
610 if constexpr(FixedWidthIntegral<T>) {
611 child_value_1 = parent->get_value() + scaling_factor;
612 child_value_2 = child_value_1 + scaling_factor;
613 } else if constexpr(GMPIntegral<T>) {
614 mpz_add(child_value_1.get_mpz_t(), parent->get_value().get_mpz_t(), scaling_factor.get_mpz_t());
615 mpz_add(child_value_2.get_mpz_t(), child_value_1.get_mpz_t(), scaling_factor.get_mpz_t());
616 }
617
618 // Instaniate the children.
619 Node<T>* child_1 = new Node<T>(child_value_1, parent);
620 Node<T>* child_2 = new Node<T>(child_value_2, parent);
621
622 // Tally them. Prune them if necessesary. Otherwise add them to the map.
623 // -- Child 1
624 bool assign_to_map = true;
625 if (_is_preserving_ancestors && child_1->is_below_ast() && ! child_1->has_ast_ancestor()) {
626 my_ancestors.push_back(new Node<T>(child_1->get_value()));
627 }
628 if (_is_verifying_non_ast_nodes) {
629 if (child_1->is_below_ast() == false && child_1->has_ast_ancestor() == false) {
630 if (Collatz<T>::st_verify(child_value_1, my_ast_value) == false) {
631 throw std::logic_error("Node value " + to_string_any(child_value_1) + " didn't verify. How?");
632 }
633 }
634 }
635 if (child_1->is_below_ast() || child_1->has_ast_ancestor()) {
636 covered_or_pruned += 1;
637 if (_is_pruning_ast_nodes) {
638 delete child_1;
639 child_1 = nullptr;
640 assign_to_map = false;
641 }
642 }
643 if (assign_to_map) {
644 _level_map[child_level][2 * parent_idx] = child_1;
645 if (! _is_pruning_ast_nodes && ! _is_pruning_parent_levels) {
646 parent->assign_child(child_1);
647 }
648 }
649 // -- Child 2
650 assign_to_map = true;
651 if (_is_preserving_ancestors && child_2->is_below_ast() && ! child_2->has_ast_ancestor()) {
652 my_ancestors.push_back(new Node<T>(child_2->get_value()));
653 }
654 if (_is_verifying_non_ast_nodes) {
655 if (child_2->is_below_ast() == false && child_2->has_ast_ancestor() == false) {
656 if (Collatz<T>::st_verify(child_value_2, my_ast_value) == false) {
657 throw std::logic_error("Node value " + to_string_any(child_value_2) + " didn't verify. How?");
658 }
659 }
660 }
661 if (child_2->is_below_ast() || child_2->has_ast_ancestor()) {
662 covered_or_pruned += 1;
663 if (_is_pruning_ast_nodes) {
664 delete child_2;
665 child_2 = nullptr;
666 assign_to_map = false;
667 }
668 }
669 if (assign_to_map) {
670 _level_map[child_level][2 * parent_idx + 1] = child_2;
671 if (! _is_pruning_ast_nodes && ! _is_pruning_parent_levels) {
672 parent->assign_child(child_2);
673 }
674 }
675 }
676 } catch (...) {
677 #pragma omp critical
678 {
679 if(!eptr) {
680 eptr = std::current_exception();
681 }
682 }
683 }
684 }
685
686 // Rethrow errors.
687 if(eptr != nullptr) {
688 std::rethrow_exception(eptr);
689 }
690
691 // Merge Ancestors.
692 if (_is_preserving_ancestors) {
693 for (std::vector<Node<T>*>& omp_ancestors : omp_local_ancestors_group) {
694 _ancestors.insert(_ancestors.end(), omp_ancestors.begin(), omp_ancestors.end());
695 }
696 // Always sort them.
697 tbb::parallel_sort(_ancestors.begin(), _ancestors.end(), [](const Node<T>* a, const Node<T>* b) {
698 if constexpr(FixedWidthIntegral<T>) {
699 return a->get_value() < b->get_value();
700 } else if constexpr(GMPIntegral<T>) {
701 return mpz_cmp(a->get_value().get_mpz_t(), b->get_value().get_mpz_t()) < 0;
702 }
703 });
704 }
705
706 // When pruning nodes,remove any nullptr (pruned) children from the vector to keep counts accurate on next loop.
707 if (_is_pruning_ast_nodes) {
708 std::vector<Node<T>*>& children = _level_map[child_level];
709 children.erase(
710 std::remove(children.begin(), children.end(), nullptr),
711 children.end()
712 );
713 children.shrink_to_fit();
714 }
715
716 // When pruning parent levels, sweep them out.
717 // Nodes default-own children. Call own_children(false) for safety.
718 if (_is_pruning_parent_levels) {
719 // When it's level 1, clear up _root too so it doesn't double-free later.
720 if (parent_level == 1) {
721 _root_node = nullptr;
722 }
723 #pragma omp parallel for
724 for (size_t i = 0; i < _level_map[parent_level].size(); i++) {
725 Node<T>* parent = _level_map[parent_level][i];
726 parent->own_children(false);
727 delete parent;
728 }
729 _level_map[parent_level].clear();
730 _level_map[parent_level].shrink_to_fit();
731 }
732
733 // Coverage is tallied above. If pruning, logic is different because ancestors were purged already, which means their
734 // descendents were purged, but still count toward coverage. Luckily, by shrinking the vector above, this method can rely
735 // on .size() to report how many are *not* covered, which means covered = total - .size().
736 T total = scaling_factor * 2;
737 T covered = covered_or_pruned;
738 if (_is_pruning_ast_nodes) {
739 covered = total - _level_map[child_level].size();
740 }
741 _coverage_map[child_level] = BinaryTreeCoverage<T>(covered, total);
742 }

753 template<size_t BUFFER_SIZE = 1 << 24>
754 void generate_value_map(ForEachPolicy policy = ForEachPolicy::SERIAL) {
755 _uncovered_values.clear();
756
757 if (policy == ForEachPolicy::SERIAL) {
758 // Just loop through the level map one-by-one.
759 std::vector<T> buffer;
760 buffer.reserve(BUFFER_SIZE);
761 for (const Node<T>* node : _level_map[_level_count]) {
762 // Non-pruned trees have covered nodes in memory. Check for them.
763 if (node->is_below_ast() || node->has_ast_ancestor()) {
764 continue;
765 }
766 buffer.push_back(node->get_value());
767 if (buffer.size() >= BUFFER_SIZE) {
768 _uncovered_values.add_many(buffer.size(), buffer.data());
769 buffer.clear();
770 }
771 }
772 _uncovered_values.add_many(buffer.size(), buffer.data());
773 } else {
774 struct ValueMapTLS {
775 NodeBitmap<T> bitmap;
776 std::vector<T> buffer;
777 };
778 std::vector<ValueMapTLS> tls_vector;
779 tls_vector.resize(omp_get_max_threads());
780 #pragma omp parallel default(none) shared(_level_map, tls_vector)
781 {
782 int my_thread_id = omp_get_thread_num();
783 ValueMapTLS& my_tls = tls_vector[my_thread_id];
784 my_tls.buffer.reserve(BUFFER_SIZE);
785 #pragma omp for
786 for (size_t i = 0; i < _level_map[_level_count].size(); i++) {
787 my_tls.buffer.push_back(_level_map[_level_count][i]->get_value());
788 if (my_tls.buffer.size() >= BUFFER_SIZE) {
789 my_tls.bitmap.add_many(my_tls.buffer.size(), my_tls.buffer.data());
790 my_tls.buffer.clear();
791 }
792 }
793 my_tls.bitmap.add_many(my_tls.buffer.size(), my_tls.buffer.data());
794 }
795 // Now merge bitmaps and remaining buffers.
796 for (ValueMapTLS& tls : tls_vector) {
797 _uncovered_values |= tls.bitmap;
798 tls.bitmap.clear();
799 _uncovered_values.add_many(tls.buffer.size(), tls.buffer.data());
800 }
801 }
802
803 // Optimize
804 _uncovered_values.optimize();
805 }
