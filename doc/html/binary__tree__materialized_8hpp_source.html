76 ~BinaryTreeMaterializedImpl() {
77 // Clean up ancestors. These are new-placement, so they aren't linked to the same memory as _level_map.
78 for (Node<T>* node : _ancestors) {
79 delete node;
80 }
81 _ancestors.clear();
82
83 // Now clear out the level_map. If pruning, handle it carefully.
84 if (_is_pruning_hwm_nodes) {
85 // Scan all levels manually and call delete explicitly.
86 for (level_t level = 0; level <= _level_count; level++) {
87 for (Node<T>* node : _level_map[level]) {
88 if (node != nullptr) {
89 node->own_children(false);
90 delete node;
91 }
92 }
93 }
94 } else {
95 // This will cascade to children because Node.own_children is default true.
96 delete _root_node;
97 }
98 }

233 size_t deep_size() const {
234 size_t total = sizeof(*this);
235
236 // Account for _level_map
237 total += sizeof(_level_map);
238 for (const auto& [level, nodes] : _level_map) {
239 total += sizeof(level);
240 total += sizeof(nodes);
241 total += nodes.capacity() * sizeof(Node<T>*);
242 for (const Node<T>* node : nodes) {
243 if (node) {
244 total += node->deep_size();
245 }
246 }
247 }
248
249 // Account for _coverage_map
250 total += sizeof(_coverage_map);
251 for (const auto& [level, coverage] : _coverage_map) {
252 total += sizeof(level);
253 total += sizeof(coverage);
254 }

256 // Unordered map buckets.
257 total += _level_map.bucket_count() * sizeof(void*);

279 static bool st_equal(const BinaryTreeMaterializedImpl<T>& first, const BinaryTreeMaterializedImpl<T>& second, std::string* err = nullptr) {
280 EqualityHelper eq(err);
281 eq.set_category("BinaryTreeMaterializedImpl");
282
283 // Flags
284 if (eq.unequal(first.is_pruning_hwm_nodes(), second.is_pruning_hwm_nodes())) {
285 return eq.fail("Is pruning hwm nodes mismatch");
286 }
287 if (eq.unequal(first.is_pruning_parent_levels(), second.is_pruning_parent_levels())) {
288 return eq.fail("Is pruning parent levels mismatch");
289 }
290
291 // Level Map
292 if (eq.unequal(first.get_level_map().size(), second.get_level_map().size())) {
293 return eq.fail("Level map count mismatch");
294 }
295 for (const auto& [level, f_nodes] : first.get_level_map()) {
296 if (! second.get_level_map().contains(level)) {
297 return eq.fail("First level map has level " + to_string_any(level) + ", but the second level map doesn't.");
298 }
299 const std::vector<Node<T>*>& s_nodes = second.get_level_map().at(level);
300 if (eq.unequal(f_nodes.size(), s_nodes.size())) {
301 return eq.fail("Node counts on level " + to_string_any(level) + " mismatch");
302 }
303 for (size_t i = 0; i < f_nodes.size(); i++) {
304 if(! Node<T>::st_equal(f_nodes.at(i), s_nodes.at(i), err)) {
305 return false;
306 }
307 }
308 }
309
310 // All good.
311 return true;
312 }

346 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
347 // !!! BIG FAT ANNOYING ERROR !!!
348 if (_is_pruning_hwm_nodes || _is_pruning_parent_levels) {
349 std::string msg = "You've activated my trap card!";
350 msg+= " Serializing and deserializing a materialized tree is already a bad idea.";
351 msg+= " Doing it with any form of node pruning enabled is NOT supported NOR a good idea.";
352 msg+= " Oh, and rebuilding a materialized tree is almost always faster than trying to import one.";
353 msg+= " Be well.";
354 throw std::logic_error(msg);
355 }
356
357 StreamHelper sh(nullptr, &out, err);
358 sh.set_category("BinaryTreeMaterializedImpl");
359
360 // Unique properties.
361 if (! sh.serialize_bool(_is_pruning_hwm_nodes)) {
362 return sh.fail("_is_pruning_hwm_nodes==" + std::to_string(_is_pruning_hwm_nodes));
363 }
364 if (! sh.serialize_bool(_is_pruning_parent_levels)) {
365 return sh.fail("_is_pruning_parent_levels==" + std::to_string(_is_pruning_parent_levels));
366 }
367
368 // Level Map
369 // Write how many levels we need to loop through later.
370 //
371 // Note: do NOT serialize level 1, because it's only the root node, which is a common item. It is linked in deserialize().
372 uint64_t u64_level_map_count = get_level_map().size() - 1;
373 if (! sh.serialize_integral(u64_level_map_count)) {
374 return sh.fail("level_map_count==" + to_string_any(u64_level_map_count));
375 }
376 // Get the level numbers in order for guaranteed write order.
377 std::vector<level_t> level_map_levels;
378 level_map_levels.reserve(_level_map.size());
379 for (const auto& [level, _] : _level_map) {
380 if (level == 1) { continue; }
381 level_map_levels.push_back(level);
382 }
383 std::sort(level_map_levels.begin(), level_map_levels.end());
384 // Loop through each level found.
385 for (level_t level : level_map_levels) {
386 const std::vector<Node<T>*>& nodes = _level_map.at(level);
387 // Write the level number and then the number of nodes on it.
388 if (! sh.serialize_integral(level)) {
389 return sh.fail("level map's level==" + to_string_any(level));
390 }
391 uint64_t u64_node_size = static_cast<uint64_t>(nodes.size());
392 if (! sh.serialize_integral(u64_node_size)) {
393 return sh.fail("level map on level==" + to_string_any(level) + " nodes.size()==" + to_string_any(nodes.size()));
394 }
395 // Write each node out.
396 for (const Node<T>* node : nodes) {
397 if (! node->serialize(out, err)) {
398 return sh.fail("level map node with value==" + to_string_any(node->get_value()));
399 }
400 }
401 }
402
403 // All good.
404 return true;
405 }

420 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
421 StreamHelper sh(&in, nullptr, err);
422 sh.set_category("BinaryTreeMaterializedImpl");
423
424 // Unique properties
425 bool b_tmp;
426 if (! sh.deserialize_bool(b_tmp)) {
427 return sh.fail("couldn't read _is_pruning_hwm_nodes");
428 }
429 _is_pruning_hwm_nodes = b_tmp;
430 if (! sh.deserialize_bool(b_tmp)) {
431 return sh.fail("couldn't read _is_pruning_parent_levels");
432 }
433 _is_pruning_parent_levels = b_tmp;
434
435 // Level map
436 // Start by assigning the root node to level 1, because it's not exported in serialize().
437 _level_map[1].resize(1);
438 _level_map[1][0] = _root_node;
439 // Now get the remaining levels, if any.
440 uint64_t u64_level_map_count = 0;
441 T parent_v;
442 T hwm_ancestor_v;
443 uint8_t child_count;
444 if (! sh.deserialize_integral(u64_level_map_count)) {
445 return sh.fail("couldn't read level_map_count");
446 }
447 for (uint64_t i = 0; i < u64_level_map_count; i++) {
448 uint64_t parent_index = 0;
449 level_t level = 0;
450 if (! sh.deserialize_integral(level)) {
451 return sh.fail("couldn't read level for index i==" + to_string_any(i));
452 }
453 uint64_t u64_node_size = 0;
454 if (! sh.deserialize_integral(u64_node_size)) {
455 return sh.fail("couldn't read node size for level==" + to_string_any(level));
456 }
457 size_t node_size = static_cast<size_t>(u64_node_size);
458 _level_map[level].reserve(node_size);
459 for (uint64_t j = 0; j < u64_node_size; j++) {
460 // Make a new heap-allocated node to deserialize into.
461 Node<T>* new_node = new Node<T>();
462 if (! new_node->deserialize(in, parent_v, hwm_ancestor_v, child_count, err)) {
463 return sh.fail("couldn't get node on level==" + to_string_any(level) + " and position j==" + to_string_any(j));
464 }
465 // Add it to the map.
466 _level_map[level].push_back(new_node);
467 // Find the parent by-value. Must be in the previous level. Use while() to avoid GMP allocs.
468 // These were added left-to-right, so we can scan with an index efficiently.
469 Node<T>* parent = nullptr;
470 while (parent_index < _level_map[level - 1].size()) {
471 if (_level_map[level - 1][parent_index]->get_value() == parent_v) {
472 parent = _level_map[level - 1][parent_index];
473 break;
474 }
475 // Didnt' find it, bump the index.
476 parent_index++;
477 }
478 if (parent == nullptr) {
479 return sh.fail("couldn't find parent on level-1==" + to_string_any(level - 1) + " with parent value==" + to_string_any(parent_v));
480 }
481 // Link up parents and children.
482 new_node->assign_parent(parent);
483 parent->assign_child(new_node);
484 // Follow the same ancestor tracking logic from Node: inherit parent's, and if still null, see if parent is one.
485 new_node->assign_hwm_ancestor(parent->get_hwm_ancestor());
486 if (new_node->get_hwm_ancestor() == nullptr && parent->is_below_high_water_mark()) {
487 new_node->assign_hwm_ancestor(parent);
488 }
489 }
490 }
491
492 // All good
493 return true;
494 }

510 void add_level() {
511 // When we have no levels, this simply crafts a 0- or 1-based root node and manually sets level map and coverage.
512 if (_level_count == 0) {
513 _level_count = 1;
514 _root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
515 _level_map[_level_count].resize(1);
516 _level_map[_level_count][0] = _root_node;
517 _coverage_map[_level_count].set_covered(0);
518 _coverage_map[_level_count].set_total(1);
519 return;
520 }
521
522 // Get the parent and child levels. Then confirm the new one will fit. Then bump count.
523 level_t parent_level = _level_count;
524 level_t child_level = _level_count + 1;
525 _level_count++;
526
527 // Build the step value. Each level doubles the tree, so we need to respect T with GMP-size values.
528 T step = BinaryTreeMath<T>::st_step(parent_level);
529
530 // Calculate parent and child counts for looping and indexing.
531 // Note: size_t is safe, because no one can fit 2^64 nodes in RAM.
532 auto& parents = _level_map[parent_level];
533 size_t parent_count = parents.size();
534 size_t child_count = parent_count * 2;
535
536 // Resize the vector to contain all children. Otherwise assigning to indexes ([]) will fail.
537 _level_map[child_level].resize(child_count);
538
539 // Keep a record of covered or pruned values for coverage math after the loop. Use guaranteed-width integral.
540 uint64_t covered_or_pruned = 0;
541
542 // Create an external tracker for ancestors for OMP local threads to utilize by-ref.
543 std::vector<std::vector<Node<T>*>> omp_local_ancestors_group(omp_get_max_threads());
544
545 // Begin the critical section, but don't loop yet.
546 std::exception_ptr eptr = nullptr;
547 #pragma omp parallel reduction(+:covered_or_pruned) default(none) shared(parents, _level_map, step, child_level, parent_count, _is_pruning_hwm_nodes, _is_preserving_ancestors, omp_local_ancestors_group, _is_verifying_non_hwm_nodes, eptr)
548 {
549 try {
550 // Use local child values.
551 T child_value_1;
552 T child_value_2;
553
554 // Grab this thread's personal ancestors vector to avoid locking.
555 auto& my_ancestors = omp_local_ancestors_group[omp_get_thread_num()];
556
557 // Track this thread's local High-Water Mark. Since nodes are not sequentially ordered, it can only set this to first node - 1.
558 T my_high_water_mark = BinaryTreeMath<T>::st_first_node_value_of_level(_level_count) - 1;
559 if (my_high_water_mark < 1) {
560 my_high_water_mark = 1;
561 }
562
563 // Now Loop
564 #pragma omp for schedule(static)
565 for(size_t parent_idx = 0; parent_idx < parent_count; parent_idx++) {
566 // Find the parent.
567 Node<T>* parent = parents[parent_idx];
568
569 // Compute the child values. Avoid alloc() with GMP with arithmetic operators.
570 if constexpr(FixedWidthIntegral<T>) {
571 child_value_1 = parent->get_value() + step;
572 child_value_2 = child_value_1 + step;
573 } else if constexpr(GMPIntegral<T>) {
574 mpz_add(child_value_1.get_mpz_t(), parent->get_value().get_mpz_t(), step.get_mpz_t());
575 mpz_add(child_value_2.get_mpz_t(), child_value_1.get_mpz_t(), step.get_mpz_t());
576 }
577
578 // Instaniate the children.
579 Node<T>* child_1 = new Node<T>(child_value_1, parent);
580 Node<T>* child_2 = new Node<T>(child_value_2, parent);
581
582 // Tally them. Prune them if necessesary. Otherwise add them to the map.
583 // -- Child 1
584 bool assign_to_map = true;
585 if (_is_preserving_ancestors && child_1->is_below_high_water_mark() && ! child_1->has_high_water_mark_ancestor()) {
586 my_ancestors.push_back(new Node<T>(child_1->get_value()));
587 }
588 if (_is_verifying_non_hwm_nodes) {
589 if (child_1->is_below_high_water_mark() == false && child_1->has_high_water_mark_ancestor() == false) {
590 if (Collatz<T>::st_verify(child_value_1, my_high_water_mark) == false) {
591 throw std::logic_error("Node value " + to_string_any(child_value_1) + " didn't verify. How?");
592 }
593 }
594 }
595 if (child_1->is_below_high_water_mark() || child_1->has_high_water_mark_ancestor()) {
596 covered_or_pruned += 1;
597 if (_is_pruning_hwm_nodes) {
598 delete child_1;
599 child_1 = nullptr;
600 assign_to_map = false;
601 }
602 }
603 if (assign_to_map) {
604 _level_map[child_level][2 * parent_idx] = child_1;
605 if (! _is_pruning_hwm_nodes && ! _is_pruning_parent_levels) {
606 parent->assign_child(child_1);
607 }
608 }
609 // -- Child 2
610 assign_to_map = true;
611 if (_is_preserving_ancestors && child_2->is_below_high_water_mark() && ! child_2->has_high_water_mark_ancestor()) {
612 my_ancestors.push_back(new Node<T>(child_2->get_value()));
613 }
614 if (_is_verifying_non_hwm_nodes) {
615 if (child_2->is_below_high_water_mark() == false && child_2->has_high_water_mark_ancestor() == false) {
616 if (Collatz<T>::st_verify(child_value_2, my_high_water_mark) == false) {
617 throw std::logic_error("Node value " + to_string_any(child_value_2) + " didn't verify. How?");
618 }
619 }
620 }
621 if (child_2->is_below_high_water_mark() || child_2->has_high_water_mark_ancestor()) {
622 covered_or_pruned += 1;
623 if (_is_pruning_hwm_nodes) {
624 delete child_2;
625 child_2 = nullptr;
626 assign_to_map = false;
627 }
628 }
629 if (assign_to_map) {
630 _level_map[child_level][2 * parent_idx + 1] = child_2;
631 if (! _is_pruning_hwm_nodes && ! _is_pruning_parent_levels) {
632 parent->assign_child(child_2);
633 }
634 }
635 }
636 } catch (...) {
637 #pragma omp critical
638 {
639 if(!eptr) {
640 eptr = std::current_exception();
641 }
642 }
643 }
644 }
645
646 // Rethrow errors.
647 if(eptr != nullptr) {
648 std::rethrow_exception(eptr);
649 }
650
651 // Merge Ancestors.
652 if (_is_preserving_ancestors) {
653 for (std::vector<Node<T>*>& omp_ancestors : omp_local_ancestors_group) {
654 _ancestors.insert(_ancestors.end(), omp_ancestors.begin(), omp_ancestors.end());
655 }
656 // Always sort them.
657 tbb::parallel_sort(_ancestors.begin(), _ancestors.end(), [](const Node<T>* a, const Node<T>* b) {
658 if constexpr(FixedWidthIntegral<T>) {
659 return a->get_value() < b->get_value();
660 } else if constexpr(GMPIntegral<T>) {
661 return mpz_cmp(a->get_value().get_mpz_t(), b->get_value().get_mpz_t()) < 0;
662 }
663 });
664 }
665
666 // When pruning nodes,remove any nullptr (pruned) children from the vector to keep counts accurate on next loop.
667 if (_is_pruning_hwm_nodes) {
668 std::vector<Node<T>*>& children = _level_map[child_level];
669 children.erase(
670 std::remove(children.begin(), children.end(), nullptr),
671 children.end()
672 );
673 children.shrink_to_fit();
674 }
675
676 // When pruning parent levels, sweep them out.
677 // Nodes default-own children. Call own_children(false) for safety.
678 if (_is_pruning_parent_levels) {
679 // When it's level 1, clear up _root too so it doesn't double-free later.
680 if (parent_level == 1) {
681 _root_node = nullptr;
682 }
683 #pragma omp parallel for
684 for (size_t i = 0; i < _level_map[parent_level].size(); i++) {
685 Node<T>* parent = _level_map[parent_level][i];
686 parent->own_children(false);
687 delete parent;
688 }
689 _level_map[parent_level].clear();
690 _level_map[parent_level].shrink_to_fit();
691 }
692
693 // Coverage is tallied above. If pruning, logic is different because ancestors were purged already, which means their
694 // descendents were purged, but still count toward coverage. Luckily, by shrinking the vector above, this method can rely
695 // on .size() to report how many are *not* covered, which means covered = total - .size().
696 T total = step * 2;
697 T covered = covered_or_pruned;
698 if (_is_pruning_hwm_nodes) {
699 covered = total - _level_map[child_level].size();
700 }
701 _coverage_map[child_level] = BinaryTreeCoverage<T>(covered, total);
702 }
