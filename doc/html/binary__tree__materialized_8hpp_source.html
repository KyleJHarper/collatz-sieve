75 ~BinaryTreeMaterializedImpl() {
76 // Clean up ancestors. These are new-placement, so they aren't linked to the same memory as _level_map.
77 for (Node<T>* node : _ancestors) {
78 delete node;
79 }
80 _ancestors.clear();
81
82 // Now clear out the level_map. If pruning, handle it carefully.
83 if (_is_pruning_hwm_nodes) {
84 // Scan all levels manually and call delete explicitly.
85 for (level_t level = 0; level <= _level_count; level++) {
86 for (Node<T>* node : _level_map[level]) {
87 if (node != nullptr) {
88 node->own_children(false);
89 delete node;
90 }
91 }
92 }
93 } else {
94 // This will cascade to children because Node.own_children is default true.
95 delete _root_node;
96 }
97 }

232 size_t deep_size() const {
233 size_t total = sizeof(*this);
234
235 // Account for _level_map
236 total += sizeof(_level_map);
237 for (const auto& [level, nodes] : _level_map) {
238 total += sizeof(level);
239 total += sizeof(nodes);
240 total += nodes.capacity() * sizeof(Node<T>*);
241 for (const Node<T>* node : nodes) {
242 if (node) {
243 total += node->deep_size();
244 }
245 }
246 }
247
248 // Account for _coverage_map
249 total += sizeof(_coverage_map);
250 for (const auto& [level, coverage] : _coverage_map) {
251 total += sizeof(level);
252 total += sizeof(coverage);
253 }

255 // Unordered map buckets.
256 total += _level_map.bucket_count() * sizeof(void*);

278 static bool st_equal(const BinaryTreeMaterializedImpl<T>& first, const BinaryTreeMaterializedImpl<T>& second, std::string* err = nullptr) {
279 EqualityHelper eq(err);
280 eq.set_category("BinaryTreeMaterializedImpl");
281
282 // Flags
283 if (eq.unequal(first.is_pruning_hwm_nodes(), second.is_pruning_hwm_nodes())) {
284 return eq.fail("Is pruning hwm nodes mismatch");
285 }
286 if (eq.unequal(first.is_pruning_parent_levels(), second.is_pruning_parent_levels())) {
287 return eq.fail("Is pruning parent levels mismatch");
288 }
289
290 // Level Map
291 if (eq.unequal(first.get_level_map().size(), second.get_level_map().size())) {
292 return eq.fail("Level map count mismatch");
293 }
294 for (const auto& [level, f_nodes] : first.get_level_map()) {
295 if (! second.get_level_map().contains(level)) {
296 return eq.fail("First level map has level " + to_string_any(level) + ", but the second level map doesn't.");
297 }
298 const std::vector<Node<T>*>& s_nodes = second.get_level_map().at(level);
299 if (eq.unequal(f_nodes.size(), s_nodes.size())) {
300 return eq.fail("Node counts on level " + to_string_any(level) + " mismatch");
301 }
302 for (size_t i = 0; i < f_nodes.size(); i++) {
303 if(! Node<T>::st_equal(f_nodes.at(i), s_nodes.at(i), err)) {
304 return false;
305 }
306 }
307 }
308
309 // All good.
310 return true;
311 }

345 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
346 // !!! BIG FAT ANNOYING ERROR !!!
347 if (_is_pruning_hwm_nodes || _is_pruning_parent_levels) {
348 std::string msg = "You've activated my trap card!";
349 msg+= " Serializing and deserializing a materialized tree is already a bad idea.";
350 msg+= " Doing it with any form of node pruning enabled is NOT supported NOR a good idea.";
351 msg+= " Oh, and rebuilding a materialized tree is almost always faster than trying to import one.";
352 msg+= " Be well.";
353 throw std::logic_error(msg);
354 }
355
356 StreamHelper sh(nullptr, &out, err);
357 sh.set_category("BinaryTreeMaterializedImpl");
358
359 // Unique properties.
360 if (! sh.serialize_bool(_is_pruning_hwm_nodes)) {
361 return sh.fail("_is_pruning_hwm_nodes==" + std::to_string(_is_pruning_hwm_nodes));
362 }
363 if (! sh.serialize_bool(_is_pruning_parent_levels)) {
364 return sh.fail("_is_pruning_parent_levels==" + std::to_string(_is_pruning_parent_levels));
365 }
366
367 // Level Map
368 // Write how many levels we need to loop through later.
369 //
370 // Note: do NOT serialize level 1, because it's only the root node, which is a common item. It is linked in deserialize().
371 uint64_t u64_level_map_count = get_level_map().size() - 1;
372 if (! sh.serialize_integral(u64_level_map_count)) {
373 return sh.fail("level_map_count==" + to_string_any(u64_level_map_count));
374 }
375 // Get the level numbers in order for guaranteed write order.
376 std::vector<level_t> level_map_levels;
377 level_map_levels.reserve(_level_map.size());
378 for (const auto& [level, _] : _level_map) {
379 if (level == 1) { continue; }
380 level_map_levels.push_back(level);
381 }
382 std::sort(level_map_levels.begin(), level_map_levels.end());
383 // Loop through each level found.
384 for (level_t level : level_map_levels) {
385 const std::vector<Node<T>*>& nodes = _level_map.at(level);
386 // Write the level number and then the number of nodes on it.
387 if (! sh.serialize_integral(level)) {
388 return sh.fail("level map's level==" + to_string_any(level));
389 }
390 uint64_t u64_node_size = static_cast<uint64_t>(nodes.size());
391 if (! sh.serialize_integral(u64_node_size)) {
392 return sh.fail("level map on level==" + to_string_any(level) + " nodes.size()==" + to_string_any(nodes.size()));
393 }
394 // Write each node out.
395 for (const Node<T>* node : nodes) {
396 if (! node->serialize(out, err)) {
397 return sh.fail("level map node with value==" + to_string_any(node->get_value()));
398 }
399 }
400 }
401
402 // All good.
403 return true;
404 }

419 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
420 StreamHelper sh(&in, nullptr, err);
421 sh.set_category("BinaryTreeMaterializedImpl");
422
423 // Unique properties
424 bool b_tmp;
425 if (! sh.deserialize_bool(b_tmp)) {
426 return sh.fail("couldn't read _is_pruning_hwm_nodes");
427 }
428 _is_pruning_hwm_nodes = b_tmp;
429 if (! sh.deserialize_bool(b_tmp)) {
430 return sh.fail("couldn't read _is_pruning_parent_levels");
431 }
432 _is_pruning_parent_levels = b_tmp;
433
434 // Level map
435 // Start by assigning the root node to level 1, because it's not exported in serialize().
436 _level_map[1].resize(1);
437 _level_map[1][0] = _root_node;
438 // Now get the remaining levels, if any.
439 uint64_t u64_level_map_count = 0;
440 T parent_v;
441 T hwm_ancestor_v;
442 uint8_t child_count;
443 if (! sh.deserialize_integral(u64_level_map_count)) {
444 return sh.fail("couldn't read level_map_count");
445 }
446 for (uint64_t i = 0; i < u64_level_map_count; i++) {
447 uint64_t parent_index = 0;
448 level_t level = 0;
449 if (! sh.deserialize_integral(level)) {
450 return sh.fail("couldn't read level for index i==" + to_string_any(i));
451 }
452 uint64_t u64_node_size = 0;
453 if (! sh.deserialize_integral(u64_node_size)) {
454 return sh.fail("couldn't read node size for level==" + to_string_any(level));
455 }
456 size_t node_size = static_cast<size_t>(u64_node_size);
457 _level_map[level].reserve(node_size);
458 for (uint64_t j = 0; j < u64_node_size; j++) {
459 // Make a new heap-allocated node to deserialize into.
460 Node<T>* new_node = new Node<T>();
461 if (! new_node->deserialize(in, parent_v, hwm_ancestor_v, child_count, err)) {
462 return sh.fail("couldn't get node on level==" + to_string_any(level) + " and position j==" + to_string_any(j));
463 }
464 // Add it to the map.
465 _level_map[level].push_back(new_node);
466 // Find the parent by-value. Must be in the previous level. Use while() to avoid GMP allocs.
467 // These were added left-to-right, so we can scan with an index efficiently.
468 Node<T>* parent = nullptr;
469 while (parent_index < _level_map[level - 1].size()) {
470 if (_level_map[level - 1][parent_index]->get_value() == parent_v) {
471 parent = _level_map[level - 1][parent_index];
472 break;
473 }
474 // Didnt' find it, bump the index.
475 parent_index++;
476 }
477 if (parent == nullptr) {
478 return sh.fail("couldn't find parent on level-1==" + to_string_any(level - 1) + " with parent value==" + to_string_any(parent_v));
479 }
480 // Link up parents and children.
481 new_node->assign_parent(parent);
482 parent->assign_child(new_node);
483 // Follow the same ancestor tracking logic from Node: inherit parent's, and if still null, see if parent is one.
484 new_node->assign_hwm_ancestor(parent->get_hwm_ancestor());
485 if (new_node->get_hwm_ancestor() == nullptr && parent->is_below_high_water_mark()) {
486 new_node->assign_hwm_ancestor(parent);
487 }
488 }
489 }
490
491 // All good
492 return true;
493 }

509 void add_level() {
510 // When we have no levels, this simply crafts a 0- or 1-based root node and manually sets level map and coverage.
511 if (_level_count == 0) {
512 _level_count = 1;
513 _root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
514 _level_map[_level_count].resize(1);
515 _level_map[_level_count][0] = _root_node;
516 _coverage_map[_level_count].set_covered(0);
517 _coverage_map[_level_count].set_total(1);
518 return;
519 }
520
521 // Get the parent and child levels. Then confirm the new one will fit. Then bump count.
522 level_t parent_level = _level_count;
523 level_t child_level = _level_count + 1;
524 _level_count++;
525
526 // Build the step value. Each level doubles the tree, so we need to respect T with GMP-size values.
527 T step = BinaryTreeMath<T>::st_step(parent_level);
528
529 // Calculate parent and child counts for looping and indexing.
530 // Note: size_t is safe, because no one can fit 2^64 nodes in RAM.
531 auto& parents = _level_map[parent_level];
532 size_t parent_count = parents.size();
533 size_t child_count = parent_count * 2;
534
535 // Resize the vector to contain all children. Otherwise assigning to indexes ([]) will fail.
536 _level_map[child_level].resize(child_count);
537
538 // Keep a record of covered or pruned values for coverage math after the loop. Use guaranteed-width integral.
539 uint64_t covered_or_pruned = 0;
540
541 // Create an external tracker for ancestors for OMP local threads to utilize by-ref.
542 std::vector<std::vector<Node<T>*>> omp_local_ancestors_group(omp_get_max_threads());
543
544 // Begin the critical section, but don't loop yet.
545 std::exception_ptr eptr = nullptr;
546 #pragma omp parallel reduction(+:covered_or_pruned) default(none) shared(parents, _level_map, step, child_level, parent_count, _is_pruning_hwm_nodes, _is_preserving_ancestors, omp_local_ancestors_group, _is_verifying_non_hwm_nodes, eptr)
547 {
548 try {
549 // Use local child values.
550 T child_value_1;
551 T child_value_2;
552
553 // Grab this thread's personal ancestors vector to avoid locking.
554 auto& my_ancestors = omp_local_ancestors_group[omp_get_thread_num()];
555
556 // Track this thread's local High-Water Mark. Since nodes are not sequentially ordered, it can only set this to first node - 1.
557 T my_high_water_mark = BinaryTreeMath<T>::st_first_node_value_of_level(_level_count) - 1;
558 if (my_high_water_mark < 1) {
559 my_high_water_mark = 1;
560 }
561
562 // Now Loop
563 #pragma omp for schedule(static)
564 for(size_t parent_idx = 0; parent_idx < parent_count; parent_idx++) {
565 // Find the parent.
566 Node<T>* parent = parents[parent_idx];
567
568 // Compute the child values. Avoid alloc() with GMP with arithmetic operators.
569 if constexpr(BuiltinIntegral<T>) {
570 child_value_1 = parent->get_value() + step;
571 child_value_2 = child_value_1 + step;
572 } else if constexpr(GMPIntegral<T>) {
573 mpz_add(child_value_1.get_mpz_t(), parent->get_value().get_mpz_t(), step.get_mpz_t());
574 mpz_add(child_value_2.get_mpz_t(), child_value_1.get_mpz_t(), step.get_mpz_t());
575 }
576
577 // Instaniate the children.
578 Node<T>* child_1 = new Node<T>(child_value_1, parent);
579 Node<T>* child_2 = new Node<T>(child_value_2, parent);
580
581 // Tally them. Prune them if necessesary. Otherwise add them to the map.
582 // -- Child 1
583 bool assign_to_map = true;
584 if (_is_preserving_ancestors && child_1->is_below_high_water_mark() && ! child_1->has_high_water_mark_ancestor()) {
585 my_ancestors.push_back(new Node<T>(child_1->get_value()));
586 }
587 if (_is_verifying_non_hwm_nodes) {
588 if (child_1->is_below_high_water_mark() == false && child_1->has_high_water_mark_ancestor() == false) {
589 if (Collatz<T>::st_verify(child_value_1, my_high_water_mark) == false) {
590 throw std::logic_error("Node value " + to_string_any(child_value_1) + " didn't verify. How?");
591 }
592 }
593 }
594 if (child_1->is_below_high_water_mark() || child_1->has_high_water_mark_ancestor()) {
595 covered_or_pruned += 1;
596 if (_is_pruning_hwm_nodes) {
597 delete child_1;
598 child_1 = nullptr;
599 assign_to_map = false;
600 }
601 }
602 if (assign_to_map) {
603 _level_map[child_level][2 * parent_idx] = child_1;
604 if (! _is_pruning_hwm_nodes && ! _is_pruning_parent_levels) {
605 parent->assign_child(child_1);
606 }
607 }
608 // -- Child 2
609 assign_to_map = true;
610 if (_is_preserving_ancestors && child_2->is_below_high_water_mark() && ! child_2->has_high_water_mark_ancestor()) {
611 my_ancestors.push_back(new Node<T>(child_2->get_value()));
612 }
613 if (_is_verifying_non_hwm_nodes) {
614 if (child_2->is_below_high_water_mark() == false && child_2->has_high_water_mark_ancestor() == false) {
615 if (Collatz<T>::st_verify(child_value_2, my_high_water_mark) == false) {
616 throw std::logic_error("Node value " + to_string_any(child_value_2) + " didn't verify. How?");
617 }
618 }
619 }
620 if (child_2->is_below_high_water_mark() || child_2->has_high_water_mark_ancestor()) {
621 covered_or_pruned += 1;
622 if (_is_pruning_hwm_nodes) {
623 delete child_2;
624 child_2 = nullptr;
625 assign_to_map = false;
626 }
627 }
628 if (assign_to_map) {
629 _level_map[child_level][2 * parent_idx + 1] = child_2;
630 if (! _is_pruning_hwm_nodes && ! _is_pruning_parent_levels) {
631 parent->assign_child(child_2);
632 }
633 }
634 }
635 } catch (...) {
636 #pragma omp critical
637 {
638 if(!eptr) {
639 eptr = std::current_exception();
640 }
641 }
642 }
643 }
644
645 // Rethrow errors.
646 if(eptr != nullptr) {
647 std::rethrow_exception(eptr);
648 }
649
650 // Merge Ancestors.
651 if (_is_preserving_ancestors) {
652 for (std::vector<Node<T>*>& omp_ancestors : omp_local_ancestors_group) {
653 _ancestors.insert(_ancestors.end(), omp_ancestors.begin(), omp_ancestors.end());
654 }
655 // Always sort them.
656 tbb::parallel_sort(_ancestors.begin(), _ancestors.end(), [](const Node<T>* a, const Node<T>* b) {
657 if constexpr(BuiltinIntegral<T>) {
658 return a->get_value() < b->get_value();
659 } else {
660 return mpz_cmp(a->get_value().get_mpz_t(), b->get_value().get_mpz_t()) < 0;
661 }
662 });
663 }
664
665 // When pruning nodes,remove any nullptr (pruned) children from the vector to keep counts accurate on next loop.
666 if (_is_pruning_hwm_nodes) {
667 std::vector<Node<T>*>& children = _level_map[child_level];
668 children.erase(
669 std::remove(children.begin(), children.end(), nullptr),
670 children.end()
671 );
672 children.shrink_to_fit();
673 }
674
675 // When pruning parent levels, sweep them out.
676 // Nodes default-own children. Call own_children(false) for safety.
677 if (_is_pruning_parent_levels) {
678 // When it's level 1, clear up _root too so it doesn't double-free later.
679 if (parent_level == 1) {
680 _root_node = nullptr;
681 }
682 #pragma omp parallel for
683 for (size_t i = 0; i < _level_map[parent_level].size(); i++) {
684 Node<T>* parent = _level_map[parent_level][i];
685 parent->own_children(false);
686 delete parent;
687 }
688 _level_map[parent_level].clear();
689 _level_map[parent_level].shrink_to_fit();
690 }
691
692 // Coverage is tallied above. If pruning, logic is different because ancestors were purged already, which means their
693 // descendents were purged, but still count toward coverage. Luckily, by shrinking the vector above, this method can rely
694 // on .size() to report how many are *not* covered, which means covered = total - .size().
695 T total = step * 2;
696 T covered = covered_or_pruned;
697 if (_is_pruning_hwm_nodes) {
698 covered = total - _level_map[child_level].size();
699 }
700 _coverage_map[child_level] = BinaryTreeCoverage<T>(covered, total);
701 }
