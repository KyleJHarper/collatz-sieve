102 template<AnySupportedIntegral U>
103 inline bool verify_with_detailed_metrics(U& value, ThreadStorage& thread_storage) {
104 // Test for overflow.
105 // This method does NOT use striding for obvious reasons. Lean on the precomputed max initial values tables.
106 if constexpr(FixedWidthIntegral<U>) {
107 if constexpr(sizeof(U) * 8 < 64) {
108 // Type is less than 64 bits, just send it as a 64-bit. Even uint32_t can't reach an overflow of 64-bit.
109 // Max initial value of 64 bit is 12,327,829,502 which is greater than 2^32 which is 4,294,967,296.
110 uint64_t safe_value = value;
111 return verify_with_detailed_metrics(safe_value, thread_storage);
112 } else if constexpr(sizeof(U) * 8 == 64) {
113 // Dealing with T of exactly 64 bits. If it'll overflow, use 128-bit. Otherwise, pass through.
114 if (value > CollatzConstants::get_max_initial_value_by_type<U>()) {
115 uint128_t safe_value = value;
116 return verify_with_detailed_metrics(safe_value, thread_storage);
117 }
118 } else if constexpr(sizeof(U) * 8 == 128) {
119 // Dealing with 128 bits. If it'll overflow, use mpz_class.
120 if (value > CollatzConstants::get_max_initial_value_by_type<U>()) {
121 static thread_local mpz_class safe_value;
122 Int128::uint128_to_mpz(value, safe_value);
123 return verify_with_detailed_metrics(safe_value, thread_storage);
124 }
125 }
126 }
127
128 // May now process value, guaranteed to not overflow.
129 bool below_hwm = false;
130 size_t steps = 0;
131 using StrideTable = std::conditional_t<
132 FixedWidthIntegral<T>
133 , AffineStride::VerifyFWTable
134 , AffineStride::VerifyGMPTable
135 >;
136 if constexpr(FixedWidthIntegral<U>) {
137 U initial_value = value;
138 while (value != 1) {
139 if ((value & 1) == 0) {
140 value >>= 1;
141 } else {
142 value = (value << 1) + value + 1;
143 }
144 // Tally the local step count.
145 steps++;
146
147 // High-Water Mark Metrics
148 if (below_hwm) {
149 thread_storage.metrics.steps_skippable_by_hwm++;
150 } else {
151 if (value < initial_value) {
152 below_hwm = true;
153 // Affine stride is no longer effective. Tally it by counting how many strides it would've taken.
154 thread_storage.metrics.steps_skippable_by_affine_stride_before_hwm += (steps - (1 + (steps / StrideTable::STRIDE_SIZE)));
155 }
156 }
157 }
158 } else if constexpr(GMPIntegral<U>) {
159 static thread_local U initial_value;
160 initial_value = value;
161 while (mpz_cmp_ui(value.get_mpz_t(), 1) != 0) {
162 if (mpz_even_p(value.get_mpz_t())) {
163 mpz_tdiv_q_2exp(value.get_mpz_t(), value.get_mpz_t(), 1);
164 } else {
165 mpz_mul_ui(value.get_mpz_t(), value.get_mpz_t(), 3);
166 mpz_add_ui(value.get_mpz_t(), value.get_mpz_t(), 1);
167 }
168
169 // Tally the local step count.
170 steps++;
171
172 // High-Water Mark Metrics
173 if (below_hwm) {
174 thread_storage.metrics.steps_skippable_by_hwm++;
175 } else {
176 if (value < initial_value) {
177 below_hwm = true;
178 // Affine stride is no longer effective. Tally it by counting how many strides it would've taken.
179 thread_storage.metrics.steps_skippable_by_affine_stride_before_hwm += (steps - (1 + (steps / StrideTable::STRIDE_SIZE)));
180 }
181 }
182 }
183 }
184 // Add the local steps to the thread local total.
185 thread_storage.metrics.steps_total += steps;
186
187 // Affine Stride Metric
188 // This is simply the number of whole strides that could've been taken times the number of steps saved.
189 thread_storage.metrics.steps_skippable_by_affine_stride += ((steps / StrideTable::STRIDE_SIZE) * (StrideTable::STRIDE_SIZE - 1));
190
191 // All done. Return true.
192 return true;
193 }

230 template<bool UseIVTable, bool DetailedMetrics>
231 void run_executor_impl() {
232 // Reset variables.
233 _tls.clear();
234
235 // Process all the uncovered values.
236 this->_tree.for_each_uncovered_value_with_tls(ForEachPolicy::PARALLEL, _tls, [&](T& value, ThreadStorage& thread_storage) {
237 // Stop if done.
238 if (this->_end_value > 0 && value > this->_end_value) {
239 thread_storage.metrics.flush(this->_published_metrics, true);
240 return ForEachSignal::BREAK;
241 }
242
243 // Synchronize state and metrics if necessary.
244 if (--thread_storage.synchronization_countdown == 0) {
245 // Reset the countdown for next time. Bump the counter for metrics.
246 thread_storage.synchronization_countdown = this->_synchronization_countdown;
247 thread_storage.synchronizations_performed++;
248
249 // Flush metrics.
250 thread_storage.metrics.flush(this->_published_metrics, true);
251
252 // Wait when paused.
253 while (this->_state.load(std::memory_order_relaxed) == VerifierState::PAUSED) {
254 std::this_thread::sleep_for(std::chrono::milliseconds(this->_paused_check_ms));
255 }
256
257 // Quit if necessary.
258 if (this->_state.load(std::memory_order_relaxed) == VerifierState::STOPPING) {
259 return ForEachSignal::BREAK;
260 }
261 }
262
263 // Verify using whichever method is requested: with or without metrics.
264 if constexpr(DetailedMetrics) {
265 if (! verify_with_detailed_metrics(value, thread_storage)) {
266 throw std::logic_error("Failed to validate value: " + to_string_any(value));
267 }
268 } else {
269 // Handles overflow detection and promotes automatically.
270 if (! Collatz<T>::template st_verify<T, UseIVTable>(value, value)) {
271 throw std::logic_error("Failed to validate value: " + to_string_any(value));
272 }
273 }
274
275 // Always bump the number of nodes verified, even without DetailedMetrics enabled.
276 thread_storage.metrics.nodes_verified++;
277
278 // Continue
279 return ForEachSignal::CONTINUE;
280 }, this->_start_value);
281
282 // Set state to STOPPED. Only this thread may do that.
283 this->_state.store(VerifierState::STOPPED);
284
285 // Synchronize the timer since it's done doing any work.
286 this->sync_timer();
287 }
