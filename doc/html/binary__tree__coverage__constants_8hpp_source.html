83 template<AnySupportedIntegral T>
84 inline T get_known_coverage(level_t level) {
85 // Sanity checks.
86 if (level > MAX_KNOWN_COVERAGE_LEVEL) {
87 throw std::out_of_range("Level " + std::to_string(level) + " not found in KNOWN_COVERAGE");
88 }
89
90 // Pick the right type, and convert as-needed.
91 if constexpr(FixedWidthIntegral<T>) {
92 if (level > sizeof(T) * 8) {
93 throw std::out_of_range("Level " + std::to_string(level) + " is outside the max range for your type T: " + ABI::demangle<T>());
94 }
95 return static_cast<T>(KNOWN_COVERAGE_MAP[level]);
96 } else if constexpr(GMPIntegral<T>) {
97 return Int128::uint128_to_mpz(KNOWN_COVERAGE_MAP[level]);
98 } else {
99 throw std::logic_error("Unknown data type");
100 }
101
102 // Should never reach this.
103 throw std::logic_error("Unknown type get_known_coverage()");
104 }
