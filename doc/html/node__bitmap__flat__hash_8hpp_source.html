164 void add_range(const T& start, const T& end) {
165 // Prefix can be mpz_class, so use TLS and out&.
166 static thread_local prefix_t start_prefix;
167 static thread_local prefix_t end_prefix;
168 if constexpr(FixedWidthIntegral<T>) {
169 start_prefix = Traits::get_prefix(start);
170 end_prefix = Traits::get_prefix(end);
171 } else if constexpr(GMPIntegral<T>) {
172 Traits::get_prefix(start, start_prefix);
173 Traits::get_prefix(end, end_prefix);
174 }
175 // Suffix is just a uint32_t.
176 suffix_t start_suffix = Traits::get_suffix(start);
177 suffix_t end_suffix = Traits::get_suffix(end);
178
179 if (start_prefix == end_prefix) {
180 // They all fall into the same prefix. Call a single addRange() and then leave.
181 auto [it, inserted] = _flat_map.try_emplace(start_prefix);
182 if (inserted) {
183 add_prefix_key(start_prefix);
184 }
185 it->second.addRange(start_suffix, end_suffix);
186 } else {
187 // Insert the remainder of prefix_1 which clearly goes to SUFFIX_MAX since prefix_2 is beyond it.
188 // Since the suffix is full, we use addRangeClosed() to encapsulate it.
189 auto [first_it, first_inserted] = _flat_map.try_emplace(start_prefix);
190 if (first_inserted) {
191 add_prefix_key(start_prefix);
192 }
193 first_it->second.addRangeClosed(start_suffix, Traits::SUFFIX_MAX);
194
195 // Loop through any intermediary prefixes, which are incremental of course between prefix_start and prefix_end, non-inclusive.
196 // Again, since the suffix is guaranteed full, we use addRangeClosed() to encapsulate it.
197 for (prefix_t middle_prefix = start_prefix + 1; middle_prefix < end_prefix; middle_prefix++) {
198 auto [mid_it, mid_inserted] = _flat_map.try_emplace(middle_prefix);
199 if (mid_inserted) {
200 add_prefix_key(middle_prefix);
201 }
202 mid_it->second.addRangeClosed(0, Traits::SUFFIX_MAX);
203 }
204
205 // Add the remaining bits to the last prefix, ending at "end" of course.
206 // Reminder, we do NOT include the last value (end_suffix) because this is the half-open variant.
207 auto [last_it, last_inserted] = _flat_map.try_emplace(end_prefix);
208 if (last_inserted) {
209 add_prefix_key(end_prefix);
210 }
211 last_it->second.addRange(0, end_suffix);
212 }
213 }

234 void add_many(const size_t count, T* values) {
235 // Safety check.
236 if (count < 1) {
237 return;
238 }
239
240 // Sort the values.
241 T* end = values + count;
242 if (count < 100000) {
243 std::sort(values, end);
244 } else {
245 tbb::parallel_sort(values, end);
246 }
247
248 // Loop through the values one time, bumping the prefix only when a change is needed.
249 // By computing the prefix's max value, it can be compared in the loop instead of extracting each value's prefix.
250 prefix_t current_prefix = Traits::get_prefix(values[0]);
251 T prefix_max_value = (static_cast<T>(current_prefix + 1) << Traits::SUFFIX_BITS) - 1;
252 if constexpr(FixedWidthIntegral<T>) {
253 if (current_prefix == std::numeric_limits<prefix_t>::max()) {
254 prefix_max_value = std::numeric_limits<T>::max();
255 }
256 }
257 const size_t BUFFER_SIZE = size_t(1) << 16; // 64K * sizeof()
258 std::vector<suffix_t> suffixes;
259 suffixes.reserve(BUFFER_SIZE);
260 for(const T* value = values; value < end; value++) {
261 // Bump the prefix if necessary. Must flush too.
262 if (*value > prefix_max_value) {
263 // Must flush the current buffer due to new prefix.
264 add_many_with_prefix(current_prefix, suffixes.size(), suffixes.data());
265 suffixes.clear();
266
267 // Create new prefix and max value.
268 current_prefix = Traits::get_prefix(*value);
269 prefix_max_value = (static_cast<T>(current_prefix + 1) << Traits::SUFFIX_BITS) - 1;
270 if constexpr(FixedWidthIntegral<T>) {
271 if (current_prefix == std::numeric_limits<prefix_t>::max()) {
272 prefix_max_value = std::numeric_limits<T>::max();
273 }
274 }
275 }
276
277 // Add the suffix to the buffer.
278 suffixes.push_back(Traits::get_suffix(*value));
279
280 // Flush if necessary.
281 if (suffixes.size() >= suffixes.capacity()) {
282 add_many_with_prefix(current_prefix, suffixes.size(), suffixes.data());
283 suffixes.clear();
284 }
285 }
286
287 // Flush the remaining buffer, if it has any data.
288 if (suffixes.empty() == false) {
289 add_many_with_prefix(current_prefix, suffixes.size(), suffixes.data());
290 suffixes.clear();
291 }
292 }

364 T minimum() const {
365 if (empty()) {
366 throw std::out_of_range("Cannot request NodeBitmap minimum() when the bitmap is empty.");
367 }
368
369 // Since prefixes are sorted, the first one has the lowest values.
370 const prefix_t& prefix = _sorted_prefixes.front();
371 const roaring::Roaring& roaring_obj = _flat_map.find(prefix)->second;
372
373 // Shift and return the minimum. When T is small, prefix is nada.
374 T value = 0;
375 if constexpr(sizeof(T) > Traits::SUFFIX_BYTES) {
376 if constexpr(FixedWidthIntegral<T>) {
377 value = (static_cast<T>(prefix) << Traits::SUFFIX_BITS) + roaring_obj.minimum();
378 } else if constexpr(GMPIntegral<T>) {
379 mpz_mul_2exp(value.get_mpz_t(), prefix.get_mpz_t(), Traits::SUFFIX_BITS);
380 mpz_add_ui(value.get_mpz_t(), value.get_mpz_t(), roaring_obj.minimum());
381 }
382 } else {
383 // T is smaller than suffix_t, which means the only prefix it'll ever have is 0.
384 // Set it manually, because small-T means it can't shift SUFFIX_BITS wide.
385 value = 0 + roaring_obj.minimum();
386 }
387
388 return value;
389 }

454 void merge(const FlatHashBitmapImpl<T>& src) {
455 for (const auto& [src_prefix, src_bitmap] : src.get_map()) {
456 // Get an iterator to our keypair.

559 static bool st_equal(const FlatHashBitmapImpl<T>& first, const FlatHashBitmapImpl<T>& second, std::string* err = nullptr) {
560 EqualityHelper eq(err);
561 eq.set_category("FlatHashBitmapImpl");
562
563 // Grab the flat hash maps and prefixes.
564 const map_t& f_flat_map = first.get_map();
565 const map_t& s_flat_map = second.get_map();
566 const std::vector<prefix_t>& f_prefixes = first.get_sorted_prefixes();
567 const std::vector<prefix_t>& s_prefixes = second.get_sorted_prefixes();
568 // Check size.
569 if (! eq.equal(f_prefixes.size(), s_prefixes.size())) {
570 return eq.fail("Flat hash impl sorted_prefxies size mistmatch");
571 }
572 // Prefixes should match and are already sorted. Test each prefix, and then the CRoaring bitmap beneath it.
573 for (size_t i = 0; i < f_prefixes.size(); i++) {
574 if (! eq.equal(f_prefixes.at(i), s_prefixes.at(i))) {
575 return eq.fail("Flat hash prefix at ID " + to_string_any(i) + " mismatch");
576 }
577 const roaring::Roaring& f_roaring = f_flat_map.at(f_prefixes.at(i));
578 const roaring::Roaring& s_roaring = s_flat_map.at(s_prefixes.at(i));
579 if (! roaring::api::roaring_bitmap_equals(&(f_roaring.roaring), &(s_roaring.roaring))) {
580 return eq.fail("Bitmaps do not match for prefix " + to_string_any(f_prefixes.at(i)));
581 }
582 }
583
584 // Guess they're equal.
585 return true;
586 }

618 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
619 StreamHelper sh(nullptr, &out, err);
620 sh.set_category("FlatHashBitmapImpl");
621
622 // Write out the number of prefixes/bitmaps.
623 uint64_t u64_prefix_count = static_cast<uint64_t>(_sorted_prefixes.size());
624 if (! sh.serialize_integral(u64_prefix_count)) {
625 return sh.fail("prefix_count==" + to_string_any(u64_prefix_count));
626 }
627
628 // Now loop through each key-value pair. Make sure to do them in order so exported files remain deterministic.
629 for (const prefix_t& prefix : _sorted_prefixes) {
630 // Write the prefix.
631 if (! sh.serialize_integral(prefix)) {
632 return sh.fail("prefix==" + to_string_any(prefix));
633 }
634 // Find the bitmap.
635 auto it = _flat_map.find(prefix);
636 if (it == _flat_map.end()) {
637 return sh.fail("couldn't find bitmap for prefix==" + to_string_any(prefix));
638 }
639 const roaring::Roaring& bitmap = it->second;
640 // Write how much space the bitmap will take.
641 uint64_t u64_bitmap_size = static_cast<uint64_t>(bitmap.getSizeInBytes(true));
642 if (! sh.serialize_integral(u64_bitmap_size)) {
643 return sh.fail("bitmap_size==" + to_string_any(u64_bitmap_size));
644 }
645 // Build the buffer and flush it.
646 std::vector<char> buffer(u64_bitmap_size);
647 size_t actual_bytes = bitmap.write(buffer.data(), true);
648 if (actual_bytes != u64_bitmap_size) {
649 return sh.fail("Number of bytes to write expected by getSizeInBytes doesn't match actual bytes returned");
650 }
651 if (! sh.write_bytes(buffer.data(), u64_bitmap_size)) {
652 return sh.fail("roaring bitmap data");
653 }
654 }
655
656 // All good.
657 return true;
658 }

672 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
673 StreamHelper sh(&in, nullptr, err);
674 sh.set_category("FlatHashBitmapImpl");
675
676 // Clear it out (reset).
677 clear();
678
679 // Prefixes
680 uint64_t u64_prefix_count = 0;
681 if (! sh.deserialize_integral(u64_prefix_count)) {
682 return sh.fail("couldn't read prefix count");
683 }
684 for (uint64_t i = 0; i < u64_prefix_count; i++) {
685 prefix_t prefix;
686 if (! sh.deserialize_integral(prefix)) {
687 return sh.fail("couldn't read prefix value for index i==" + to_string_any(i));
688 }
689 uint64_t u64_bitmap_size = 0;
690 if (! sh.deserialize_integral(u64_bitmap_size)) {
691 return sh.fail("couldn't read bitmap size for prefix==" + to_string_any(prefix));
692 }
693 std::vector<char> buffer(u64_bitmap_size);
694 if (! sh.read_bytes(buffer.data(), static_cast<size_t>(u64_bitmap_size))) {
695 return sh.fail("couldn't read bitmap data for prefix==" + to_string_any(prefix));
696 }
697 // Try to place the prefix, which will build out bitmap too.
698 auto [it, inserted] = _flat_map.try_emplace(prefix);
699 if (inserted) {
700 add_prefix_key(prefix);
701 }
702 it->second = roaring::Roaring::read(buffer.data(), true);
703 }
704
705 // All good
706 return true;
707 }

729 template<typename Func>
730 void for_each_value(ForEachPolicy policy, Func&& callback) const {
731 // Do not allow non-ref callbacks. Otherwise we make GMP over and over.
732 static_assert(std::is_invocable_v<Func, T&>, "Callback must be callable with (T&)");
733 // Require ForEachSignal return type.
734 static_assert(std::is_same_v<std::invoke_result_t<Func, T&>, ForEachSignal>, "Callback must return ForEachSignal");
735
736 std::vector<uint8_t> dummy_tls;
737 for_each_value_with_tls(policy, dummy_tls, [&](T& value, auto&) {
738 return callback(value);
739 });
740 }

772 template<typename Func, typename TLS_Type>
773 void for_each_value_with_tls(ForEachPolicy policy, std::vector<TLS_Type>& tls, Func&& callback) const {
774 // Do not allow non-ref callbacks. Otherwise we make GMP over and over.
775 static_assert(std::is_invocable_v<Func, T&, TLS_Type&>, "Callback must be callable with (T&, TLS_Type&)");
776 // Require ForEachSignal return type.
777 static_assert(std::is_same_v<std::invoke_result_t<Func, T&, TLS_Type&>, ForEachSignal>, "Callback must return ForEachSignal");
778
779 // Ensure the TLS has enough elements before proceeding.
780 const size_t max_threads = policy == ForEachPolicy::SERIAL ? 1 : static_cast<size_t>(omp_get_max_threads());
781 if (tls.size() < max_threads) {
782 tls.resize(max_threads);
783 }
784
785 // Setup a control-stop.
786 std::atomic<ForEachSignal> atomic_signal = ForEachSignal::CONTINUE;
787 ForEachSignal signal = ForEachSignal::CONTINUE;
788
789 // Process according to policy.
790 if (policy == ForEachPolicy::SERIAL) {
791 // Serial Path
792 // Guarantees sequential processing. Can efficiently use exposed iterators.
793
794 // Hoist these, just in case of temp allocs.
795 T value;
796 T shifted_prefix;
797 suffix_t suffix;
798
799 // Even though this path is not threaded, get a thread ID for clarity.
800 size_t my_thread_id = static_cast<size_t>(omp_get_thread_num());
801
802 // The TLS should only need index 0.
803 TLS_Type& my_tls = tls[my_thread_id];
804
805 // Loop through prefixes in order.
806 for (const prefix_t& prefix : _sorted_prefixes) {
807 // Shift the prefix once instead of deeper inside the while() loop. For smaller types, set to 0.
808 if constexpr(sizeof(T) > Traits::SUFFIX_BYTES) {
809 // T can hold the prefix. Let it shift.
810 if constexpr(FixedWidthIntegral<T>) {
811 shifted_prefix = static_cast<T>(prefix) << Traits::SUFFIX_BITS;
812 } else if constexpr(GMPIntegral<T>) {
813 mpz_mul_2exp(shifted_prefix.get_mpz_t(), prefix.get_mpz_t(), Traits::SUFFIX_BITS);
814 }
815 } else {
816 // T is smaller than suffix_t, which means the only prefix it'll ever have is 0.
817 // Set it manually, because small-T means it can't shift SUFFIX_BITS wide.
818 shifted_prefix = 0;
819 }
820
821 // Find the bitmap and connect an iterator.
822 const roaring::Roaring& roaring_obj = _flat_map.find(prefix)->second;
823 roaring::api::roaring_uint32_iterator_t bitmap_iterator;
824 roaring::api::roaring_iterator_init(&roaring_obj.roaring, &bitmap_iterator);
825
826 // Loop this iterator until done. Use addition instead of bitwise-OR (|) to avoid alloc on GMP path.
827 while (bitmap_iterator.has_value) {
828 suffix = bitmap_iterator.current_value;
829 if constexpr(FixedWidthIntegral<T>) {
830 value = shifted_prefix + suffix;
831 } else if constexpr(GMPIntegral<T>) {
832 mpz_add_ui(value.get_mpz_t(), shifted_prefix.get_mpz_t(), suffix);
833 }
834 signal = callback(value, my_tls);
835 if (signal == ForEachSignal::BREAK) { break; }
836 roaring::api::roaring_uint32_iterator_advance(&bitmap_iterator);
837 }
838
839 // If stopping, break from the prefix for() loop too.
840 if (signal == ForEachSignal::BREAK) { break; }
841 }
842 } else {
843 // Parallel Path
844 // Work is threaded, breaking sequential guarantee. Uses CRoaring internals for performance.
845 //
846 // While throwing _sorted_prefixes at OMP might seem ideal, it would result in only a single thread working until level
847 // 32. This is because CRoaring uses a 32-bit map (16 bit key, 16 bit suffix) internally.
848 //
849 // As such, this will read the internal high-low containers and iterate based on their type (array, bitmap, RLE).
850
851 // Open the parallel region early and establish thread-locals.
852 #pragma omp parallel default(none) shared(tls, atomic_signal, callback)
853 {
854 // Get a thread ID.
855 size_t my_thread_id = static_cast<size_t>(omp_get_thread_num());
856
857 // Pick an element from the TLS.
858 TLS_Type& my_tls = tls[my_thread_id];
859
860 // Track a local-stop.
861 ForEachSignal local_signal = ForEachSignal::CONTINUE;
862
863 // Hoist a few locals to avoid lots of potential alloc on GMP path.
864 T value;
865 T shifted_prefix;
866 T shifted_prefix_and_key;
867
868 // Loop through prefixes.
869 for (size_t prefix_index = 0; prefix_index < _sorted_prefixes.size(); prefix_index++) {
870 // Do not test for atomic signal here. Doing so before an OMP-for region is a race condition.
871 // Grab the prefix and then build a shifted_prefix so it isn't recomputed.
872 const prefix_t& prefix = _sorted_prefixes[prefix_index];
873 // Shift the prefix once instead of deeper inside the while()/for() loops. For smaller types, set to 0.
874 if constexpr(sizeof(T) > Traits::SUFFIX_BYTES) {
875 // T can hold the prefix. Let it shift.
876 if constexpr(FixedWidthIntegral<T>) {
877 shifted_prefix = static_cast<T>(prefix) << Traits::SUFFIX_BITS;
878 } else if constexpr(GMPIntegral<T>) {
879 mpz_mul_2exp(shifted_prefix.get_mpz_t(), prefix.get_mpz_t(), Traits::SUFFIX_BITS);
880 }
881 } else {
882 // T is smaller than suffix_t, which means the only prefix it'll ever have is 0.
883 // Set it manually, because small-T means it can't shift SUFFIX_BITS wide.
884 shifted_prefix = 0;
885 }
886
887 // Find our bitmap and connect to the internal storage components.
888 auto it = _flat_map.find(prefix);
889 const roaring::api::roaring_bitmap_t* roaring_bitmap = &(it->second.roaring);
890 const roaring::api::roaring_array_t* high_low_container = &(roaring_bitmap->high_low_container);
891 // Hoist container bits. Probably optimized away, but meh.
892 const auto* high_low_container__keys = high_low_container->keys;
893 const auto* high_low_container__containers = high_low_container->containers;
894 const auto* high_low_container__typecodes = high_low_container->typecodes;
895
896 // Build the inner loop around the keys and sub containers. Split it up with OMP.
897 #pragma omp for schedule(dynamic)
898 for (int high_low_index = 0; high_low_index < high_low_container->size; high_low_index++) {
899 // Check for stop again at the chunk level.
900 if (atomic_signal.load(std::memory_order_relaxed) == ForEachSignal::BREAK) { continue; }
901
902 // Grab the roaring key, the container, and type code.
903 roaring_key_t suffix_key = high_low_container__keys[high_low_index];
904 const roaring::api::container_s* container = high_low_container__containers[high_low_index];
905 roaring_typecode_t typecode = high_low_container__typecodes[high_low_index];
906
907 // Hoist and promote+shift the key into full suffix_t space.
908 suffix_t shifted_key = (suffix_t(suffix_key)) << Traits::ROARING_KEY_BITS;
909 suffix_t suffix_val;
910
911 // Build the shifted prefix and key combo so it isn't repeated below.
912 if constexpr(FixedWidthIntegral<T>) {
913 shifted_prefix_and_key = shifted_prefix + shifted_key;
914 } else if constexpr(GMPIntegral<T>) {
915 mpz_add_ui(shifted_prefix_and_key.get_mpz_t(), shifted_prefix.get_mpz_t(), shifted_key);
916 }
917
918 // Select the correct iteration method based on the typecode to retrieve and process suffixes.
919 // Use addition instead of bitwise-OR (|) to avoid alloc on GMP path.
920 switch (typecode) {
921 case ARRAY_CONTAINER_TYPE: {
922 const roaring::internal::array_container_t* array_container = (roaring::internal::array_container_t*)container;
923 for (int array_index = 0; array_index < array_container->cardinality; array_index++) {
924 suffix_val = array_container->array[array_index];
925 if constexpr(FixedWidthIntegral<T>) {
926 value = shifted_prefix_and_key + suffix_val;
927 } else if constexpr(GMPIntegral<T>) {
928 mpz_add_ui(value.get_mpz_t(), shifted_prefix_and_key.get_mpz_t(), suffix_val);
929 }
930 local_signal = callback(value, my_tls);
931 if (local_signal == ForEachSignal::BREAK) {
932 atomic_signal.store(ForEachSignal::BREAK, std::memory_order_relaxed);
933 break;
934 }
935 }
936 break;
937 }
938
939 case BITSET_CONTAINER_TYPE: {
940 const roaring::internal::bitset_container_t* bitset_container = (const roaring::internal::bitset_container_t*)container;
941 for (int word_index = 0; word_index < roaring::internal::BITSET_CONTAINER_SIZE_IN_WORDS; word_index++) {
942 roaring_word_t word = bitset_container->words[word_index];
943 // Loop until there are no more on-bits in the word.
944 while (word != 0) {
945 // Calculate the offset by counting the number of trailing zeros, giving us the next "1" position.
946 int offset = __builtin_ctzll(word);
947 // Multiply out the word index by the number of bits per word, then add the offset. That's the next "on" node.
948 suffix_val = (word_index * Traits::ROARING_WORD_BITS) + offset;
949 if constexpr(FixedWidthIntegral<T>) {
950 value = shifted_prefix_and_key + suffix_val;
951 } else if constexpr(GMPIntegral<T>) {
952 mpz_add_ui(value.get_mpz_t(), shifted_prefix_and_key.get_mpz_t(), suffix_val);
953 }
954 local_signal = callback(value, my_tls);
955 if (local_signal == ForEachSignal::BREAK) {
956 atomic_signal.store(ForEachSignal::BREAK, std::memory_order_relaxed);
957 break;
958 }
959 // Now bitwise-AND the word minus one to wipe out the least-significant 1s place which was just processed.
960 word &= (word - 1);
961 }
962 }
963 break;
964 }
965
966 case RUN_CONTAINER_TYPE: {
967 const roaring::internal::run_container_t* run_container = (const roaring::internal::run_container_t*)container;
968 for (int run_index = 0; run_index < run_container->n_runs; run_index++) {
969 roaring_value_t start = run_container->runs[run_index].value;
970 roaring_value_t length = run_container->runs[run_index].length;
971 suffix_t end = start + length;
972 for (suffix_val = start; suffix_val <= end; suffix_val++) {
973 if constexpr(FixedWidthIntegral<T>) {
974 value = shifted_prefix_and_key + suffix_val;
975 } else if constexpr(GMPIntegral<T>) {
976 mpz_add_ui(value.get_mpz_t(), shifted_prefix_and_key.get_mpz_t(), suffix_val);
977 }
978 local_signal = callback(value, my_tls);
979 if (local_signal == ForEachSignal::BREAK) {
980 atomic_signal.store(ForEachSignal::BREAK, std::memory_order_relaxed);
981 break;
982 }
983 }
984 }
985 break;
986 }
987 }
988 } // end omp for loop
989 } // end for prefix_index
990 } // end omp parallel region
991 } // end if serial/parallel
992 }
993
994};
