64 void add_prefix_key(const prefix_t& prefix) {
65 auto it = std::lower_bound(_sorted_prefixes.begin(), _sorted_prefixes.end(), prefix);
66 // Check for duplicates and only insert if there are none.
67 if (it == _sorted_prefixes.end() || *it != prefix) {

133 void add(const T& value) {
134 // Prefix can be mpz_class, so use TLS and out&.
135 static thread_local prefix_t prefix;
136 if constexpr(FixedWidthIntegral<T>) {
137 prefix = Traits::get_prefix(value);
138 } else if constexpr(GMPIntegral<T>) {
139 Traits::get_prefix(value, prefix);
140 }
141 // Suffix is just a uint32_t.
142 suffix_t suffix = Traits::get_suffix(value);
143
144 auto [it, inserted] = _flat_map.try_emplace(prefix);
145 if (inserted) {
146 add_prefix_key(prefix);
147 }
148 it->second.add(suffix);

157 void add_range(const T& start, const T& end) {
158 // Prefix can be mpz_class, so use TLS and out&.
159 static thread_local prefix_t start_prefix;
160 static thread_local prefix_t end_prefix;
161 if constexpr(FixedWidthIntegral<T>) {
162 start_prefix = Traits::get_prefix(start);
163 end_prefix = Traits::get_prefix(end);
164 } else if constexpr(GMPIntegral<T>) {
165 Traits::get_prefix(start, start_prefix);
166 Traits::get_prefix(end, end_prefix);
167 }
168 // Suffix is just a uint32_t.
169 suffix_t start_suffix = Traits::get_suffix(start);
170 suffix_t end_suffix = Traits::get_suffix(end);
171
172 if (start_prefix == end_prefix) {
173 // They all fall into the same prefix. Call a single addRange() and then leave.
174 auto [it, inserted] = _flat_map.try_emplace(start_prefix);
175 if (inserted) {
176 add_prefix_key(start_prefix);
177 }
178 it->second.addRange(start_suffix, end_suffix);
179 } else {
180 // Insert the remainder of prefix_1 which clearly goes to SUFFIX_MAX since prefix_2 is beyond it.
181 // Since the suffix is full, we use addRangeClosed() to encapsulate it.
182 auto [first_it, first_inserted] = _flat_map.try_emplace(start_prefix);
183 if (first_inserted) {
184 add_prefix_key(start_prefix);
185 }
186 first_it->second.addRangeClosed(start_suffix, Traits::SUFFIX_MAX);
187
188 // Loop through any intermediary prefixes, which are incremental of course between prefix_start and prefix_end, non-inclusive.
189 // Again, since the suffix is guaranteed full, we use addRangeClosed() to encapsulate it.
190 for (prefix_t middle_prefix = start_prefix + 1; middle_prefix < end_prefix; middle_prefix++) {
191 auto [mid_it, mid_inserted] = _flat_map.try_emplace(middle_prefix);
192 if (mid_inserted) {
193 add_prefix_key(middle_prefix);
194 }
195 mid_it->second.addRangeClosed(0, Traits::SUFFIX_MAX);
196 }
197
198 // Add the remaining bits to the last prefix, ending at "end" of course.
199 // Reminder, we do NOT include the last value (end_suffix) because this is the half-open variant.
200 auto [last_it, last_inserted] = _flat_map.try_emplace(end_prefix);
201 if (last_inserted) {
202 add_prefix_key(end_prefix);
203 }
204 last_it->second.addRange(0, end_suffix);
205 }
206 }

249 void remove(const T& value) {
250 // Prefix can be mpz_class, so use TLS and out&.
251 static thread_local prefix_t prefix;
252 if constexpr(FixedWidthIntegral<T>) {
253 prefix = Traits::get_prefix(value);

296 void merge(const FlatHashBitmapImpl<T>& src) {
297 for (const auto& [src_prefix, src_bitmap] : src.get_map()) {
298 // Get an iterator to our keypair.
299 auto [it, inserted] = _flat_map.try_emplace(src_prefix);
300 // Add the prefix if it was missing.
301 if (inserted) {
302 add_prefix_key(src_prefix);
303 }
304 // Merge using bitwise |=.
305 it->second |= src_bitmap;
306 }

401 static bool st_equal(const FlatHashBitmapImpl<T>& first, const FlatHashBitmapImpl<T>& second, std::string* err = nullptr) {
402 EqualityHelper eq(err);
403 eq.set_category("FlatHashBitmapImpl");
404
405 // Grab the flat hash maps and prefixes.
406 const map_t& f_flat_map = first.get_map();
407 const map_t& s_flat_map = second.get_map();
408 const std::vector<prefix_t>& f_prefixes = first.get_sorted_prefixes();
409 const std::vector<prefix_t>& s_prefixes = second.get_sorted_prefixes();
410 // Check size.
411 if (! eq.equal(f_prefixes.size(), s_prefixes.size())) {
412 return eq.fail("Flat hash impl sorted_prefxies size mistmatch");
413 }
414 // Prefixes should match and are already sorted. Test each prefix, and then the CRoaring bitmap beneath it.
415 for (size_t i = 0; i < f_prefixes.size(); i++) {
416 if (! eq.equal(f_prefixes.at(i), s_prefixes.at(i))) {
417 return eq.fail("Flat hash prefix at ID " + to_string_any(i) + " mismatch");
418 }
419 const roaring::Roaring& f_roaring = f_flat_map.at(f_prefixes.at(i));
420 const roaring::Roaring& s_roaring = s_flat_map.at(s_prefixes.at(i));
421 if (! roaring::api::roaring_bitmap_equals(&(f_roaring.roaring), &(s_roaring.roaring))) {
422 return eq.fail("Bitmaps do not match for prefix " + to_string_any(f_prefixes.at(i)));
423 }
424 }
425
426 // Guess they're equal.
427 return true;
428 }

460 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
461 StreamHelper sh(nullptr, &out, err);
462 sh.set_category("FlatHashBitmapImpl");
463
464 // Write out the number of prefixes/bitmaps.
465 uint64_t u64_prefix_count = static_cast<uint64_t>(_sorted_prefixes.size());
466 if (! sh.serialize_integral(u64_prefix_count)) {
467 return sh.fail("prefix_count==" + to_string_any(u64_prefix_count));
468 }
469
470 // Now loop through each key-value pair. Make sure to do them in order so exported files remain deterministic.
471 for (const prefix_t& prefix : _sorted_prefixes) {
472 // Write the prefix.
473 if (! sh.serialize_integral(prefix)) {
474 return sh.fail("prefix==" + to_string_any(prefix));
475 }
476 // Find the bitmap.
477 auto it = _flat_map.find(prefix);
478 if (it == _flat_map.end()) {
479 return sh.fail("couldn't find bitmap for prefix==" + to_string_any(prefix));
480 }
481 const roaring::Roaring& bitmap = it->second;
482 // Write how much space the bitmap will take.
483 uint64_t u64_bitmap_size = static_cast<uint64_t>(bitmap.getSizeInBytes(true));
484 if (! sh.serialize_integral(u64_bitmap_size)) {
485 return sh.fail("bitmap_size==" + to_string_any(u64_bitmap_size));
486 }
487 // Build the buffer and flush it.
488 std::vector<char> buffer(u64_bitmap_size);
489 size_t actual_bytes = bitmap.write(buffer.data(), true);
490 if (actual_bytes != u64_bitmap_size) {
491 return sh.fail("Number of bytes to write expected by getSizeInBytes doesn't match actual bytes returned");
492 }
493 if (! sh.write_bytes(buffer.data(), u64_bitmap_size)) {
494 return sh.fail("roaring bitmap data");
495 }
496 }
497
498 // All good.
499 return true;
500 }

514 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
515 StreamHelper sh(&in, nullptr, err);
516 sh.set_category("FlatHashBitmapImpl");
517
518 // Clear it out (reset).
519 clear();
520
521 // Prefixes
522 uint64_t u64_prefix_count = 0;
523 if (! sh.deserialize_integral(u64_prefix_count)) {
524 return sh.fail("couldn't read prefix count");
525 }
526 for (uint64_t i = 0; i < u64_prefix_count; i++) {
527 prefix_t prefix;
528 if (! sh.deserialize_integral(prefix)) {
529 return sh.fail("couldn't read prefix value for index i==" + to_string_any(i));
530 }
531 uint64_t u64_bitmap_size = 0;
532 if (! sh.deserialize_integral(u64_bitmap_size)) {
533 return sh.fail("couldn't read bitmap size for prefix==" + to_string_any(prefix));
534 }
535 std::vector<char> buffer(u64_bitmap_size);
536 if (! sh.read_bytes(buffer.data(), static_cast<size_t>(u64_bitmap_size))) {
537 return sh.fail("couldn't read bitmap data for prefix==" + to_string_any(prefix));
538 }
539 // Try to place the prefix, which will build out bitmap too.
540 auto [it, inserted] = _flat_map.try_emplace(prefix);
541 if (inserted) {
542 add_prefix_key(prefix);
543 }
544 it->second = roaring::Roaring::read(buffer.data(), true);
545 }
546
547 // All good
548 return true;
549 }

604 template<typename Func, typename TLS_Type>
605 void for_each_transformer(BitmapTransformerPolicy policy, std::vector<TLS_Type>& tls, Func&& callback) {
606 // Do not allow non-ref callbacks. Otherwise we make GMP over and over.
607 static_assert(std::is_invocable_r_v<bool, Func, const T&, TLS_Type&>, "Callback must be callable as void(const T&, TLS_Type&)");
608
609 // Ensure the TLS has enough elements before proceeding.
610 const size_t max_threads = policy == BitmapTransformerPolicy::SERIAL ? 1 : static_cast<size_t>(omp_get_max_threads());
611 if (tls.size() < max_threads) {
612 tls.resize(max_threads);
613 }
614
615 // Setup a control-stop.
616 std::atomic<bool> atomic_stop = false;
617 bool stop;
618
619 // Process according to policy.
620 if (policy == BitmapTransformerPolicy::SERIAL) {
621 // Serial Path
622 // Guarantees sequential processing. Can efficiently use exposed iterators.
623
624 // Hoist these, just in case of temp allocs.
625 T value;
626 T shifted_prefix;
627 suffix_t suffix;
628
629 // Even though this path is not threaded, get a thread ID for clarity.
630 size_t my_thread_id = static_cast<size_t>(omp_get_thread_num());
631
632 // The TLS should only need index 0.
633 TLS_Type& my_tls = tls[my_thread_id];
634
635 // Loop through prefixes in order.
636 for (const prefix_t& prefix : _sorted_prefixes) {
637 // Shift the prefix once instead of deeper inside the while() loop. For smaller types, set to 0.
638 if constexpr(sizeof(T) > Traits::SUFFIX_BYTES) {
639 // T can hold the prefix. Let it shift.
640 if constexpr(FixedWidthIntegral<T>) {
641 shifted_prefix = static_cast<T>(prefix) << Traits::SUFFIX_BITS;
642 } else if constexpr(GMPIntegral<T>) {
643 mpz_mul_2exp(shifted_prefix.get_mpz_t(), prefix.get_mpz_t(), Traits::SUFFIX_BITS);
644 }
645 } else {
646 // T is smaller than suffix_t, which means the only prefix it'll ever have is 0.
647 // Set it manually, because small-T means it can't shift SUFFIX_BITS wide.
648 shifted_prefix = 0;
649 }
650
651 // Find the bitmap and connect an iterator.
652 roaring::Roaring& roaring_obj = _flat_map.find(prefix)->second;
653 roaring::api::roaring_uint32_iterator_t bitmap_iterator;
654 roaring::api::roaring_iterator_init(&roaring_obj.roaring, &bitmap_iterator);
655
656 // Loop this iterator until done. Use addition instead of bitwise-OR (|) to avoid alloc on GMP path.
657 while (bitmap_iterator.has_value) {
658 suffix = bitmap_iterator.current_value;
659 if constexpr(FixedWidthIntegral<T>) {
660 value = shifted_prefix + suffix;
661 } else if constexpr(GMPIntegral<T>) {
662 mpz_add_ui(value.get_mpz_t(), shifted_prefix.get_mpz_t(), suffix);
663 }
664 stop = callback(value, my_tls);
665 if (stop) { break; }
666 roaring::api::roaring_uint32_iterator_advance(&bitmap_iterator);
667 }
668
669 // If stopping, break from the prefix for() loop too.
670 if (stop) { break; }
671 }
672 } else {
673 // Parallel Path
674 // Work is threaded, breaking sequential guarantee. Uses CRoaring internals for performance.
675 //
676 // While throwing _sorted_prefixes at OMP might seem ideal, it would result in only a single thread working until level
677 // 32. This is because CRoaring uses a 32-bit map (16 bit key, 16 bit suffix) internally.
678 //
679 // As such, this will read the internal high-low containers and iterate based on their type (array, bitmap, RLE).
680
681 // Open the parallel region early and establish thread-locals.
682 #pragma omp parallel default(none) shared(tls, atomic_stop, callback)
683 {
684 // Get a thread ID.
685 size_t my_thread_id = static_cast<size_t>(omp_get_thread_num());
686
687 // Pick an element from the TLS.
688 TLS_Type& my_tls = tls[my_thread_id];
689
690 // Track a local-stop.
691 bool local_stop = false;
692
693 // Hoist a few locals to avoid lots of potential alloc on GMP path.
694 T value;
695 T shifted_prefix;
696 T shifted_prefix_and_key;
697
698 // Loop through prefixes.
699 for (size_t prefix_index = 0; prefix_index < _sorted_prefixes.size(); prefix_index++) {
700 // Stop if needed.
701 if (atomic_stop.load(std::memory_order_relaxed)) { continue; }
702
703 // Grab the prefix and then build a shifted_prefix so it isn't recomputed.
704 const prefix_t& prefix = _sorted_prefixes[prefix_index];
705 // Shift the prefix once instead of deeper inside the while()/for() loops. For smaller types, set to 0.
706 if constexpr(sizeof(T) > Traits::SUFFIX_BYTES) {
707 // T can hold the prefix. Let it shift.
708 if constexpr(FixedWidthIntegral<T>) {
709 shifted_prefix = static_cast<T>(prefix) << Traits::SUFFIX_BITS;
710 } else if constexpr(GMPIntegral<T>) {
711 mpz_mul_2exp(shifted_prefix.get_mpz_t(), prefix.get_mpz_t(), Traits::SUFFIX_BITS);
712 }
713 } else {
714 // T is smaller than suffix_t, which means the only prefix it'll ever have is 0.
715 // Set it manually, because small-T means it can't shift SUFFIX_BITS wide.
716 shifted_prefix = 0;
717 }
718
719 // Find our bitmap and connect to the internal storage components.
720 auto it = _flat_map.find(prefix);
721 const roaring::api::roaring_bitmap_t* roaring_bitmap = &(it->second.roaring);
722 const roaring::api::roaring_array_t* high_low_container = &(roaring_bitmap->high_low_container);
723 // Hoist container bits. Probably optimized away, but meh.
724 const auto* high_low_container__keys = high_low_container->keys;
725 const auto* high_low_container__containers = high_low_container->containers;
726 const auto* high_low_container__typecodes = high_low_container->typecodes;
727
728 // Build the inner loop around the keys and sub containers. Split it up with OMP.
729 #pragma omp for schedule(dynamic)
730 for (int high_low_index = 0; high_low_index < high_low_container->size; high_low_index++) {
731 // Check for stop again at the chunk level.
732 if (atomic_stop.load(std::memory_order_relaxed)) { continue; }
733
734 // Grab the roaring key, the container, and type code.
735 roaring_key_t suffix_key = high_low_container__keys[high_low_index];
736 const roaring::api::container_s* container = high_low_container__containers[high_low_index];
737 roaring_typecode_t typecode = high_low_container__typecodes[high_low_index];
738
739 // Hoist and promote+shift the key into full suffix_t space.
740 suffix_t shifted_key = (suffix_t(suffix_key)) << Traits::ROARING_KEY_BITS;
741 suffix_t suffix_val;
742
743 // Build the shifted prefix and key combo so it isn't repeated below.
744 if constexpr(FixedWidthIntegral<T>) {
745 shifted_prefix_and_key = shifted_prefix + shifted_key;
746 } else if constexpr(GMPIntegral<T>) {
747 mpz_add_ui(shifted_prefix_and_key.get_mpz_t(), shifted_prefix.get_mpz_t(), shifted_key);
748 }
749
750 // Select the correct iteration method based on the typecode to retrieve and process suffixes.
751 // Use addition instead of bitwise-OR (|) to avoid alloc on GMP path.
752 switch (typecode) {
753 case ARRAY_CONTAINER_TYPE: {
754 const roaring::internal::array_container_t* array_container = (roaring::internal::array_container_t*)container;
755 for (int array_index = 0; array_index < array_container->cardinality; array_index++) {
756 suffix_val = array_container->array[array_index];
757 if constexpr(FixedWidthIntegral<T>) {
758 value = shifted_prefix_and_key + suffix_val;
759 } else if constexpr(GMPIntegral<T>) {
760 mpz_add_ui(value.get_mpz_t(), shifted_prefix_and_key.get_mpz_t(), suffix_val);
761 }
762 local_stop = callback(value, my_tls);
763 if (local_stop) {
764 atomic_stop.store(true, std::memory_order_relaxed);
765 break;
766 }
767 }
768 break;
769 }
770
771 case BITSET_CONTAINER_TYPE: {
772 const roaring::internal::bitset_container_t* bitset_container = (const roaring::internal::bitset_container_t*)container;
773 for (int word_index = 0; word_index < roaring::internal::BITSET_CONTAINER_SIZE_IN_WORDS; word_index++) {
774 roaring_word_t word = bitset_container->words[word_index];
775 // Loop until there are no more on-bits in the word.
776 while (word != 0) {
777 // Calculate the offset by counting the number of trailing zeros, giving us the next "1" position.
778 int offset = __builtin_ctzll(word);
779 // Multiply out the word index by the number of bits per word, then add the offset. That's the next "on" node.
780 suffix_val = (word_index * Traits::ROARING_WORD_BITS) + offset;
781 if constexpr(FixedWidthIntegral<T>) {
782 value = shifted_prefix_and_key + suffix_val;
783 } else if constexpr(GMPIntegral<T>) {
784 mpz_add_ui(value.get_mpz_t(), shifted_prefix_and_key.get_mpz_t(), suffix_val);
785 }
786 local_stop = callback(value, my_tls);
787 if (local_stop) {
788 atomic_stop.store(true, std::memory_order_relaxed);
789 break;
790 }
791 // Now bitwise-AND the word minus one to wipe out the least-significant 1s place which was just processed.
792 word &= (word - 1);
