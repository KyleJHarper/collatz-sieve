64 auto it = std::lower_bound(_sorted_prefixes.begin(), _sorted_prefixes.end(), prefix);

120 static thread_local prefix_t prefix;
121 if constexpr(BuiltinIntegral<T>) {
122 prefix = Traits::get_prefix(value);
123 } else {
124 Traits::get_prefix(value, prefix);
125 }
126 // Suffix is just a uint32_t.

128
129 auto [it, inserted] = _flat_map.try_emplace(prefix);
130 if (inserted) {
131 add_prefix_key(prefix);
132 }
133 it->second.add(suffix);
134 }

142 void add_range(const T& start, const T& end) {
143 // Prefix can be mpz_class, so use TLS and out&.
144 static thread_local prefix_t start_prefix;
145 static thread_local prefix_t end_prefix;
146 if constexpr(BuiltinIntegral<T>) {
147 start_prefix = Traits::get_prefix(start);
148 end_prefix = Traits::get_prefix(end);
149 } else {
150 Traits::get_prefix(start, start_prefix);
151 Traits::get_prefix(end, end_prefix);
152 }
153 // Suffix is just a uint32_t.
154 suffix_t start_suffix = Traits::get_suffix(start);
155 suffix_t end_suffix = Traits::get_suffix(end);
156
157 if (start_prefix == end_prefix) {
158 // They all fall into the same prefix. Call a single addRange() and then leave.
159 auto [it, inserted] = _flat_map.try_emplace(start_prefix);
160 if (inserted) {
161 add_prefix_key(start_prefix);
162 }
163 it->second.addRange(start_suffix, end_suffix);
164 } else {
165 // Insert the remainder of prefix_1 which clearly goes to SUFFIX_MAX since prefix_2 is beyond it.
166 // Since the suffix is full, we use addRangeClosed() to encapsulate it.
167 auto [first_it, first_inserted] = _flat_map.try_emplace(start_prefix);
168 if (first_inserted) {
169 add_prefix_key(start_prefix);
170 }
171 first_it->second.addRangeClosed(start_suffix, Traits::SUFFIX_MAX);
172
173 // Loop through any intermediary prefixes, which are incremental of course between prefix_start and prefix_end, non-inclusive.
174 // Again, since the suffix is guaranteed full, we use addRangeClosed() to encapsulate it.
175 for (prefix_t middle_prefix = start_prefix + 1; middle_prefix < end_prefix; middle_prefix++) {
176 auto [mid_it, mid_inserted] = _flat_map.try_emplace(middle_prefix);
177 if (mid_inserted) {
178 add_prefix_key(middle_prefix);
179 }
180 mid_it->second.addRangeClosed(0, Traits::SUFFIX_MAX);
181 }
182
183 // Add the remaining bits to the last prefix, ending at "end" of course.
184 // Reminder, we do NOT include the last value (end_suffix) because this is the half-open variant.
185 auto [last_it, last_inserted] = _flat_map.try_emplace(end_prefix);
186 if (last_inserted) {
187 add_prefix_key(end_prefix);
188 }
189 last_it->second.addRange(0, end_suffix);
190 }
191 }

281 void merge(const FlatHashBitmapImpl<T>& src) {
282 for (const auto& [src_prefix, src_bitmap] : src.get_map()) {
283 // Get an iterator to our keypair.
284 auto [it, inserted] = _flat_map.try_emplace(src_prefix);
285 // Add the prefix if it was missing.
286 if (inserted) {
287 add_prefix_key(src_prefix);
288 }
289 // Merge using bitwise |=.
290 it->second |= src_bitmap;
291 }

310 clear();
311 for (const auto& [src_prefix, src_bitmap] : src.get_map()) {
312 // Add to the prefix key list.

314 // Build the key on our map.
315 auto [it, last_inserted] = _flat_map.try_emplace(src_prefix);
316 // Clone the src to our new bitmap using the overwrite method.
317 bool successful = roaring::api::roaring_bitmap_overwrite(it->second, src_bitmap);
318 if (!successful) {
319 throw std::runtime_error("Unable to clone a bitmap using roaring_bitmap_overwite when calling clone().");

389 static bool st_equal(const FlatHashBitmapImpl<T>& first, const FlatHashBitmapImpl<T>& second, std::string* err = nullptr) {
390 EqualityHelper eq(err);
391 eq.set_category("FlatHashBitmapImpl");
392
393 // Grab the flat hash maps and prefixes.
394 const map_t& f_flat_map = first.get_map();
395 const map_t& s_flat_map = second.get_map();
396 const std::vector<prefix_t>& f_prefixes = first.get_sorted_prefixes();
397 const std::vector<prefix_t>& s_prefixes = second.get_sorted_prefixes();
398 // Check size.
399 if (eq.unequal(f_prefixes.size(), s_prefixes.size())) {
400 return eq.fail("Flat hash impl sorted_prefxies size mistmatch");
401 }
402 // Prefixes should match and are already sorted. Test each prefix, and then the CRoaring bitmap beneath it.
403 for (size_t i = 0; i < f_prefixes.size(); i++) {
404 if (eq.unequal(f_prefixes.at(i), s_prefixes.at(i))) {
405 return eq.fail("Flat hash prefix at ID " + to_string_any(i) + " mismatch");
406 }
407 const roaring::Roaring& f_roaring = f_flat_map.at(f_prefixes.at(i));
408 const roaring::Roaring& s_roaring = s_flat_map.at(s_prefixes.at(i));
409 if (! roaring::api::roaring_bitmap_equals(&(f_roaring.roaring), &(s_roaring.roaring))) {
410 return eq.fail("Bitmaps do not match for prefix " + to_string_any(f_prefixes.at(i)));
411 }
412 }
413
414 // Guess they're equal.
415 return true;
416 }

448 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
449 StreamHelper sh(nullptr, &out, err);
450 sh.set_category("FlatHashBitmapImpl");
451
452 // Write out the number of prefixes/bitmaps.
453 uint64_t u64_prefix_count = static_cast<uint64_t>(_sorted_prefixes.size());
454 if (! sh.serialize_integral(u64_prefix_count)) {
455 return sh.fail("prefix_count==" + to_string_any(u64_prefix_count));
456 }
457
458 // Now loop through each key-value pair. Make sure to do them in order so exported files remain deterministic.
459 for (const prefix_t& prefix : _sorted_prefixes) {
460 // Write the prefix.
461 if (! sh.serialize_integral(prefix)) {
462 return sh.fail("prefix==" + to_string_any(prefix));
463 }
464 // Find the bitmap.
465 auto it = _flat_map.find(prefix);
466 if (it == _flat_map.end()) {
467 return sh.fail("couldn't find bitmap for prefix==" + to_string_any(prefix));
468 }
469 const roaring::Roaring& bitmap = it->second;
470 // Write how much space the bitmap will take.
471 uint64_t u64_bitmap_size = static_cast<uint64_t>(bitmap.getSizeInBytes(true));
472 if (! sh.serialize_integral(u64_bitmap_size)) {
473 return sh.fail("bitmap_size==" + to_string_any(u64_bitmap_size));
474 }
475 // Build the buffer and flush it.
476 std::vector<char> buffer(u64_bitmap_size);
477 size_t actual_bytes = bitmap.write(buffer.data(), true);
478 if (actual_bytes != u64_bitmap_size) {
479 return sh.fail("Number of bytes to write expected by getSizeInBytes doesn't match actual bytes returned");
480 }
481 if (! sh.write_bytes(buffer.data(), u64_bitmap_size)) {
482 return sh.fail("roaring bitmap data");
483 }
484 }
485
486 // All good.
487 return true;
488 }

502 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
503 StreamHelper sh(&in, nullptr, err);
504 sh.set_category("FlatHashBitmapImpl");
505
506 // Clear it out (reset).
507 clear();
508
509 // Prefixes
510 uint64_t u64_prefix_count = 0;
511 if (! sh.deserialize_integral(u64_prefix_count)) {
512 return sh.fail("couldn't read prefix count");
513 }
514 for (uint64_t i = 0; i < u64_prefix_count; i++) {
515 prefix_t prefix;
516 if (! sh.deserialize_integral(prefix)) {
517 return sh.fail("couldn't read prefix value for index i==" + to_string_any(i));
518 }
519 uint64_t u64_bitmap_size = 0;
520 if (! sh.deserialize_integral(u64_bitmap_size)) {
521 return sh.fail("couldn't read bitmap size for prefix==" + to_string_any(prefix));
522 }
523 std::vector<char> buffer(u64_bitmap_size);
524 if (! sh.read_bytes(buffer.data(), static_cast<size_t>(u64_bitmap_size))) {
525 return sh.fail("couldn't read bitmap data for prefix==" + to_string_any(prefix));
526 }
527 // Try to place the prefix, which will build out bitmap too.
528 auto [it, inserted] = _flat_map.try_emplace(prefix);
529 if (inserted) {
530 add_prefix_key(prefix);
531 }
532 it->second = roaring::Roaring::read(buffer.data(), true);
533 }
534
535 // All good
536 return true;

592 template<typename Func, typename TLS_Type>
593 void for_each_transformer(BitmapTransformerPolicy policy, std::vector<TLS_Type>& tls, Func&& callback) {
594 // Do not allow non-ref callbacks. Otherwise we make GMP over and over.
595 static_assert(std::is_invocable_r_v<bool, Func, const T&, TLS_Type&>, "Callback must be callable as void(const T&, TLS_Type&)");
596
597 // Ensure the TLS has enough elements before proceeding.
598 const size_t max_threads = policy == BitmapTransformerPolicy::SERIAL ? 1 : static_cast<size_t>(omp_get_max_threads());
599 if (tls.size() < max_threads) {
600 tls.resize(max_threads);
601 }
602
603 // Setup a control-stop.
604 std::atomic<bool> atomic_stop = false;
605 bool stop;
606
607 // Process according to policy.
608 if (policy == BitmapTransformerPolicy::SERIAL) {
609 // Serial Path
610 // Guarantees sequential processing. Can efficiently use exposed iterators.
611
612 // Hoist these, just in case of temp allocs.
613 T value;
614 T shifted_prefix;
615 suffix_t suffix;
616
617 // Even though this path is not threaded, get a thread ID for clarity.
618 size_t my_thread_id = static_cast<size_t>(omp_get_thread_num());
619
620 // The TLS should only need index 0.
621 TLS_Type& my_tls = tls[my_thread_id];
622
623 // Loop through prefixes in order.
624 for (const prefix_t& prefix : _sorted_prefixes) {
625 // Shift the prefix once instead of deeper inside the while() loop. For smaller types, set to 0.
626 if constexpr(sizeof(T) > Traits::SUFFIX_BYTES) {
627 // T can hold the prefix. Let it shift.
628 if constexpr(BuiltinIntegral<T>) {
629 shifted_prefix = static_cast<T>(prefix) << Traits::SUFFIX_BITS;
630 } else {
631 mpz_mul_2exp(shifted_prefix.get_mpz_t(), prefix.get_mpz_t(), Traits::SUFFIX_BITS);
632 }
633 } else {
634 // T is smaller than suffix_t, which means the only prefix it'll ever have is 0.
635 // Set it manually, because small-T means it can't shift SUFFIX_BITS wide.
636 shifted_prefix = 0;
637 }
638
639 // Find the bitmap and connect an iterator.
640 roaring::Roaring& roaring_obj = _flat_map.find(prefix)->second;
641 roaring::api::roaring_uint32_iterator_t bitmap_iterator;
642 roaring::api::roaring_iterator_init(&roaring_obj.roaring, &bitmap_iterator);
643
644 // Loop this iterator until done. Use addition instead of bitwise-OR (|) to avoid alloc on GMP path.
645 while (bitmap_iterator.has_value) {
646 suffix = bitmap_iterator.current_value;
647 if constexpr(BuiltinIntegral<T>) {
648 value = shifted_prefix + suffix;
649 } else {
650 mpz_add_ui(value.get_mpz_t(), shifted_prefix.get_mpz_t(), suffix);
651 }
652 stop = callback(value, my_tls);
653 if (stop) { break; }
654 roaring::api::roaring_uint32_iterator_advance(&bitmap_iterator);
655 }
656
657 // If stopping, break from the prefix for() loop too.
658 if (stop) { break; }
659 }
660 } else {
661 // Parallel Path
662 // Work is threaded, breaking sequential guarantee. Uses CRoaring internals for performance.
663 //
664 // While throwing _sorted_prefixes at OMP might seem ideal, it would result in only a single thread working until level
665 // 32. This is because CRoaring uses a 32-bit map (16 bit key, 16 bit suffix) internally.
666 //
667 // As such, this will read the internal high-low containers and iterate based on their type (array, bitmap, RLE).
668
669 // Open the parallel region early and establish thread-locals.
670 #pragma omp parallel default(none) shared(tls, atomic_stop, callback)
671 {
672 // Get a thread ID.
673 size_t my_thread_id = static_cast<size_t>(omp_get_thread_num());
674
675 // Pick an element from the TLS.
676 TLS_Type& my_tls = tls[my_thread_id];
677
678 // Track a local-stop.
679 bool local_stop = false;
680
681 // Hoist a few locals to avoid lots of potential alloc on GMP path.
682 T value;
683 T shifted_prefix;
684 T shifted_prefix_and_key;
685
686 // Loop through prefixes.
687 for (size_t prefix_index = 0; prefix_index < _sorted_prefixes.size(); prefix_index++) {
688 // Stop if needed.
689 if (atomic_stop.load(std::memory_order_relaxed)) { continue; }
690
691 // Grab the prefix and then build a shifted_prefix so it isn't recomputed.
692 const prefix_t& prefix = _sorted_prefixes[prefix_index];
693 // Shift the prefix once instead of deeper inside the while()/for() loops. For smaller types, set to 0.
694 if constexpr(sizeof(T) > Traits::SUFFIX_BYTES) {
695 // T can hold the prefix. Let it shift.
696 if constexpr(BuiltinIntegral<T>) {
697 shifted_prefix = static_cast<T>(prefix) << Traits::SUFFIX_BITS;
698 } else {
699 mpz_mul_2exp(shifted_prefix.get_mpz_t(), prefix.get_mpz_t(), Traits::SUFFIX_BITS);
700 }
701 } else {
702 // T is smaller than suffix_t, which means the only prefix it'll ever have is 0.
703 // Set it manually, because small-T means it can't shift SUFFIX_BITS wide.
704 shifted_prefix = 0;
705 }
706
707 // Find our bitmap and connect to the internal storage components.
708 auto it = _flat_map.find(prefix);
709 const roaring::api::roaring_bitmap_t* roaring_bitmap = &(it->second.roaring);
710 const roaring::api::roaring_array_t* high_low_container = &(roaring_bitmap->high_low_container);
711 // Hoist container bits. Probably optimized away, but meh.
712 const auto* high_low_container__keys = high_low_container->keys;
713 const auto* high_low_container__containers = high_low_container->containers;
714 const auto* high_low_container__typecodes = high_low_container->typecodes;
715
716 // Build the inner loop around the keys and sub containers. Split it up with OMP.
717 #pragma omp for schedule(dynamic)
718 for (int high_low_index = 0; high_low_index < high_low_container->size; high_low_index++) {
719 // Check for stop again at the chunk level.
720 if (atomic_stop.load(std::memory_order_relaxed)) { continue; }
721
722 // Grab the roaring key, the container, and type code.
723 roaring_key_t suffix_key = high_low_container__keys[high_low_index];
724 const roaring::api::container_s* container = high_low_container__containers[high_low_index];
725 roaring_typecode_t typecode = high_low_container__typecodes[high_low_index];
726
727 // Hoist and promote+shift the key into full suffix_t space.
728 suffix_t shifted_key = (suffix_t(suffix_key)) << Traits::ROARING_KEY_BITS;
729 suffix_t suffix_val;
730
731 // Build the shifted prefix and key combo so it isn't repeated below.
732 if constexpr(BuiltinIntegral<T>) {
733 shifted_prefix_and_key = shifted_prefix + shifted_key;
734 } else {
735 mpz_add_ui(shifted_prefix_and_key.get_mpz_t(), shifted_prefix.get_mpz_t(), shifted_key);
736 }
737
738 // Select the correct iteration method based on the typecode to retrieve and process suffixes.
739 // Use addition instead of bitwise-OR (|) to avoid alloc on GMP path.
740 switch (typecode) {
741 case ARRAY_CONTAINER_TYPE: {
742 const roaring::internal::array_container_t* array_container = (roaring::internal::array_container_t*)container;
743 for (int array_index = 0; array_index < array_container->cardinality; array_index++) {
744 suffix_val = array_container->array[array_index];
745 if constexpr(BuiltinIntegral<T>) {
746 value = shifted_prefix_and_key + suffix_val;
747 } else {
748 mpz_add_ui(value.get_mpz_t(), shifted_prefix_and_key.get_mpz_t(), suffix_val);
749 }
750 local_stop = callback(value, my_tls);
751 if (local_stop) {
752 atomic_stop.store(true, std::memory_order_relaxed);
753 break;
754 }
755 }
756 break;
757 }
758
759 case BITSET_CONTAINER_TYPE: {
760 const roaring::internal::bitset_container_t* bitset_container = (const roaring::internal::bitset_container_t*)container;
761 for (int word_index = 0; word_index < roaring::internal::BITSET_CONTAINER_SIZE_IN_WORDS; word_index++) {
762 roaring_word_t word = bitset_container->words[word_index];
763 // Loop until there are no more on-bits in the word.
764 while (word != 0) {
765 // Calculate the offset by counting the number of trailing zeros, giving us the next "1" position.
766 int offset = __builtin_ctzll(word);
767 // Multiply out the word index by the number of bits per word, then add the offset. That's the next "on" node.
768 suffix_val = (word_index * Traits::ROARING_WORD_BITS) + offset;
769 if constexpr(BuiltinIntegral<T>) {
770 value = shifted_prefix_and_key + suffix_val;
771 } else {
772 mpz_add_ui(value.get_mpz_t(), shifted_prefix_and_key.get_mpz_t(), suffix_val);
773 }
774 local_stop = callback(value, my_tls);
775 if (local_stop) {
776 atomic_stop.store(true, std::memory_order_relaxed);
777 break;
778 }
779 // Now bitwise-AND the word minus one to wipe out the least-significant 1s place which was just processed.
780 word &= (word - 1);
781 }
782 }
783 break;
784 }
785
786 case RUN_CONTAINER_TYPE: {
787 const roaring::internal::run_container_t* run_container = (const roaring::internal::run_container_t*)container;
788 for (int run_index = 0; run_index < run_container->n_runs; run_index++) {
789 roaring_value_t start = run_container->runs[run_index].value;
790 roaring_value_t length = run_container->runs[run_index].length;
