163 void run_executor() {
164 // Sanity check the scaling runs is 1+.
165 if (_scales_per_run < 1) {
166 throw std::invalid_argument("Cannot send a scales-per-run of zero. Must be 1+.");
167 }
168
169 // Find out how much memory is available for the GPU. Start by assuming the caller sent a fixed byte amount.
170 _gpu_buffer_used = _gpu_buffer_limit;
171
172 // If unsent, aim for 90% of the free bytes or total needed (if lesser).
173 if (_gpu_buffer_limit == 0) {
174 size_t free_bytes = 0;
175 size_t total_bytes = 0;
176 cudaError_t status = cudaMemGetInfo(&free_bytes, &total_bytes);
177 if (status != cudaSuccess) {
178 throw std::runtime_error(std::string("CUDA Error: ") + cudaGetErrorString(status));
179 }
180 _gpu_buffer_used = free_bytes * 9 / 10; // 90%
181 }
182 // If the survivor set is smaller than the available buffer space on the GPU, reduce the allocation size needed.
183 const size_t uncovered_values_size = this->_tree.get_uncovered_values().cardinality() * sizeof(T);
184 if (uncovered_values_size < _gpu_buffer_used) {
185 _gpu_buffer_used = uncovered_values_size;
186 }
187
188 // Use the total buffer limit to decide how large each thread's buffer can be.
189 // Add 1-2 elements worth of buffer in case the full survivor set fits in VRAM. Otherwise, integer truncation might cause
190 // edge-cases where a handful are excluded and require a whole round-trip. It's unlikely, but why risk it?
191 const size_t thread_count = static_cast<size_t>(omp_get_max_threads());
192 const size_t buffer_limit_per_thread = (_gpu_buffer_used / thread_count) + (sizeof(T) * 2);
193
194 // Use size_of to determine how many elements that works out to be. It'll be used by the buffer's constructor.
195 const uint64_t elements_per_thread = buffer_limit_per_thread / sizeof(T);
196
197 // Build the TLS objects so element count can be sent. Otherwise NodeBitmap will default-initialiaze and fail.
198 _tls.resize(thread_count);
199 for (ThreadStorage& tls : _tls) {
200 tls.buffer_ptr = std::make_unique<CudaStreamBuffer<T, GPUVerifierResultData>>(elements_per_thread);
201 }
202
203 // Link to to the uncovered values (root survivors).
204 const NodeBitmap<T>& uncovered_values = this->_tree.get_uncovered_values();
205
206 // Update the multiplier to use with the scaling factor, which grows by scales_per_run with each loop.
207 _base_multiplier = 1;
208
209 // Bump the multiplier if a start value requires it. Take the maximum value of the tree and subtract it from the start
210 // value requested, then divide it by the scaling factor to get the mutliplier.
211 if (this->_start_value > uncovered_values.maximum()) {
212 _base_multiplier += ((this->_start_value - uncovered_values.maximum()) / _scaling_factor);
213 }
214
215 // Create a max multiplier which is the highest (inclusive) multiplier based on the scales_per_run.
216 _max_multiplier = _base_multiplier + _scales_per_run - 1;
217
218 // Hoist a total scaling factor based on the maximum multiplier the GPU will receive.
219 T max_scaling_factor = _scaling_factor * _max_multiplier;
220
221 // Loop through all survivors forever, or until end_value is reached.
222 // It's okay if this last_loop is non-atomic. It's only ever checked outside OMP.
223 bool last_loop = false;
224 while(! last_loop) {
225 uncovered_values.for_each_value_with_tls(ForEachPolicy::PARALLEL, _tls, [&](T& root_survivor_value, ThreadStorage& thread_storage) {
226 // Stop if done. Since a multiplier range is in effect, it breaks the contract of always working within one prefix
227 // of a NodeBitmap (2^32 range). Therefore, don't return BREAK, just flag the atomic for last loop.
228 if (this->_end_value > 0 && (max_scaling_factor + root_survivor_value) > this->_end_value) {
229 last_loop = true;
230 }
231
232 // If the buffer is full, send it to the GPU for processing.
233 if (thread_storage.buffer_ptr->is_full()) {
234 // Call helper to process on the GPU.
235 process_buffer(thread_storage);
236
237 // This is a natual point to check for pausing, stopping, etc.
238 while (this->_state.load(std::memory_order_relaxed) == VerifierState::PAUSED) {
239 std::this_thread::sleep_for(std::chrono::milliseconds(this->_paused_check_ms));
240 }
241 if (this->_state.load(std::memory_order_relaxed) == VerifierState::STOPPING) {
242 last_loop = true;
243 }
244 thread_storage.synchronizations_performed++;
245 }
246
247 // Buffer has space. Add the current value to the next index, incrementing it too.
248 thread_storage.buffer_ptr->get_host_data_ptr()[thread_storage.buffer_ptr->use_next_index()] = root_survivor_value;
249
250 // Continue
251 return ForEachSignal::CONTINUE;
252 });
253
254 // Flush any pending buffers.
255 for (ThreadStorage& thread_storage : _tls) {
256 if (thread_storage.buffer_ptr->get_next_index() > 0) {
257 process_buffer(thread_storage);
258 }
259 }
260
261 // Bump the multipliers.
262 _base_multiplier += _scales_per_run;
263 _max_multiplier += _scales_per_run;
264
265 // Recompute the max scaling factor.
266 max_scaling_factor = _scaling_factor * _max_multiplier;
267 }
268
269 // Set state to STOPPED. Only this thread may do that.
270 this->_state.store(VerifierState::STOPPED);
271
272 // Synchronize the timer since it's done doing any work.
273 this->sync_timer();
274
275 // Reset the GPU used value.
276 _gpu_buffer_used = 0;
277 }

282 void process_buffer(ThreadStorage& thread_storage) {
283 // Mark the buffer READY.
284 thread_storage.buffer_ptr->set_state(CudaStreamBufferState::READY);
285
286 // Launch the kernel for the GPU to execute when it's able to.
287 if (this->_enable_max_iv_table) {
288 thread_storage.buffer_ptr->launch_kernel(launch_gpu_verify_kernel<T, true>, _base_multiplier, _max_multiplier, _scaling_factor);
289 } else {
290 thread_storage.buffer_ptr->launch_kernel(launch_gpu_verify_kernel<T, false>, _base_multiplier, _max_multiplier, _scaling_factor);
291 }
292
293 // Wait for the GPU to mark the buffer PROCESSED.
294 thread_storage.buffer_ptr->wait_until_state(CudaStreamBufferState::PROCESSED);
295
296 // Process the results. Check for overflows and handle them.
297 GPUVerifierResultData* host_results_ptr = thread_storage.buffer_ptr->get_host_results_ptr();
298 T* host_data_ptr = thread_storage.buffer_ptr->get_host_data_ptr();
299 size_t scales_ran = _max_multiplier - _base_multiplier + 1;
300 if (host_results_ptr->overflow_count > 0) {
301 // Had at least one overflow. Check if we contained them.
302 if (host_results_ptr->overflow_exceeded) {
303 // Ran out of overflow slots. Must reprocess the whole batch on CPU.
304 for (uint64_t i = 0; i < thread_storage.buffer_ptr->get_next_index(); i++) {
305 T base_value = *(host_data_ptr + i);
306 for (size_t multiplier = _base_multiplier; multiplier <= _max_multiplier; multiplier++) {
307 T initial_value = base_value + (multiplier * _scaling_factor);
308 if (! Collatz<T>::st_verify(initial_value, initial_value)) {
309 throw std::logic_error("Failed to verify value: " + to_string_any(initial_value));
310 }
311 }
312 }
313 // Update the metrics for this, including the overflow-exceeded counter.
314 this->_published_metrics.gpu_overflow_buffer_exceeded_atomic.fetch_add(1);
315 this->_published_metrics.gpu_overflows_processed_atomic.fetch_add(((thread_storage.buffer_ptr->get_next_index() + 1) * scales_ran));
316 } else {
317 // Contained the overflow. Process only the offenders on CPU.
318 for (uint64_t i = 0; i < host_results_ptr->overflow_count; i++) {
319 T base_value = *(host_data_ptr + host_results_ptr->overflow_indexes[i]);
320 for (size_t multiplier = _base_multiplier; multiplier <= _max_multiplier; multiplier++) {
321 T initial_value = base_value + (multiplier * _scaling_factor);
322 if (!Collatz<T>::st_verify(initial_value, initial_value)) {
323 throw std::logic_error("Failed to verify value: " + to_string_any(initial_value));
324 }
325 }
326 }
327 // Update the metrics for this.
328 this->_published_metrics.gpu_overflows_processed_atomic.fetch_add(host_results_ptr->overflow_count * scales_ran);
329 }
330 }
331
332 // Reset the results. They'll be sync'd as-such to the device.
333 host_results_ptr->reset();
334
335 // Update metrics.
336 this->_published_metrics.nodes_verified_atomic.fetch_add(thread_storage.buffer_ptr->get_next_index() * scales_ran);
337 this->_published_metrics.gpu_kernel_launches_atomic.fetch_add(1);
338
339 // Recycle the buffer. This resets counters and sets state to FILLING.
340 thread_storage.buffer_ptr->recycle();
341 }
