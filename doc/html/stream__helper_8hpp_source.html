122 [[nodiscard]] bool serialize_integral(const T& value) {
123 std::ostream& out = get_out();
124 if constexpr(FixedWidthIntegral<T>) {
125 T final_value = value;
126 // Flip endianness if needed.
127 if constexpr(sizeof(T) > 1 && Endian::is_big_endian()) {
128 final_value = Bit::byteswap(final_value);
129 }
130 out.write(reinterpret_cast<const char*>(&final_value), sizeof(final_value));
131 if (! out) {
132 return fail("Unable to write integral to out in serialize_integral");
133 }
134 } else if constexpr (GMPIntegral<T>) {
135 // Sign: 0 = zero, 1 = positive, -1 = negative
136 int8_t sign = mpz_sgn(value.get_mpz_t());
137 if (! serialize_integral(sign)) {
138 return fail("MPZ sign serialize failed");
139 }
140
141 // When zero, just skip anything else.
142 if (sign == 0) {
143 return true;
144 }
145
146 // Export limbs (little-endian, 1 byte per limb chunk)
147 uint64_t size = 0;
148 std::vector<uint8_t> buffer(mpz_sizeinbase(value.get_mpz_t(), 2) / 8 + 1);
149 mpz_export(
150 buffer.data(),
151 &size, // Track the count for writing later.
152 -1, // Least significant word first.
153 sizeof(uint8_t), // Using 1-byte at a time.
154 -1, // Least significant byte first
155 0,
156 value.get_mpz_t()
157 );
158
159 // Upgrade the count to uint64_t. Then write it out for deserialization later.
160 if (! serialize_integral(size)) {
161 return fail("MPZ size serialize");
162 }
163
164 // Write the data now.
165 if (! write_bytes(buffer.data(), size)) {
166 return fail("Unable to write mpz_class data payload in serialize_integral");
167 }
168 } else {
169 return fail("Unsupported type sent to seralize_integral(): " + ABI::demangle<T>());
170 }
171
172 // All good.
173 return true;
174 }

191 template<AnySupportedIntegral T>
192 [[nodiscard]] bool deserialize_integral(T& rv) {
193 std::istream& in = get_in();
194 if constexpr(FixedWidthIntegral<T>) {
195 in.read(reinterpret_cast<char*>(&rv), sizeof(rv));
196 if (! in) {
197 return fail("Failed to read integral from deserialize_integral");
198 }
199 // Convert from little-endian if this system is big endian.
200 if constexpr (sizeof(T) > 1 && Endian::is_big_endian()) {
201 rv = Bit::byteswap(rv);
202 }
203 } else if constexpr (GMPIntegral<T>) {
204 // First byte is the sign.
205 int8_t sign;
206 if (! deserialize_integral(sign)) {
207 return fail("MPZ sign deserialize failed");
208 }
209
210 // When the sign is zero, there's no size or data to read.
211 if (sign == 0) {
212 return true;
213 }
214
215 // Read the size into a uint64_t from the next 8 bytes.
216 uint64_t size;
217 if (! deserialize_integral(size)) {
218 return fail("MPZ size deserialize");
219 }
220
221 // Build a buffer, and read the next <size> bytes in from it for limb data.
222 std::vector<uint8_t> buffer(size);
223 if (! read_bytes(buffer.data(), size)) {
224 return fail("Failed to read mpz_class data");
225 }
226
227 // Import the data to an mpz_class object.
228 mpz_import(
229 rv.get_mpz_t(),
230 size,
231 -1, // Least significant word first.
232 sizeof(uint8_t), // Using 1 byte at a time.
233 -1, // Least significant byte first.
234 0,
235 buffer.data()
236 );
237
238 // Check the sign and flip it if necessary.
239 if (sign < 0) {
240 rv = -rv;
241 }
242 } else {
243 return fail("Unsupported type sent to deseralize_integral(): " + ABI::demangle<T>());
244 }
245
246 // All good.
247 return true;
248 }
