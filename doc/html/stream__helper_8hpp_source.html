71 [[nodiscard]] bool serialize_integral_impl(const T& value) {
72 std::ostream& out = get_out();
73 if constexpr(FixedWidthIntegral<T>) {
74 T final_value = value;
75 // Flip endianness if needed.
76 if constexpr(sizeof(T) > 1 && Endian::is_big_endian()) {
77 final_value = Bit::byteswap(final_value);
78 }
79 out.write(reinterpret_cast<const char*>(&final_value), sizeof(final_value));
80 if (! out) {
81 return fail("Unable to write integral to out in serialize_integral");
82 }
83 } else if constexpr (GMPIntegral<T>) {
84 // Sign: 0 = zero, 1 = positive, -1 = negative
85 int8_t sign = mpz_sgn(value.get_mpz_t());
86 if (! serialize_integral(sign)) {
87 return fail("MPZ sign serialize failed");
88 }
89
90 // When zero, just skip anything else.
91 if (sign == 0) {
92 return true;
93 }
94
95 // Export limbs (little-endian, 1 byte per limb chunk)
96 uint64_t size = 0;
97 std::vector<uint8_t> buffer(mpz_sizeinbase(value.get_mpz_t(), 2) / 8 + 1);
98 mpz_export(
99 buffer.data(),
100 &size, // Track the count for writing later.
101 -1, // Least significant word first.
102 sizeof(uint8_t), // Using 1-byte at a time.
103 -1, // Least significant byte first
104 0,
105 value.get_mpz_t()
106 );
107
108 // Upgrade the count to uint64_t. Then write it out for deserialization later.
109 if (! serialize_integral(size)) {
110 return fail("MPZ size serialize");
111 }
112
113 // Write the data now.
114 if (! write_bytes(buffer.data(), size)) {
115 return fail("Unable to write mpz_class data payload in serialize_integral");
116 }
117 } else {
118 return fail("Unsupported type sent to seralize_integral(): " + ABI::demangle<T>());
119 }
120
121 // All good.
122 return true;
123 }

140 template<AnySupportedIntegral T>
141 [[nodiscard]] bool deserialize_integral_impl(T& rv) {
142 std::istream& in = get_in();
143 if constexpr(FixedWidthIntegral<T>) {
144 static_assert(GuaranteedWidthIntegral<T>);
145 in.read(reinterpret_cast<char*>(&rv), sizeof(rv));
146 if (! in) {
147 return fail("Failed to read integral from deserialize_integral");
148 }
149 // Convert from little-endian if this system is big endian.
150 if constexpr (sizeof(T) > 1 && Endian::is_big_endian()) {
151 rv = Bit::byteswap(rv);
152 }
153 } else if constexpr (GMPIntegral<T>) {
154 // First byte is the sign.
155 int8_t sign;
156 if (! deserialize_integral(sign)) {
157 return fail("MPZ sign deserialize failed");
158 }
159
160 // When the sign is zero, there's no size or data to read.
161 if (sign == 0) {
162 return true;
163 }
164
165 // Read the size into a uint64_t from the next 8 bytes.
166 uint64_t size;
167 if (! deserialize_integral(size)) {
168 return fail("MPZ size deserialize");
169 }
170
171 // Build a buffer, and read the next <size> bytes in from it for limb data.
172 std::vector<uint8_t> buffer(size);
173 if (! read_bytes(buffer.data(), size)) {
174 return fail("Failed to read mpz_class data");
175 }
176
177 // Import the data to an mpz_class object.
178 mpz_import(
179 rv.get_mpz_t(),
180 size,
181 -1, // Least significant word first.
182 sizeof(uint8_t), // Using 1 byte at a time.
183 -1, // Least significant byte first.
184 0,
185 buffer.data()
186 );
187
188 // Check the sign and flip it if necessary.
189 if (sign < 0) {
190 rv = -rv;
191 }
192 } else {
193 return fail("Unsupported type sent to deseralize_integral(): " + ABI::demangle<T>());
194 }
195
196 // All good.
197 return true;
198 }
