121 //
122 // Serialize Integral
123 // Will serialize any integral of type T. When type is fixed-width, the caller is expected to deserialize using the same type
124 // to ensure the correct number of bytes are read, and therefore no size data is emitted to the ostream. When the type is
125 // variable length (mpz_class), we will emit sign and (if non-zero) chunk count and then limb data.
126 //
127 // Note, you must use a GuaranteedWidthIntegral. Things like size_t and int/uint are not portable.
128 //
129 template<AnySupportedIntegral T>
130 [[nodiscard]] bool serialize_integral(const T& value) {
131 std::ostream& out = get_out();
132 if constexpr(BuiltinIntegral<T>) {
133 static_assert(GuaranteedWidthIntegral<T>);
134 T final_value = value;
135 // Flip endianness if needed.
136 if constexpr(sizeof(T) > 1 && is_big_endian()) {
137 final_value = byteswap(final_value);
138 }
139 out.write(reinterpret_cast<const char*>(&final_value), sizeof(final_value));
140 if (! out) {
141 return fail("Unable to write integral to out in serialize_integral");
142 }
143 } else {
144 // Sign: 0 = zero, 1 = positive, -1 = negative
145 int8_t sign = mpz_sgn(value.get_mpz_t());
146 if (! serialize_integral(sign)) {
147 return fail("MPZ sign serialize failed");
148 }
149
150 // When zero, we can just skip anything else.
151 if (sign == 0) {
152 return true;
153 }
154
155 // Export limbs (little-endian, 1 byte per limb chunk)
156 uint64_t size = 0;
157 std::vector<uint8_t> buffer(mpz_sizeinbase(value.get_mpz_t(), 2) / 8 + 1);
158 mpz_export(
159 buffer.data(),
160 &size, // Track the count for writing later.
161 -1, // Least significant word first.
162 sizeof(uint8_t), // We're using 1-byte at a time.
163 -1, // Least significant byte first
164 0,
165 value.get_mpz_t()
166 );
167
168 // Upgrade the count to uint64_t. Then write it out for deserialization later.
169 if (! serialize_integral(size)) {
170 return fail("MPZ size serialize");
171 }
172
173 // Write the data now.
174 if (! write_bytes(buffer.data(), size)) {
175 return fail("Unable to write mpz_class data payload in serialize_integral");
176 }
177 }
178
179 // All good.
180 return true;
181 }

185 //
186 // Deserialize Integral
187 // Will deserialize any integral of type T from "in". When the type of T is fixed-width, this function uses the sizeof(T) to
188 // determine how many bytes to read. When T is variable length (mpz_class), it will expect a 8 bytes (unit64_t) to indicate
189 // how many bytes of limb data should be read next.
190 //
191 // We will write the result to a return value (rv) to avoid allocs on the GMP path.
192 //
193 // Note, you must use a GuaranteedWidthIntegral. Things like size_t and int/uint are not portable.
194 //
195 template<AnySupportedIntegral T>
196 [[nodiscard]] bool deserialize_integral(T& rv) {
197 std::istream& in = get_in();
198 if constexpr(BuiltinIntegral<T>) {
199 static_assert(GuaranteedWidthIntegral<T>);
200 in.read(reinterpret_cast<char*>(&rv), sizeof(rv));
201 if (! in) {
202 return fail("Failed to read integral from deserialize_integral");
203 }
204 // Convert from little-endian if this system is big endian.
205 if constexpr (sizeof(T) > 1 && is_big_endian()) {
206 rv = byteswap(rv);
207 }
208 } else {
209 // First byte is the sign.
210 int8_t sign;
211 if (! deserialize_integral(sign)) {
212 return fail("MPZ sign deserialize failed");
213 }
214
215 // When the sign is zero, there's no size or data to read.
216 if (sign == 0) {
217 return true;
218 }
219
220 // Read the size into a uint64_t from the next 8 bytes.
221 uint64_t size;
222 if (! deserialize_integral(size)) {
223 return fail("MPZ size deserialize");
224 }
225
226 // Build a buffer, and read the next <size> bytes in from it for limb data.
227 std::vector<uint8_t> buffer(size);
228 if (! read_bytes(buffer.data(), size)) {
229 return fail("Failed to read mpz_class data");
230 }
231
232 // Import the data to an mpz_class object.
233 mpz_import(
234 rv.get_mpz_t(),
235 size,
236 -1, // Least significant word first.
237 sizeof(uint8_t), // Using 1 byte at a time.
238 -1, // Least significant byte first.
239 0,
240 buffer.data()
241 );
242
243 // Check the sign and flip it if necessary.
244 if (sign < 0) {
245 rv = -rv;
246 }
247 }
248
249 // All good.
250 return true;
251 }

287 //
288 // ByteSwap
289 // Perform a standard bswap on any builtin type. Requires __builtin_bswapXX().
290 //
291 template<BuiltinIntegral T>
292 static T byteswap(T value) {
293 if constexpr (sizeof(T) == 1) {
294 return value;
295 } else if constexpr (sizeof(T) == 2) {
296 return __builtin_bswap16(value);
297 } else if constexpr (sizeof(T) == 4) {
298 return __builtin_bswap32(value);
299 } else if constexpr (sizeof(T) == 8) {
300 return __builtin_bswap64(value);
301 } else if constexpr (sizeof(T) == 16) {
302 // 128-bit needs broken into two chunks.
303 uint64_t high = static_cast<uint64_t>(value >> 64);
304 uint64_t low = static_cast<uint64_t>(value);
305 high = __builtin_bswap64(high);
306 low = __builtin_bswap64(low);
307 return (static_cast<T>(low) << 64) | high;
308 } else {
309 static_assert(sizeof(T) <= 16, "Unsupported size for byteswap");
310 return value;
311 }
312 }
