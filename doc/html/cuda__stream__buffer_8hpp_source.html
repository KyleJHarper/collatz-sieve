228 template<typename Kernel>
229 void launch_kernel(Kernel&& kernel_func, size_t base_multiplier = 1, size_t max_multiplier = 1, T scaling_factor = 0) {
230 // Make sure this buffer is READY. Otherwise, error.
231 if (get_state() != CudaStreamBufferState::READY) {
232 throw std::runtime_error("A buffer must be in the READY state before launching a kernel.");
233 }
234
235 // Synchronize data from host to device.
236 sync_host_data_to_device();
237 sync_host_results_to_device();
238
239 // Launch the user-provided kernel wrapper.
240 kernel_func(_device_data, _device_results, _next_index, _stream, base_multiplier, max_multiplier, scaling_factor);
241
242 // Sync the results back. Don't need the device buffer (data).
243 sync_device_results_to_host();
244
245 // Inject a state = PROCESSED flag operation directly into the stream. Guarantees ordering after GPU threads are done.
246 cudaLaunchHostFunc(_stream, [](void* p) {
247 static_cast<CudaStreamBuffer*>(p)->set_state(CudaStreamBufferState::PROCESSED);
248 }, this);
