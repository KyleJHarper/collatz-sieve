241 msg += to_string_any(bits) + " bits. A Collatz sequence will overflow.";
242 msg += " Max level for this type is " + to_string_any(max_level) + ".";
243 throw std::out_of_range(msg);

326 static bool st_equal(const BinaryTree<T, TreeType>& first, const BinaryTree<T, TreeType>& second, std::string* err = nullptr) {
327 // Common to all trees.
328 EqualityHelper eq(err);
329 eq.set_category("BinaryTree");
330
331 // Root node
332 if (! Node<T>::st_equal(first.get_root_node(), second.get_root_node(), err)) {
333 return eq.fail("Root nodes don't match.");
334 }
335
336 // Level count
337 if (eq.unequal(first.get_level_count(), second.get_level_count())) {
338 return eq.fail("Level counts differ");
339 }
340
341 // Flags
342 if (eq.unequal(first.is_preserving_ancestors(), second.is_preserving_ancestors())) {
343 return eq.fail("Is preserving ancestors mismatch");
344 }
345 if (eq.unequal(first.is_verifying_non_hwm_nodes(), second.is_verifying_non_hwm_nodes())) {
346 return eq.fail("Is verifying hwm nodes mismatch");
347 }
348 if (eq.unequal(first.is_initialized(), second.is_initialized())) {
349 return eq.fail("Is initialized mismatch");
350 }
351
352 // Coverage
353 // Ensure the same size because it's cheap.
354 if (eq.unequal(first.get_coverage_map().size(), second.get_coverage_map().size())) {
355 return eq.fail("Coverage map size differs");
356 }
357 // Compare each level+coverage combo.
358 for (const auto& [level, f_coverage] : first.get_coverage_map()) {
359 if (! second.get_coverage_map().contains(level)) {
360 return eq.fail("Coverage map in 'first' has level " + to_string_any(level) + " but 'second doesn't.");
361 }
362 const BinaryTreeCoverage<T>& s_coverage = second.get_coverage_map().at(level);
363 if (eq.unequal(f_coverage.get_covered(), s_coverage.get_covered())) {
364 return eq.fail("Coverage get_covered() mismatch");
365 }
366 if (eq.unequal(f_coverage.get_total(), s_coverage.get_total())) {
367 return eq.fail("Coverage get_total() mismatch");
368 }
369 }
370
371 // Ancestors
372 // For this to work, they must be a consistent order, which the impls guarantee on add_level();
373 const std::vector<Node<T>*>& f_ancestors = first.get_ancestors();
374 const std::vector<Node<T>*>& s_ancestors = second.get_ancestors();
375 if (eq.unequal(f_ancestors.size(), s_ancestors.size())) {
376 return eq.fail("Ancestor count mismatch");
377 }
378 for (size_t i = 0; i < f_ancestors.size(); i++) {
379 if (! Node<T>::st_equal(f_ancestors.at(i), s_ancestors.at(i), err)) {
380 return eq.fail("Ancestor node mismatch");
381 }
382 }
383
384 // Implementation-specific checks.
385 if (! TreeType::st_equal(first.get_impl(), second.get_impl(), err)) {
386 return eq.fail("Implementation-specific details mismatched");
387 }
388
389 // Everything is okay.
390 return true;
391 }

420 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
421 StreamHelper sh(nullptr, &out, err);
422 sh.set_category("BinaryTree");
423
424 // Header
425 const BinaryTreeFileHeader header;
426 if (! sh.write_bytes(&(header.magic), header.MAGIC_SIZE)) {
427 return sh.fail("header magic string==" + std::string(header.magic));
428 }
429 if (! sh.serialize_integral(header.version)) {
430 return sh.fail("header version==" + to_string_any(header.version));
431 }
432
433 // Common Metadata
434 uint8_t tree_type = IsImplicitTree<TreeType> ? TreeTypeEnum::IMPLICIT : TreeTypeEnum::MATERIALIZED;
435 uint32_t level_count = get_level_count();
436 bool b_is_preserving_ancestors = is_preserving_ancestors();
437 bool b_is_verifying_non_hwm_nodes = is_verifying_non_hwm_nodes();
438 if (! sh.serialize_integral(tree_type)) {
439 return sh.fail("tree_type==" + to_string_any(tree_type));
440 }
441 if (! sh.serialize_integral(level_count)) {
442 return sh.fail("level_count==" + to_string_any(level_count));
443 }
444 if (! sh.serialize_bool(b_is_preserving_ancestors)) {
445 return sh.fail("b_is_preserving_ancestors==" + std::to_string(b_is_preserving_ancestors));
446 }
447 if (! sh.serialize_bool(b_is_verifying_non_hwm_nodes)) {
448 return sh.fail("b_is_verifying_non_hwm_nodes==" + std::to_string(b_is_verifying_non_hwm_nodes));
449 }
450 // The is_initialized must be true, so we ignore it.
451
452 // Root node
453 bool b_has_root_node = get_root_node() != nullptr;
454 if (! sh.serialize_bool(b_has_root_node)) {
455 return sh.fail("b_has_root_node");
456 }
457 if (b_has_root_node) {
458 if (! get_root_node()->serialize(out, err)) {
459 return sh.fail("root node");
460 }
461 }
462
463 // Coverage
464 // Write out how many entries exist so we know how many loops to make.
465 const std::unordered_map<level_t, BinaryTreeCoverage<T>>& coverage_map = _impl.get_coverage_map();
466 uint64_t u64_entry_count = coverage_map.size();
467 if (! sh.serialize_integral(u64_entry_count)) {
468 return sh.fail("coverage entry_count==" + to_string_any(u64_entry_count));
469 }
470 // Loop through them, emitting the level and then the coverage itself.
471 // Sort the levels to ensure consistent output. Nodes themselves are always ordered consistently.
472 std::vector<level_t> coverage_levels;
473 coverage_levels.reserve(coverage_map.size());
474 for (const auto& [level, _] : coverage_map) {
475 coverage_levels.push_back(level);
476 }
477 std::sort(coverage_levels.begin(), coverage_levels.end());
478 for (level_t level : coverage_levels) {
479 uint64_t u64_level = static_cast<uint64_t>(level);
480 const BinaryTreeCoverage<T>& coverage = coverage_map.at(level);
481 if (! sh.serialize_integral(u64_level)) {
482 return sh.fail("u64_level==" + to_string_any(u64_level));
483 }
484 if (! coverage.serialize(out, err)) {
485 return sh.fail("coverage on level " + to_string_any(level));
486 }
487 }
488
489 // Ancestors
490 // These are new-placement constructed (no children). Just store the value.
491 // First, write out how many there are.
492 const std::vector<Node<T>*>& ancestors = get_ancestors();
493 uint64_t u64_size = static_cast<uint64_t>(ancestors.size());
494 if (! sh.serialize_integral(u64_size)) {
495 return sh.fail("get_ancestors().size()==" + to_string_any(u64_size));
496 }
497 // Now loop through them.
498 for (const Node<T>* ancestor : ancestors) {
499 if (! ancestor->serialize(out, err)) {
500 return sh.fail("ancestor with value==" + to_string_any(ancestor->get_value()));
501 }
502 }
503
504 // Implementation-Specific Hook
505 if (! _impl.serialize(out, err)) {
506 return sh.fail("implementation-specific hook issues");
507 }
508
509 // All Good
510 return true;
511 }

525 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
526 StreamHelper sh(&in, nullptr, err);
527 sh.set_category("BinaryTree");
528 EqualityHelper eq(err);
529
530 // Reset this object.
531 _impl.reset();
532
533 // Header
534 const BinaryTreeFileHeader header;
535 char magic[header.MAGIC_SIZE];
536 if (! sh.read_bytes(&magic, header.MAGIC_SIZE)) {
537 return sh.fail("couldn't read header magic string");
538 }
539 if (! eq.equal(std::string(magic, header.MAGIC_SIZE), std::string(header.magic, header.MAGIC_SIZE))) {
540 return sh.fail("Magic in header doesn't match the known value. Wrong file?");
541 }
542 uint32_t version;
543 if (! sh.deserialize_integral(version)) {
544 return sh.fail("couldn't read header version");
545 }
546 if (! eq.equal(version, header.version)) {
547 return sh.fail("Version mismatch");
548 }
549
550 // Common Metadata
551 uint8_t tree_type;
552 uint32_t level_count;
553 bool b_is_preserving_ancestors;
554 bool b_is_verifying_non_hwm_nodes;
555 // Tree type. Has to match.
556 if (! sh.deserialize_integral(tree_type)) {
557 return sh.fail("couldn't read tree_type");
558 }
559 if (! eq.equal(tree_type, static_cast<uint8_t>(_impl.get_tree_type()))) {
560 return sh.fail("Tree types don't match");
561 }
562 // Level count. Type T needs to support this many levels.
563 if (! sh.deserialize_integral(level_count)) {
564 return sh.fail("couldn't read level_count");
565 }
566 if (! BinaryTreeMath<T>::st_level_will_fit(level_count)) {
567 std::string msg = "Level count found ("
568 + to_string_any(level_count)
569 + ") exceeds the amount permitted ("
570 + to_string_any(BinaryTreeMath<T>::st_max_level_of_type())
571 + ") by the type T you specified: "
572 + ABIHelpers::demangle(typeid(T).name())
573 + ".";
574 return sh.fail(msg);
575 }
576 _impl.set_level_count(level_count);
577 // Is preserving ancestors
578 if (! sh.deserialize_bool(b_is_preserving_ancestors)) {
579 return sh.fail("couldn't read is_preserving_ancestors");
580 }
581 _impl.set_is_preserving_ancestors(b_is_preserving_ancestors);
582 // Is verifying non-hwm nodes
583 if (! sh.deserialize_bool(b_is_verifying_non_hwm_nodes)) {
584 return sh.fail("couldn't read is_verifying_non_hwm_nodes");
585 }
586 if (b_is_verifying_non_hwm_nodes) {
587 enable_non_hwm_node_verification();
588 } else {
589 disable_non_hwm_node_verification();
590 }
591 // The is_initialized must be true, so we set it manually.
592 _impl.set_is_initialized(true);
593
594 // Root node
595 bool source_has_root_node;
596 if (! sh.deserialize_bool(source_has_root_node)) {
597 return sh.fail("has_root_node");
598 }
599 if (source_has_root_node) {
600 Node<T>*& root_node = _impl.get_root_node_rw();
601 // Root node can be null on *this* object, so make one for it to use.
602 if (root_node == nullptr) {
603 root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
604 }
605 T root_parent_v;
606 T root_hwm_ancestor_v;
607 uint8_t child_count;
608 if (! root_node->deserialize(in, root_parent_v, root_hwm_ancestor_v, child_count, err)) {
609 return sh.fail("root node");
610 }
611 if (root_parent_v != 0) {
612 return sh.fail("Root node's parent value must be 0 (absent), but it's: " + to_string_any(root_parent_v));
613 }
614 if (root_hwm_ancestor_v != 0) {
615 return sh.fail("Root node's hwm ancestor value must be 0 (absent), but it's: " + to_string_any(root_hwm_ancestor_v));
616 }
617 } else {
618 // The backed up file didn't have a root node... which means it was a 0-level tree :/
619 if (_impl.get_root_node() != nullptr) {
620 Node<T>*& root_node = _impl.get_root_node_rw();
621 delete root_node;
622 root_node = nullptr;
623 }
624 }
625
626 // Coverage
627 std::unordered_map<level_t, BinaryTreeCoverage<T>>& coverage_map = _impl.get_coverage_map_rw();
628 uint64_t u64_entry_count = 0;
629 if (! sh.deserialize_integral(u64_entry_count)) {
630 return sh.fail("couldn't get coverage's entry count");
631 }
632 for (uint64_t i = 0; i < u64_entry_count; i++) {
633 uint64_t u64_level;
634 if (! sh.deserialize_integral(u64_level)) {
635 return sh.fail("couldn't read coverage level for index i==" + to_string_any(i));
636 }
637 level_t level = static_cast<level_t>(u64_level);
638 BinaryTreeCoverage<T> coverage;
639 if (! coverage.deserialize(in, err)) {
640 return sh.fail("coverage on level " + to_string_any(level));
641 }
642 // Level and coverage are okay. Throw them in the map.
643 coverage_map[level] = coverage;
644 }
645
646 // Ancestors
647 // These are new-placement (no children). Slam them into the vector.
648 uint64_t u64_size = 0;
649 std::vector<Node<T>*>& ancestors_rw = _impl.get_ancestors_rw();
650 if (! sh.deserialize_integral(u64_size)) {
651 return sh.fail("couldn't read ancestors.size() count");
652 }
653 ancestors_rw.reserve(static_cast<size_t>(u64_size));
654 T parent_v;
655 T hwm_ancestor_v;
656 uint8_t child_count;
657 for (uint64_t i = 0; i < u64_size; i++) {
658 Node<T>* new_node = new Node<T>(T(1));
659 if (! new_node->deserialize(in, parent_v, hwm_ancestor_v, child_count, err)) {
660 return sh.fail("couldn't deserialize ancestor at index i==" + to_string_any(i));
661 }
662 ancestors_rw.push_back(new_node);
663 }
664
665 // Implementation-specific hook
666 if (! _impl.deserialize(in, err)) {
667 return sh.fail("implementation-specific hook issues");
668 }
669
670 // All good
671 return true;
672 }

689 bool save(const std::string& path, int compression_level = 22) {
690 // Sanity checks.
691 if (compression_level > 22) {
692 throw std::out_of_range("Zstd compression cannot exceed 22. You sent " + to_string_any(compression_level) + " to BinaryTree.save()");
693 }
694 if (compression_level < 0) {
695 throw std::out_of_range("While Zstd supports negative compression levels, we do not. You sent " + to_string_any(compression_level) + " to BinaryTree.save()");
696 }
697
698 // Setup some pointers to the file output and a potential zstd out.
699 std::ofstream file_out;
700 std::ostream* final_out = nullptr;
701 std::unique_ptr<zstd_ostream> zstd_out;
702
703 // Open the file stream.
704 const bool use_compression = compression_level > 0;
705 if (path == "-") {
706 // Write to stdout. Ensure it's not a terminal.
707 if (isatty(fileno(stdout))) {
708 throw std::runtime_error("Refusing to write binary data to a terminal. Redirect to a file, pipe, etc.");
709 }
710 final_out = &std::cout;
711 } else {
712 // Write to a normal file.
713 file_out.open(path, std::ios::binary);
714 if (!file_out) {
715 throw std::runtime_error("Failed to open file for writing");
716 }
717 final_out = &file_out;
718 }
719
720 // If compression is enabled, thread it into our zstd_ostream helper.
721 if (use_compression) {
722 zstd_out = std::make_unique<zstd_ostream>(*final_out, compression_level);
723 final_out = zstd_out.get();
724 }
725
726 // Perform the serialization and see if it worked.
727 std::string err;
728 bool ok = serialize(*final_out, &err);
729
730 // If compression is used, zstd requires finialization logic to finish a frame.
731 if (use_compression) {
732 zstd_out->finalize();
733 } else {
734 // We'll flush a raw file here just for good practice.
735 final_out->flush();
736 }
737 if (! ok) {
738 throw std::runtime_error("Failed to serialize data. Error chain is: " + err);
739 }
740
741 return ok && final_out->good();
742 }

751 bool load(const std::string& path) {
752 // Open the file stream.
753 std::ifstream f_in(path, std::ios::binary);
754 if (!f_in) {
755 throw std::runtime_error("Failed to open file for reading");
756 }
757
758 // Link an istream to the file input to start.
759 std::istream* final_in = &f_in;
760 std::unique_ptr<zstd_istream> zstd_in;
761
762 // Check for Zstd compression. Then rewind our cursor.
763 std::string err;
764 uint32_t four_byte_magic;
765 {
766 StreamHelper sh(final_in, nullptr, &err);
767 std::streampos pos = f_in.tellg();
768 if (! sh.deserialize_integral(four_byte_magic)) {
769 throw std::runtime_error("Failed to deserialize first four bytes to detect Zstd compression. Error chain is: " + err);
770 }
771 f_in.clear();
772 f_in.seekg(pos);
773 }
774
775 // If we have Zstd compression, we need to link the file to our decompressor and pass on a different istream.
776 const bool is_zstd = four_byte_magic == ZSTD_MAGICNUMBER;
777 if (is_zstd) {
778 zstd_in = std::make_unique<zstd_istream>(*final_in);
779 final_in = zstd_in.get();
780 }
781
782 // Now we can deserialize and it'll only receive the raw data.
783 err.clear();
784 if (! deserialize(*final_in, &err)) {
785 throw std::runtime_error("Failed to deserialize data (zstd compressed=" + to_string_any(is_zstd) + "). Error chain is: " + err);
786 }
787
788 return true;
789 }
