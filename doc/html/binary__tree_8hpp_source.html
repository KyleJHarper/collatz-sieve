275 std::string msg = "Cannot build a BinaryTree with ";
276 msg += to_string_any(level) + " levels and type '" + typeid(T).name() + "' with ";
277 msg += to_string_any(bits) + " bits. A Collatz sequence will overflow.";
278 msg += " Max level for this type is " + to_string_any(max_level) + ".";
279 throw std::out_of_range(msg);

363 static bool st_equal(const BinaryTree<T, TreeType>& first, const BinaryTree<T, TreeType>& second, std::string* err = nullptr) {
364 // Common to all trees.
365 EqualityHelper eq(err);
366 eq.set_category("BinaryTree");
367
368 // Root node
369 if (! Node<T>::st_equal(first.get_root_node(), second.get_root_node(), err)) {
370 return eq.fail("Root nodes don't match.");
371 }
372
373 // Level count
374 if (! eq.equal(first.get_level_count(), second.get_level_count())) {
375 return eq.fail("Level counts differ");
376 }
377
378 // Scaling factor
379 if (! eq.equal(first.get_scaling_factor(), second.get_scaling_factor())) {
380 return eq.fail("Scaling factors differ");
381 }
382
383 // Flags
384 if (! eq.equal(first.is_preserving_ancestors(), second.is_preserving_ancestors())) {
385 return eq.fail("Is preserving ancestors mismatch");
386 }
387 if (! eq.equal(first.is_verifying_non_ast_nodes(), second.is_verifying_non_ast_nodes())) {
388 return eq.fail("Is verifying ast nodes mismatch");
389 }
390 if (! eq.equal(first.is_initialized(), second.is_initialized())) {
391 return eq.fail("Is initialized mismatch");
392 }
393
394 // Coverage
395 // Ensure the same size because it's cheap.
396 if (! eq.equal(first.get_coverage_map().size(), second.get_coverage_map().size())) {
397 return eq.fail("Coverage map size differs");
398 }
399 // Compare each level+coverage combo.
400 for (const auto& [level, f_coverage] : first.get_coverage_map()) {
401 if (! second.get_coverage_map().contains(level)) {
402 return eq.fail("Coverage map in 'first' has level " + to_string_any(level) + " but 'second doesn't.");
403 }
404 const BinaryTreeCoverage<T>& s_coverage = second.get_coverage_map().at(level);
405 if (! f_coverage.equal(s_coverage, err)) {
406 return eq.fail("Coverage doesn't match");
407 }
408 }
409
410 // Ancestors
411 // For this to work, they must be a consistent order, which the impls guarantee on add_level();
412 const std::vector<Node<T>*>& f_ancestors = first.get_ancestors();
413 const std::vector<Node<T>*>& s_ancestors = second.get_ancestors();
414 if (! eq.equal(f_ancestors.size(), s_ancestors.size())) {
415 return eq.fail("Ancestor count mismatch");
416 }
417 for (size_t i = 0; i < f_ancestors.size(); i++) {
418 if (! Node<T>::st_equal(f_ancestors.at(i), s_ancestors.at(i), err)) {
419 return eq.fail("Ancestor node mismatch");
420 }
421 }
422
423 // Uncovered Values Bitmap
424 if (! first.get_uncovered_values().equal(second.get_uncovered_values())) {
425 return eq.fail("Value bitmaps mismatch");
426 }
427
428 // Implementation-specific checks.
429 if (! TreeType::st_equal(first.get_impl(), second.get_impl(), err)) {
430 return eq.fail("Implementation-specific details mismatched");
431 }
432
433 // Everything is okay.
434 return true;
435 }

464 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
465 StreamHelper sh(nullptr, &out, err);
466 sh.set_category("BinaryTree");
467
468 // Header
469 const BinaryTreeFileHeader header;
470 if (! sh.write_bytes(&(header.magic), header.MAGIC_SIZE)) {
471 return sh.fail("header magic string==" + std::string(header.magic));
472 }
473 if (! sh.serialize_integral(header.version)) {
474 return sh.fail("header version==" + to_string_any(header.version));
475 }
476
477 // Common Metadata
478 // Currently, only mpz_class is arbitrary, so "0" (zero) will mean "mpz_class".
479 uint8_t tree_type = IsImplicitTree<TreeType> ? TreeTypeEnum::IMPLICIT : TreeTypeEnum::MATERIALIZED;
480 uint16_t tree_type_bit_width_id = ABI::get_bit_width_id<T>();
481 level_t level_count = get_level_count();
482 bool b_is_preserving_ancestors = is_preserving_ancestors();
483 bool b_is_verifying_non_ast_nodes = is_verifying_non_ast_nodes();
484 if (! sh.serialize_integral(tree_type)) {
485 return sh.fail("tree_type==" + to_string_any(tree_type));
486 }
487 if (! sh.serialize_integral(tree_type_bit_width_id)) {
488 return sh.fail("tree_type_bit_width_id=" + to_string_any(tree_type_bit_width_id));
489 }
490 if (! sh.serialize_integral(level_count)) {
491 return sh.fail("level_count==" + to_string_any(level_count));
492 }
493 if (! sh.serialize_integral(_scaling_factor)) {
494 return sh.fail("scaling_factor==" + to_string_any(_scaling_factor));
495 }
496 if (! sh.serialize_bool(b_is_preserving_ancestors)) {
497 return sh.fail("b_is_preserving_ancestors==" + std::to_string(b_is_preserving_ancestors));
498 }
499 if (! sh.serialize_bool(b_is_verifying_non_ast_nodes)) {
500 return sh.fail("b_is_verifying_non_ast_nodes==" + std::to_string(b_is_verifying_non_ast_nodes));
501 }
502 // The is_initialized must be true, so we ignore it.
503
504 // Root node
505 bool b_has_root_node = get_root_node() != nullptr;
506 if (! sh.serialize_bool(b_has_root_node)) {
507 return sh.fail("b_has_root_node");
508 }
509 if (b_has_root_node) {
510 if (! get_root_node()->serialize(out, err)) {
511 return sh.fail("root node");
512 }
513 }
514
515 // Coverage
516 // Write out how many entries exist so we know how many loops to make.
517 const std::unordered_map<level_t, BinaryTreeCoverage<T>>& coverage_map = _impl.get_coverage_map();
518 uint64_t u64_entry_count = coverage_map.size();
519 if (! sh.serialize_integral(u64_entry_count)) {
520 return sh.fail("coverage entry_count==" + to_string_any(u64_entry_count));
521 }
522 // Loop through them, emitting the level and then the coverage itself.
523 // Sort the levels to ensure consistent output. Nodes themselves are always ordered consistently.
524 std::vector<level_t> coverage_levels;
525 coverage_levels.reserve(coverage_map.size());
526 for (const auto& [level, _] : coverage_map) {
527 coverage_levels.push_back(level);
528 }
529 std::sort(coverage_levels.begin(), coverage_levels.end());
530 for (level_t level : coverage_levels) {
531 uint64_t u64_level = static_cast<uint64_t>(level);
532 const BinaryTreeCoverage<T>& coverage = coverage_map.at(level);
533 if (! sh.serialize_integral(u64_level)) {
534 return sh.fail("u64_level==" + to_string_any(u64_level));
535 }
536 if (! coverage.serialize(out, err)) {
537 return sh.fail("coverage on level " + to_string_any(level));
538 }
539 }
540
541 // Ancestors
542 // These are new-placement constructed (no children). Just store the value.
543 // First, write out how many there are.
544 const std::vector<Node<T>*>& ancestors = get_ancestors();
545 uint64_t u64_size = static_cast<uint64_t>(ancestors.size());
546 if (! sh.serialize_integral(u64_size)) {
547 return sh.fail("get_ancestors().size()==" + to_string_any(u64_size));
548 }
549 // Now loop through them.
550 for (const Node<T>* ancestor : ancestors) {
551 if (! ancestor->serialize(out, err)) {
552 return sh.fail("ancestor with value==" + to_string_any(ancestor->get_value()));
553 }
554 }
555
556 // Uncovered Values Bitmap
557 if (! get_uncovered_values().serialize(out, err)) {
558 return sh.fail("uncovered values error");
559 }
560
561 // Implementation-Specific Hook
562 if (! _impl.serialize(out, err)) {
563 return sh.fail("implementation-specific hook issues");
564 }
565
566 // All Good
567 return true;
568 }

582 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
583 StreamHelper sh(&in, nullptr, err);
584 sh.set_category("BinaryTree");
585 EqualityHelper eq(err);
586
587 // Reset this object.
588 _impl.reset();
589
590 // Header
591 const BinaryTreeFileHeader header;
592 char magic[header.MAGIC_SIZE];
593 if (! sh.read_bytes(&magic, header.MAGIC_SIZE)) {
594 return sh.fail("couldn't read header magic string");
595 }
596 if (! eq.equal(std::string(magic, header.MAGIC_SIZE), std::string(header.magic, header.MAGIC_SIZE))) {
597 return sh.fail("Magic in header doesn't match the known value. Wrong file?");
598 }
599 uint32_t version;
600 if (! sh.deserialize_integral(version)) {
601 return sh.fail("couldn't read header version");
602 }
603 if (! eq.equal(version, header.version)) {
604 return sh.fail("Version mismatch");
605 }
606
607 // Common Metadata
608 uint8_t tree_type;
609 uint16_t tree_type_bit_width_id;
610 level_t level_count;
611 bool b_is_preserving_ancestors;
612 bool b_is_verifying_non_ast_nodes;
613 // Tree type. Has to match.
614 if (! sh.deserialize_integral(tree_type)) {
615 return sh.fail("couldn't read tree_type");
616 }
617 if (! eq.equal(tree_type, static_cast<uint8_t>(_impl.get_tree_type()))) {
618 return sh.fail("Tree types don't match");
619 }
620 // Tree type bit width ID. Has to match.
621 if (! sh.deserialize_integral(tree_type_bit_width_id)) {
622 return sh.fail("couldn't read tree_type_bit_width_id (for type T inference)");
623 }
624 if (!eq.equal(tree_type_bit_width_id, static_cast<uint16_t>(ABI::get_bit_width_id<T>()))) {
625 return sh.fail("Tree types do not have matching bit width type IDs");
626 }
627 // Level count. Type T needs to support this many levels.
628 if (! sh.deserialize_integral(level_count)) {
629 return sh.fail("couldn't read level_count");
630 }
631 if (! BinaryTreeMath<T>::st_level_will_fit(level_count)) {
632 std::string msg = "Level count found ("
633 + to_string_any(level_count)
634 + ") exceeds the amount permitted ("
635 + to_string_any(BinaryTreeMath<T>::st_max_level_of_type())
636 + ") by the type T you specified: "
637 + ABI::demangle(typeid(T).name())
638 + ".";
639 return sh.fail(msg);
640 }
641 _impl.set_level_count(level_count);
642 // Scaling factor.
643 if (! sh.deserialize_integral(_scaling_factor)) {
644 return sh.fail("couldn't read scaling_factor");
645 }
646 // Is preserving ancestors
647 if (! sh.deserialize_bool(b_is_preserving_ancestors)) {
648 return sh.fail("couldn't read is_preserving_ancestors");
649 }
650 _impl.set_is_preserving_ancestors(b_is_preserving_ancestors);
651 // Is verifying non-ast nodes
652 if (! sh.deserialize_bool(b_is_verifying_non_ast_nodes)) {
653 return sh.fail("couldn't read is_verifying_non_ast_nodes");
654 }
655 if (b_is_verifying_non_ast_nodes) {
656 enable_non_ast_node_verification();
657 } else {
658 disable_non_ast_node_verification();
659 }
660 // The is_initialized must be true, so we set it manually.
661 _impl.set_is_initialized(true);
662
663 // Root node
664 bool source_has_root_node;
665 if (! sh.deserialize_bool(source_has_root_node)) {
666 return sh.fail("has_root_node");
667 }
668 if (source_has_root_node) {
669 Node<T>*& root_node = _impl.get_root_node_rw();
670 // Root node can be null on *this* object, so make one for it to use.
671 if (root_node == nullptr) {
672 root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
673 }
674 T root_parent_v;
675 T root_ast_ancestor_v;
676 uint8_t child_count;
677 if (! root_node->deserialize(in, root_parent_v, root_ast_ancestor_v, child_count, err)) {
678 return sh.fail("root node");
679 }
680 if (root_parent_v != 0) {
681 return sh.fail("Root node's parent value must be 0 (absent), but it's: " + to_string_any(root_parent_v));
682 }
683 if (root_ast_ancestor_v != 0) {
684 return sh.fail("Root node's ast ancestor value must be 0 (absent), but it's: " + to_string_any(root_ast_ancestor_v));
685 }
686 } else {
687 // The backed up file didn't have a root node... which means it was a 0-level tree :/
688 if (_impl.get_root_node() != nullptr) {
689 Node<T>*& root_node = _impl.get_root_node_rw();
690 delete root_node;
691 root_node = nullptr;
692 }
693 }
694
695 // Coverage
696 std::unordered_map<level_t, BinaryTreeCoverage<T>>& coverage_map = _impl.get_coverage_map_rw();
697 uint64_t u64_entry_count = 0;
698 if (! sh.deserialize_integral(u64_entry_count)) {
699 return sh.fail("couldn't get coverage's entry count");
700 }
701 for (uint64_t i = 0; i < u64_entry_count; i++) {
702 uint64_t u64_level;
703 if (! sh.deserialize_integral(u64_level)) {
704 return sh.fail("couldn't read coverage level for index i==" + to_string_any(i));
705 }
706 level_t level = static_cast<level_t>(u64_level);
707 BinaryTreeCoverage<T> coverage;
708 if (! coverage.deserialize(in, err)) {
709 return sh.fail("coverage on level " + to_string_any(level));
710 }
711 // Level and coverage are okay. Throw them in the map.
712 coverage_map[level] = coverage;
713 }
714
715 // Ancestors
716 // These are new-placement (no children). Slam them into the vector.
717 uint64_t u64_size = 0;
718 std::vector<Node<T>*>& ancestors_rw = _impl.get_ancestors_rw();
719 if (! sh.deserialize_integral(u64_size)) {
720 return sh.fail("couldn't read ancestors.size() count");
721 }
722 ancestors_rw.reserve(static_cast<size_t>(u64_size));
723 T parent_v;
724 T ast_ancestor_v;
725 uint8_t child_count;
726 for (uint64_t i = 0; i < u64_size; i++) {
727 Node<T>* new_node = new Node<T>(T(1));
728 if (! new_node->deserialize(in, parent_v, ast_ancestor_v, child_count, err)) {
729 return sh.fail("couldn't deserialize ancestor at index i==" + to_string_any(i));
730 }
731 ancestors_rw.push_back(new_node);
732 }
733
734 // Uncovered Values Bitmap
735 if (! get_uncovered_values_rw().deserialize(in, err)) {
736 return sh.fail("uncovered values bitmap");
737 }
738
739 // Implementation-specific hook
740 if (! _impl.deserialize(in, err)) {
741 return sh.fail("implementation-specific hook issues");
742 }
743
744 // All good
745 return true;
746 }

763 bool save(const std::string& path, int compression_level = 22) {
764 // Sanity checks.
765 if (compression_level > 22) {
766 throw std::out_of_range("Zstd compression cannot exceed 22. You sent " + to_string_any(compression_level) + " to BinaryTree.save()");
767 }
768 if (compression_level < 0) {
769 throw std::out_of_range("While Zstd supports negative compression levels, we do not. You sent " + to_string_any(compression_level) + " to BinaryTree.save()");
770 }
771
772 // Setup some pointers to the file output and a potential zstd out.
773 std::ofstream file_out;
774 std::ostream* final_out = nullptr;
775 std::unique_ptr<zstd_ostream> zstd_out;
776
777 // Open the file stream.
778 const bool use_compression = compression_level > 0;
779 if (path == "-") {
780 // Write to stdout. Ensure it's not a terminal.
781 if (isatty(fileno(stdout))) {
782 throw std::runtime_error("Refusing to write binary data to a terminal. Redirect to a file, pipe, etc.");
783 }
784 final_out = &std::cout;
785 } else {
786 // Write to a normal file.
787 file_out.open(path, std::ios::binary);
788 if (!file_out) {
789 throw std::runtime_error("Failed to open file for writing");
790 }
791 final_out = &file_out;
792 }
793
794 // If compression is enabled, thread it into our zstd_ostream helper.
795 if (use_compression) {
796 zstd_out = std::make_unique<zstd_ostream>(*final_out, compression_level);
797 final_out = zstd_out.get();
798 }
799
800 // Perform the serialization and see if it worked.
801 std::string err;
802 bool ok = serialize(*final_out, &err);
803
804 // If compression is used, zstd requires finialization logic to finish a frame.
805 if (use_compression) {
806 zstd_out->finalize();
807 } else {
808 // We'll flush a raw file here just for good practice.
809 final_out->flush();
810 }
811 if (! ok) {
812 throw std::runtime_error("Failed to serialize data. Error chain is: " + err);
813 }
814
815 return ok && final_out->good();
816 }

830 static void get_load_ptrs(
831 const std::string& path
832 , std::ifstream& file_in
833 , std::unique_ptr<zstd_istream>& zstd_in
834 , std::istream*& final_in
835 , bool& is_zstd
836 ) {
837 // Open the file stream.
838 if (path == "-") {
839 // Read from stdin.
840 final_in = &std::cin;
841 } else {
842 // Read from a normal file.
843 file_in.open(path, std::ios::binary);
844 if (!file_in) {
845 throw std::runtime_error("Failed to open file for reading");
846 }
847 final_in = &file_in;
848 }
849
850 // Check for Zstd compression. Then rewind our cursor.
851 std::string err;
852 uint32_t four_byte_magic;
853 {
854 StreamHelper sh(final_in, nullptr, &err);
855 std::streampos pos = final_in->tellg();
856 if (! sh.deserialize_integral(four_byte_magic)) {
857 throw std::runtime_error("Failed to deserialize first four bytes to detect Zstd compression. Error chain is: " + err);
858 }
859 final_in->clear();
860 final_in->seekg(pos);
861 }

864 is_zstd = four_byte_magic == ZSTD_MAGICNUMBER;
865 if (is_zstd) {
866 zstd_in = std::make_unique<zstd_istream>(*final_in);
867 final_in = zstd_in.get();
868 }

903 const std::string& path
904 , uint32_t& version
905 , uint8_t& tree_type
906 , uint16_t& tree_type_bit_width_id
907 , level_t& level_count

928 return sh.fail("Magic in header doesn't match the known value. Wrong file?");
929 }
930 if (! sh.deserialize_integral(version)) {
931 return sh.fail("couldn't read header version");
932 }
933 if (! eq.equal(version, header.version)) {
934 return sh.fail("Version mismatch");
935 }
936
937 // Common Metadata
938 if (! sh.deserialize_integral(tree_type)) {
939 return sh.fail("couldn't read tree_type");
940 }
941 if (! sh.deserialize_integral(tree_type_bit_width_id)) {
942 return sh.fail("couldn't read tree_type_bit_width_id (for type T inference)");
943 }
944 // Level count. Type T needs to support this many levels.
945 if (! sh.deserialize_integral(level_count)) {
946 return sh.fail("couldn't read level_count");
947 }
948
949 return true;

1033 template<typename Func, typename TLS_Type, AnySupportedIntegral U = T>
1034 void for_each_uncovered_value_with_tls(ForEachPolicy policy, std::vector<TLS_Type>& tls, Func&& callback, const U start = 0) const {
1035 // Do not allow non-ref callbacks. Otherwise we make GMP over and over.
1036 static_assert(std::is_invocable_v<Func, U&, TLS_Type&>, "Callback must be callable with (U&, TLS_Type&)");
1037 // Require ForEachSignal return type.
1038 static_assert(std::is_same_v<std::invoke_result_t<Func, U&, TLS_Type&>, ForEachSignal>, "Callback must return ForEachSignal");
1039 // Tree type and value type must align on certain criteria.
1040 if constexpr(FixedWidthIntegral<T>) {
1041 // Tree is fixed-width.
1042 // Value type U must be >= T if it is also fixed width.
1043 if constexpr(FixedWidthIntegral<U>) {
1044 static_assert(sizeof(U) >= sizeof(T), "Value type (U) must have a width >= tree type (T).");
1045 }
1046 } else if constexpr(GMPIntegral<T>) {
1047 // Tree is GMPIntegral.
1048 // Value type U must be GMPIntegral as well.
1049 static_assert(GMPIntegral<U>, "Value type (U) must be GMPIntegral (mpz_class) when tree type (T) is GMPIntegral.");
1050 }
1051
1052 // NodeBitmap handles the tls resizing, so leave it alone here.
1053
1054 // Pull the impl's reference to uncovered values.
1055 const NodeBitmap<T>& uncovered_values = get_uncovered_values();
1056
1057 // Throw an error if they didn't generate the value map.
1058 if (uncovered_values.empty()) {
1059 throw std::runtime_error("Value map not generated. Must call generate_value_map() before iterating this.");
1060 }
1061
1062 // Create the multiplier to use with the scaling factor, which grows by one with each loop.
1063 T multiplier = 1;
1064
1065 // Define the maximum multiplier before scaling factor and/or total would overflow.
1066 U max_multiplier = 0;
1067 if constexpr(FixedWidthIntegral<U>) {
1068 if (_scaling_factor == 0) {
1069 // When tree is small (L=1) scaling factor will be zero. Don't get a division-by-zero error.
1070 max_multiplier = std::numeric_limits<U>::max() - uncovered_values.maximum();
1071 } else {
1072 max_multiplier = (std::numeric_limits<U>::max() - uncovered_values.maximum()) / _scaling_factor;
1073 }
1074 }
1075
1076 // Bump the multiplier if a start value requires it. Take the maximum value of the tree and subtract it from the start
1077 // value requested, then divide it by the scaling factor to get the mutliplier.
1078 if (start > uncovered_values.maximum()) {
1079 multiplier += ((start - uncovered_values.maximum()) / _scaling_factor);
1080 }
1081
1082 // Loop until the caller is done.
1083 std::atomic<ForEachSignal> a_signal = ForEachSignal::CONTINUE;
1084 while(a_signal.load(std::memory_order_relaxed) == ForEachSignal::CONTINUE) {
1085 // Calculate the total scaling factor.
1086 U total_scaling_factor = _scaling_factor * multiplier;
1087
1088 // Ensure overflow won't be hit. If it will, throw. No promotion possible because caller is locked to type U.
1089 if constexpr(FixedWidthIntegral<U>) {
1090 if (multiplier > max_multiplier) {
1091 throw std::overflow_error(
1092 std::format(
1093 "Cannot iterate uncovered values further. Value type (U) is {} bits wide. Must upgrade to wider type. May resume from start={}."
1094 , sizeof(U) * 8
1095 , uncovered_values.minimum() + total_scaling_factor
1096 )
1097 );
1098 }
1099 }
1100
1101 // Loop through each position, adjusting it with the total scaling factor before sending it back.
1102 uncovered_values.for_each_value_with_tls(policy, tls, [&](T& value, TLS_Type& my_tls) {
1103 if constexpr(FixedWidthIntegral<U>) {
1104 // Value type is fixed-width. T must be fixed-width too (due to static_asserts above).
1105 // A stack variable is fine/fast here.
1106 U total = total_scaling_factor + value;
1107
1108 // Run the callback.
1109 if (callback(total, my_tls) == ForEachSignal::BREAK) {
1110 a_signal.store(ForEachSignal::BREAK, std::memory_order_relaxed);
1111 return ForEachSignal::BREAK;
1112 }
1113 return ForEachSignal::CONTINUE;
1114 } else if constexpr(GMPIntegral<U>) {
1115 // Value type is MPZ. T might be MPZ or fixed-width.
1116 // Use a thread-local for MPZ to avoid allocs on return value (total).
1117 static thread_local mpz_class total;
1118
1119 // Call the correct mpz_add_*() based on tree type T.
1120 if constexpr(FixedWidthIntegral<T>) {
1121 // Tree is fixed width. Juggle according to bit width since mpz_add_ui() has a 64-bit limit.
1122 if constexpr(sizeof(T) * 8 <= 64) {
1123 // Type is small enough to fix into mpz_add_ui().
1124 mpz_add_ui(total.get_mpz_t(), total_scaling_factor.get_mpz_t(), value);
1125 } else if constexpr(sizeof(T) * 8 == 128) {
1126 // Type is 128-bit. Must convert to GMP. Use another TLS to avoid alloc.
1127 static thread_local mpz_class value_tmp_mpz;
1128 Int128::uint128_to_mpz(value, value_tmp_mpz);
1129 mpz_add(total.get_mpz_t(), total_scaling_factor.get_mpz_t(), value_tmp_mpz.get_mpz_t());
1130 }
1131 } else if constexpr(GMPIntegral<T>) {
1132 // Tree is GMP type as well.
1133 // Simply use mpz_add directly.
1134 mpz_add(total.get_mpz_t(), total_scaling_factor.get_mpz_t(), value.get_mpz_t());
1135 }
1136
1137 // Run the callback.
1138 if (callback(total, my_tls) == ForEachSignal::BREAK) {
1139 a_signal.store(ForEachSignal::BREAK, std::memory_order_relaxed);
1140 return ForEachSignal::BREAK;
1141 }
1142 return ForEachSignal::CONTINUE;
1143 }
1144 });
1145
1146 // Bump the multiplier for the next loop.
1147 multiplier++;
1148 }
1149 }
