340 static bool st_equal(const BinaryTree<T, TreeType>& first, const BinaryTree<T, TreeType>& second, std::string* err = nullptr) {
341 // Common to all trees.
342 EqualityHelper eq(err);
343 eq.set_category("BinaryTree");
344
345 // Root node
346 if (! Node<T>::st_equal(first.get_root_node(), second.get_root_node(), err)) {
347 return eq.fail("Root nodes don't match.");
348 }
349
350 // Level count
351 if (! eq.equal(first.get_level_count(), second.get_level_count())) {
352 return eq.fail("Level counts differ");
353 }
354
355 // Flags
356 if (! eq.equal(first.is_preserving_ancestors(), second.is_preserving_ancestors())) {
357 return eq.fail("Is preserving ancestors mismatch");
358 }
359 if (! eq.equal(first.is_verifying_non_hwm_nodes(), second.is_verifying_non_hwm_nodes())) {
360 return eq.fail("Is verifying hwm nodes mismatch");
361 }
362 if (! eq.equal(first.is_initialized(), second.is_initialized())) {
363 return eq.fail("Is initialized mismatch");
364 }
365
366 // Coverage
367 // Ensure the same size because it's cheap.
368 if (! eq.equal(first.get_coverage_map().size(), second.get_coverage_map().size())) {
369 return eq.fail("Coverage map size differs");
370 }
371 // Compare each level+coverage combo.
372 for (const auto& [level, f_coverage] : first.get_coverage_map()) {
373 if (! second.get_coverage_map().contains(level)) {
374 return eq.fail("Coverage map in 'first' has level " + to_string_any(level) + " but 'second doesn't.");
375 }
376 const BinaryTreeCoverage<T>& s_coverage = second.get_coverage_map().at(level);
377 if (! f_coverage.equal(s_coverage, err)) {
378 return eq.fail("Coverage doesn't match");
379 }
380 }
381
382 // Ancestors
383 // For this to work, they must be a consistent order, which the impls guarantee on add_level();
384 const std::vector<Node<T>*>& f_ancestors = first.get_ancestors();
385 const std::vector<Node<T>*>& s_ancestors = second.get_ancestors();
386 if (! eq.equal(f_ancestors.size(), s_ancestors.size())) {
387 return eq.fail("Ancestor count mismatch");
388 }
389 for (size_t i = 0; i < f_ancestors.size(); i++) {
390 if (! Node<T>::st_equal(f_ancestors.at(i), s_ancestors.at(i), err)) {
391 return eq.fail("Ancestor node mismatch");
392 }
393 }
394
395 // Implementation-specific checks.
396 if (! TreeType::st_equal(first.get_impl(), second.get_impl(), err)) {
397 return eq.fail("Implementation-specific details mismatched");
398 }
399
400 // Everything is okay.
401 return true;
402 }

431 [[nodiscard]] bool serialize(std::ostream& out, std::string* err = nullptr) const {
432 StreamHelper sh(nullptr, &out, err);
433 sh.set_category("BinaryTree");
434
435 // Header
436 const BinaryTreeFileHeader header;
437 if (! sh.write_bytes(&(header.magic), header.MAGIC_SIZE)) {
438 return sh.fail("header magic string==" + std::string(header.magic));
439 }
440 if (! sh.serialize_integral(header.version)) {
441 return sh.fail("header version==" + to_string_any(header.version));
442 }
443
444 // Common Metadata
445 uint8_t tree_type = IsImplicitTree<TreeType> ? TreeTypeEnum::IMPLICIT : TreeTypeEnum::MATERIALIZED;
446 uint32_t level_count = get_level_count();
447 bool b_is_preserving_ancestors = is_preserving_ancestors();
448 bool b_is_verifying_non_hwm_nodes = is_verifying_non_hwm_nodes();
449 if (! sh.serialize_integral(tree_type)) {
450 return sh.fail("tree_type==" + to_string_any(tree_type));
451 }
452 if (! sh.serialize_integral(level_count)) {
453 return sh.fail("level_count==" + to_string_any(level_count));
454 }
455 if (! sh.serialize_bool(b_is_preserving_ancestors)) {
456 return sh.fail("b_is_preserving_ancestors==" + std::to_string(b_is_preserving_ancestors));
457 }
458 if (! sh.serialize_bool(b_is_verifying_non_hwm_nodes)) {
459 return sh.fail("b_is_verifying_non_hwm_nodes==" + std::to_string(b_is_verifying_non_hwm_nodes));
460 }
461 // The is_initialized must be true, so we ignore it.
462
463 // Root node
464 bool b_has_root_node = get_root_node() != nullptr;
465 if (! sh.serialize_bool(b_has_root_node)) {
466 return sh.fail("b_has_root_node");
467 }
468 if (b_has_root_node) {
469 if (! get_root_node()->serialize(out, err)) {
470 return sh.fail("root node");
471 }
472 }
473
474 // Coverage
475 // Write out how many entries exist so we know how many loops to make.
476 const std::unordered_map<level_t, BinaryTreeCoverage<T>>& coverage_map = _impl.get_coverage_map();
477 uint64_t u64_entry_count = coverage_map.size();
478 if (! sh.serialize_integral(u64_entry_count)) {
479 return sh.fail("coverage entry_count==" + to_string_any(u64_entry_count));
480 }
481 // Loop through them, emitting the level and then the coverage itself.
482 // Sort the levels to ensure consistent output. Nodes themselves are always ordered consistently.
483 std::vector<level_t> coverage_levels;
484 coverage_levels.reserve(coverage_map.size());
485 for (const auto& [level, _] : coverage_map) {
486 coverage_levels.push_back(level);
487 }
488 std::sort(coverage_levels.begin(), coverage_levels.end());
489 for (level_t level : coverage_levels) {
490 uint64_t u64_level = static_cast<uint64_t>(level);
491 const BinaryTreeCoverage<T>& coverage = coverage_map.at(level);
492 if (! sh.serialize_integral(u64_level)) {
493 return sh.fail("u64_level==" + to_string_any(u64_level));
494 }
495 if (! coverage.serialize(out, err)) {
496 return sh.fail("coverage on level " + to_string_any(level));
497 }
498 }
499
500 // Ancestors
501 // These are new-placement constructed (no children). Just store the value.
502 // First, write out how many there are.
503 const std::vector<Node<T>*>& ancestors = get_ancestors();
504 uint64_t u64_size = static_cast<uint64_t>(ancestors.size());
505 if (! sh.serialize_integral(u64_size)) {
506 return sh.fail("get_ancestors().size()==" + to_string_any(u64_size));
507 }
508 // Now loop through them.
509 for (const Node<T>* ancestor : ancestors) {
510 if (! ancestor->serialize(out, err)) {
511 return sh.fail("ancestor with value==" + to_string_any(ancestor->get_value()));
512 }
513 }
514
515 // Implementation-Specific Hook
516 if (! _impl.serialize(out, err)) {
517 return sh.fail("implementation-specific hook issues");
518 }
519
520 // All Good
521 return true;
522 }

536 [[nodiscard]] bool deserialize(std::istream& in, std::string* err = nullptr) {
537 StreamHelper sh(&in, nullptr, err);
538 sh.set_category("BinaryTree");
539 EqualityHelper eq(err);
540
541 // Reset this object.
542 _impl.reset();
543
544 // Header
545 const BinaryTreeFileHeader header;
546 char magic[header.MAGIC_SIZE];
547 if (! sh.read_bytes(&magic, header.MAGIC_SIZE)) {
548 return sh.fail("couldn't read header magic string");
549 }
550 if (! eq.equal(std::string(magic, header.MAGIC_SIZE), std::string(header.magic, header.MAGIC_SIZE))) {
551 return sh.fail("Magic in header doesn't match the known value. Wrong file?");
552 }
553 uint32_t version;
554 if (! sh.deserialize_integral(version)) {
555 return sh.fail("couldn't read header version");
556 }
557 if (! eq.equal(version, header.version)) {
558 return sh.fail("Version mismatch");
559 }
560
561 // Common Metadata
562 uint8_t tree_type;
563 uint32_t level_count;
564 bool b_is_preserving_ancestors;
565 bool b_is_verifying_non_hwm_nodes;
566 // Tree type. Has to match.
567 if (! sh.deserialize_integral(tree_type)) {
568 return sh.fail("couldn't read tree_type");
569 }
570 if (! eq.equal(tree_type, static_cast<uint8_t>(_impl.get_tree_type()))) {
571 return sh.fail("Tree types don't match");
572 }
573 // Level count. Type T needs to support this many levels.
574 if (! sh.deserialize_integral(level_count)) {
575 return sh.fail("couldn't read level_count");
576 }
577 if (! BinaryTreeMath<T>::st_level_will_fit(level_count)) {
578 std::string msg = "Level count found ("
579 + to_string_any(level_count)
580 + ") exceeds the amount permitted ("
581 + to_string_any(BinaryTreeMath<T>::st_max_level_of_type())
582 + ") by the type T you specified: "
583 + ABI::demangle(typeid(T).name())
584 + ".";
585 return sh.fail(msg);
586 }
587 _impl.set_level_count(level_count);
588 // Is preserving ancestors
589 if (! sh.deserialize_bool(b_is_preserving_ancestors)) {
590 return sh.fail("couldn't read is_preserving_ancestors");
591 }
592 _impl.set_is_preserving_ancestors(b_is_preserving_ancestors);
593 // Is verifying non-hwm nodes
594 if (! sh.deserialize_bool(b_is_verifying_non_hwm_nodes)) {
595 return sh.fail("couldn't read is_verifying_non_hwm_nodes");
596 }
597 if (b_is_verifying_non_hwm_nodes) {
598 enable_non_hwm_node_verification();
599 } else {
600 disable_non_hwm_node_verification();
601 }
602 // The is_initialized must be true, so we set it manually.
603 _impl.set_is_initialized(true);
604
605 // Root node
606 bool source_has_root_node;
607 if (! sh.deserialize_bool(source_has_root_node)) {
608 return sh.fail("has_root_node");
609 }
610 if (source_has_root_node) {
611 Node<T>*& root_node = _impl.get_root_node_rw();
612 // Root node can be null on *this* object, so make one for it to use.
613 if (root_node == nullptr) {
614 root_node = new Node<T>(BinaryTreeMath<T>::get_root_value());
615 }
616 T root_parent_v;
617 T root_hwm_ancestor_v;
618 uint8_t child_count;
619 if (! root_node->deserialize(in, root_parent_v, root_hwm_ancestor_v, child_count, err)) {
620 return sh.fail("root node");
621 }
622 if (root_parent_v != 0) {
623 return sh.fail("Root node's parent value must be 0 (absent), but it's: " + to_string_any(root_parent_v));
624 }
625 if (root_hwm_ancestor_v != 0) {
626 return sh.fail("Root node's hwm ancestor value must be 0 (absent), but it's: " + to_string_any(root_hwm_ancestor_v));
627 }
628 } else {
629 // The backed up file didn't have a root node... which means it was a 0-level tree :/
630 if (_impl.get_root_node() != nullptr) {
631 Node<T>*& root_node = _impl.get_root_node_rw();
632 delete root_node;
633 root_node = nullptr;
634 }
635 }
636
637 // Coverage
638 std::unordered_map<level_t, BinaryTreeCoverage<T>>& coverage_map = _impl.get_coverage_map_rw();
639 uint64_t u64_entry_count = 0;
640 if (! sh.deserialize_integral(u64_entry_count)) {
641 return sh.fail("couldn't get coverage's entry count");
642 }
643 for (uint64_t i = 0; i < u64_entry_count; i++) {
644 uint64_t u64_level;
645 if (! sh.deserialize_integral(u64_level)) {
646 return sh.fail("couldn't read coverage level for index i==" + to_string_any(i));
647 }
648 level_t level = static_cast<level_t>(u64_level);
649 BinaryTreeCoverage<T> coverage;
650 if (! coverage.deserialize(in, err)) {
651 return sh.fail("coverage on level " + to_string_any(level));
652 }
653 // Level and coverage are okay. Throw them in the map.
654 coverage_map[level] = coverage;
655 }
656
657 // Ancestors
658 // These are new-placement (no children). Slam them into the vector.
659 uint64_t u64_size = 0;
660 std::vector<Node<T>*>& ancestors_rw = _impl.get_ancestors_rw();
661 if (! sh.deserialize_integral(u64_size)) {
662 return sh.fail("couldn't read ancestors.size() count");
663 }
664 ancestors_rw.reserve(static_cast<size_t>(u64_size));
665 T parent_v;
666 T hwm_ancestor_v;
667 uint8_t child_count;
668 for (uint64_t i = 0; i < u64_size; i++) {
669 Node<T>* new_node = new Node<T>(T(1));
670 if (! new_node->deserialize(in, parent_v, hwm_ancestor_v, child_count, err)) {
671 return sh.fail("couldn't deserialize ancestor at index i==" + to_string_any(i));
672 }
673 ancestors_rw.push_back(new_node);
674 }
675
676 // Implementation-specific hook
677 if (! _impl.deserialize(in, err)) {
678 return sh.fail("implementation-specific hook issues");
679 }
680
681 // All good
682 return true;
683 }

700 bool save(const std::string& path, int compression_level = 22) {
701 // Sanity checks.
702 if (compression_level > 22) {
703 throw std::out_of_range("Zstd compression cannot exceed 22. You sent " + to_string_any(compression_level) + " to BinaryTree.save()");
704 }
705 if (compression_level < 0) {
706 throw std::out_of_range("While Zstd supports negative compression levels, we do not. You sent " + to_string_any(compression_level) + " to BinaryTree.save()");
707 }
708
709 // Setup some pointers to the file output and a potential zstd out.
710 std::ofstream file_out;
711 std::ostream* final_out = nullptr;
712 std::unique_ptr<zstd_ostream> zstd_out;
713
714 // Open the file stream.
715 const bool use_compression = compression_level > 0;
716 if (path == "-") {
717 // Write to stdout. Ensure it's not a terminal.
718 if (isatty(fileno(stdout))) {
719 throw std::runtime_error("Refusing to write binary data to a terminal. Redirect to a file, pipe, etc.");
720 }
721 final_out = &std::cout;
722 } else {
723 // Write to a normal file.
724 file_out.open(path, std::ios::binary);
725 if (!file_out) {
726 throw std::runtime_error("Failed to open file for writing");
727 }
728 final_out = &file_out;
729 }
730
731 // If compression is enabled, thread it into our zstd_ostream helper.
732 if (use_compression) {
733 zstd_out = std::make_unique<zstd_ostream>(*final_out, compression_level);
734 final_out = zstd_out.get();
735 }
736
737 // Perform the serialization and see if it worked.
738 std::string err;
739 bool ok = serialize(*final_out, &err);
740
741 // If compression is used, zstd requires finialization logic to finish a frame.
742 if (use_compression) {
743 zstd_out->finalize();
744 } else {
745 // We'll flush a raw file here just for good practice.
746 final_out->flush();
747 }
748 if (! ok) {
749 throw std::runtime_error("Failed to serialize data. Error chain is: " + err);
750 }
751
752 return ok && final_out->good();
753 }

762 bool load(const std::string& path) {
763 // Setup pointers to the file input and a potential zstd in.
764 std::ifstream file_in;
765 std::istream* final_in = nullptr;
766 std::unique_ptr<zstd_istream> zstd_in;
767
768 // Open the file stream.
769 if (path == "-") {
770 // Read from stdin.
771 final_in = &std::cin;
772 } else {
773 // Read from a normal file.
774 file_in.open(path, std::ios::binary);
775 if (!file_in) {
776 throw std::runtime_error("Failed to open file for reading");
777 }
778 final_in = &file_in;
779 }
780
781 // Check for Zstd compression. Then rewind our cursor.
782 std::string err;
783 uint32_t four_byte_magic;
784 {
785 StreamHelper sh(final_in, nullptr, &err);
786 std::streampos pos = final_in->tellg();
787 if (! sh.deserialize_integral(four_byte_magic)) {
788 throw std::runtime_error("Failed to deserialize first four bytes to detect Zstd compression. Error chain is: " + err);
789 }
790 final_in->clear();
791 final_in->seekg(pos);
792 }
793
794 // If we have Zstd compression, we need to link the file to our decompressor and pass on a different istream.
795 const bool is_zstd = four_byte_magic == ZSTD_MAGICNUMBER;
796 if (is_zstd) {
797 zstd_in = std::make_unique<zstd_istream>(*final_in);
798 final_in = zstd_in.get();
799 }
800
801 // Now we can deserialize and it'll only receive the raw data.
802 err.clear();
803 if (! deserialize(*final_in, &err)) {
804 throw std::runtime_error("Failed to deserialize data (zstd compressed=" + to_string_any(is_zstd) + "). Error chain is: " + err);
805 }
806
807 return true;
808 }
