44 template<size_t bits>
45 std::array<Stride, 1ULL << bits> build_stride_table() {
46 static_assert(bits <= MAX_STRIDE, "Stride bits exceed MAX_STRIDE");
47
48 // Build table.
49 const uint64_t table_size = uint64_t(1) << bits;
50 std::array<Stride, 1ULL << bits> stride_table{};
51
52 // With the table preallocated, it can use index-based iteration. Since it's looking at LSBs, we naturally get the order
53 // and can simply loop through them sequentially.
54 for (uint64_t i = 0; i < table_size; i++) {
55 uint64_t n = i;
56 Stride stride;
57 for (size_t step_count = 0; step_count < bits; step_count++) {
58 if (n & 1) {
59 stride.add = ((stride.add << 1) + stride.add) + (uint64_t(1) << stride.shift);
60 stride.multiply *= 3;
61 stride.shift++;
62 // Mimic the effects on n now.
63 n = (3 * n + 1) >> 1;
64 } else {
65 stride.shift++;
66 // Mimic the effects on n now.
67 n >>= 1;
68 }
69 }
70 // Calculate the number of bits required to apply this stride to a fixed-width type. Add one for the .add factor.
71 stride.bits_required = std::bit_width(stride.multiply - 1) + 1;
72 // Assign to the table.
73 stride_table[i] = stride;
74 }
75
76 return stride_table;
77 }

89 static_assert(bits <= MAX_STRIDE, "Stride bits exceed MAX_STRIDE");
90
91 // Build the table.
92 std::array<Stride, bits + 1> stride_table_of_ones{};
93
94 // Create the zero stride, which is no movement at all.
95 Stride stride;
96 stride_table_of_ones[0] = stride;
97
98 // Now loop over consecutive ones, which is simply applying the F step to each previous stride.
99 for (size_t i = 1; i <= bits; i++) {
100 stride.multiply = stride.multiply * 3;
101 stride.add = stride.add * 3 + (uint64_t(1) << stride.shift);
102 stride.shift++;
103
104 // Calculate the number of bits required to apply this stride to a fixed-width type. Add one for the .add factor.
105 stride.bits_required = std::bit_width(stride.multiply - 1) + 1;
106 // Assign to the table.
107 stride_table_of_ones[i] = stride;
108 }

156 if constexpr(FixedWidthIntegral<T>) {
157 // Basic arithemtic and masking is fine on this path.
158 const AffineStride::Stride& stride = get_stride(value);
159 value = ((value * stride.multiply) + stride.add) >> stride.shift;
160 } else if constexpr(GMPIntegral<T>) {
161 // GMP needs the UI extracted and cleaner calls to avoid temps.
162 const AffineStride::Stride& stride = get_stride(value);
163 mpz_mul_ui(value.get_mpz_t(), value.get_mpz_t(), stride.multiply);
164 mpz_add_ui(value.get_mpz_t(), value.get_mpz_t(), stride.add);
