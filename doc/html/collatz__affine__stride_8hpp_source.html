50 template<typename StrideType, size_t bits>
51 constexpr std::array<StrideType, 1ULL << bits> build_stride_table() {
52 if constexpr (std::same_as<StrideType, Stride>) {
53 static_assert(bits <= MAX_STRIDE, "Stride bits exceed MAX_STRIDE");
54 } else if constexpr (std::same_as<StrideType, LongStride>) {
55 static_assert(bits <= MAX_LONG_STRIDE, "Stride bits exceed MAX_LONG_STRIDE");
56 }
57
58 // Build table.
59 const uint64_t table_size = uint64_t(1) << bits;
60 std::array<StrideType, 1ULL << bits> stride_table{};
61
62 // With the table preallocated, it can use index-based iteration. Since it's looking at LSBs, we naturally get the order
63 // and can simply loop through them sequentially.
64 for (uint64_t i = 0; i < table_size; i++) {
65 uint64_t n = i;
66 StrideType stride;
67 for (size_t step_count = 0; step_count < bits; step_count++) {
68 if (n & 1) {
69 stride.add = ((stride.add << 1) + stride.add) + (uint64_t(1) << stride.shift);
70 stride.multiply *= 3;
71 stride.shift++;
72 stride.f_steps++;
73 // Mimic the effects on n now.
74 n = (3 * n + 1) >> 1;
75 } else {
76 stride.shift++;
77 // Mimic the effects on n now.
78 n >>= 1;
79 }
80 }
81 stride_table[i] = stride;
82 }
83
84 return stride_table;
85 }

110 template<AnySupportedIntegral T>
111 static inline void apply_stride(T& value) {
112 if constexpr(FixedWidthIntegral<T>) {
113 // Basic arithemtic and masking is fine on this path.
114 const AffineStride::Stride& stride = AffineStride::STRIDE_TABLE[value & AffineStride::STRIDE_MASK];
115 value = ((value * stride.multiply) + stride.add) >> stride.shift;
116 } else if constexpr(GMPIntegral<T>) {
117 // GMP needs the UI extracted and cleaner calls to avoid temps.
118 uint64_t u64_value = value.get_ui();
119 const AffineStride::Stride& stride = AffineStride::STRIDE_TABLE[u64_value & AffineStride::STRIDE_MASK];
120 mpz_mul_ui(value.get_mpz_t(), value.get_mpz_t(), stride.multiply);
121 mpz_add_ui(value.get_mpz_t(), value.get_mpz_t(), stride.add);
122 mpz_fdiv_q_2exp(value.get_mpz_t(), value.get_mpz_t(), stride.shift);
123 }
124 }
